### 试题一：谈判

> ![image.png](https://pub-ccd5622eabb54b4a81678003778ddce1.r2.dev/PicGo/335b214b790e52c75408d69b35850bd82dc7d5249f44558f5224fda4c788c3f7.png)

【思路分析】

> 解析：运用贪心的思想，每次选两个人数最少的部落谈判，从局部最优达到全局最优。最后剩一个部落结束。输出结果。
> ![144fa499943162c0f5dad0cb6817bac5.jpg](https://pub-ccd5622eabb54b4a81678003778ddce1.r2.dev/PicGo/f46a0b98b4f1fd307eaa09d441141aa665fc754e70911901214f5e3ca2e208f0.jpg)

```java
import java.io.BufferedInputStream;
import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStream;
import java.io.InputStreamReader;
import java.io.StreamTokenizer;
import java.util.ArrayList;
import java.util.Collections;
import java.util.HashMap;
import java.util.Map;
import java.util.Scanner;
import java.util.StringTokenizer;

public class Main {
	public static void main(String[] args) {
		int n = nextInt();
		ArrayList<Integer> list = new ArrayList<>();

		for (int i = 0; i < n; i++) {
			list.add(nextInt());
		}
		int p = list.size();
		Collections.sort(list);

		long sum = 0;
		while (p > 1) {
			int a = list.remove(0);
            int b = list.remove(0);
            int t = a + b;
			sum += t;
			list.remove(0);
			list.add(t);
			Collections.sort(list);
			p--;
		}
		System.out.println(sum);

	}

	static BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));

	static StringTokenizer token = new StringTokenizer("");

	static String next() {
		while (!token.hasMoreTokens()) {
			try {
				token = new StringTokenizer(reader.readLine());
			} catch (IOException e) {
				// TODO Auto-generated catch block
				e.printStackTrace();
			}
		}
		return token.nextToken();
	}

	static int nextInt() {
		return Integer.parseInt(next());
	}

}
```

### 试题二：重新排序

> ![image.png](https://pub-ccd5622eabb54b4a81678003778ddce1.r2.dev/PicGo/cc7a767f2a7c6549c65a4ab1e87f13f735d544ddbcfafcdd825c326d798a408a.png)

【思路分析】

> 使用差分算法记录每个数的累加次数，运用贪心算法，使得出现次数最多的数字最大，从局部最优做全局最优。
> ![b6180cad5e383b1b7c3f2f721506f520.jpg](https://pub-ccd5622eabb54b4a81678003778ddce1.r2.dev/PicGo/44d03ba0e8f3d985b2cfe607fbafd442e58a0188acc32a9f608d796566708fed.jpg)

```java
import java.io.BufferedInputStream;
import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStream;
import java.io.InputStreamReader;
import java.io.StreamTokenizer;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.Collections;
import java.util.HashMap;
import java.util.Map;
import java.util.StringTokenizer;

public class Main {
    public static void main(String[] args) {
        int n = nextInt();
        int[] a = new int[n+1];
        //long防止前缀和爆掉。一定要是long的
        long[] d = new long[n+2];
        //d初始时全是0
        for(int i = 1; i<=n; i++) {
            a[i] = nextInt();
        }
        int m = nextInt();
        for(int i = 0; i<m; i++) {
            int L = nextInt();
            int R = nextInt();
            d[L] +=1;
            d[R+1] -=1;
        }
        long old = 0;

        for(int i = 1; i<=n; i++) {
            d[i] +=d[i-1];
            // 需要添加long转换，就算old是long但是乘法优先级高于加法，再添加之前就溢出了
            old +=(long)(a[i] * d[i]);
        }
        //保证多余n+1项不会进入排序中，sort是左闭右开区间
        Arrays.sort(d,0,d.length-1);
        Arrays.sort(a);
        long cur = 0;
        for(int i = n; i>=1; i--) {
            cur += (long)(a[i] * d[i]);
        }
        System.out.println(cur - old);
    }

    static BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));

    static StringTokenizer token = new StringTokenizer("");

    static String next() {
        while (!token.hasMoreTokens()) {
            try {
                token = new StringTokenizer(reader.readLine());
            } catch (IOException e) {
                // TODO Auto-generated catch block
                e.printStackTrace();
            }
        }
        return token.nextToken();
    }

    static int nextInt() {
        return Integer.parseInt(next());
    }

}
```
