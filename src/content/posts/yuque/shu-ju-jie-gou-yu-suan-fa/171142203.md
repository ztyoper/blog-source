【思路分析】

> ![tmp53B8.png](https://pub-ccd5622eabb54b4a81678003778ddce1.r2.dev/PicGo/b2acfc429e45f3adbc7f8d60264d5f2fc0a5637968d6b9f13167ecf810333742.png)![tmp7DC7.png](https://pub-ccd5622eabb54b4a81678003778ddce1.r2.dev/PicGo/5a9ec00cc40264efb6e298607a696b533a9774b2587e5c9f3ee0119e7d2315c1.png)
> 我们最终是将最后两堆石子合并成一堆，因此是f[i][j] = min{f[i][k] + f[k+1][j] + sum(i,j)}
> ![tmp6758.png](https://pub-ccd5622eabb54b4a81678003778ddce1.r2.dev/PicGo/fa7a59a5e7314be177e59b4d96795280089e966f3a6099affd25b711b557d7af.png)![tmp94BD.png](https://pub-ccd5622eabb54b4a81678003778ddce1.r2.dev/PicGo/110e2cb9e3d21506344f52c25c84dddafdb1b90602d68f3f266ccfddca23aedf.png)
> ![52aa6dc874d62de820e41b39df89e25c.jpg](https://pub-ccd5622eabb54b4a81678003778ddce1.r2.dev/PicGo/448d4d2bd2b92e132598fce1d659640eb99f692847dab7d9de713dc7f0f128c8.jpg)
> ![tmp6CF4.png](https://pub-ccd5622eabb54b4a81678003778ddce1.r2.dev/PicGo/659e2abc848b2a04d620e62567b1457de605b50ff7881090cfbc7d598331488d.png)![906815bb6e13197afcd905a40011b7a8.jpg](https://pub-ccd5622eabb54b4a81678003778ddce1.r2.dev/PicGo/af1986702dd9f0a8155a62b3b633fc7c737b6c5041fe679380511f766e2db85c.jpg)

```java
import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.StringTokenizer;

public class Main {
	public static void main(String[] args) throws NumberFormatException, IOException {
		int n = nextInt();
		int[] a = new int[n + 1];
		int[][] f = new int[n + 1][n + 1];
		int[] sum = new int[n + 1];

		for (int i = 0; i <= n; i++) {
			Arrays.fill(f[i], Integer.MAX_VALUE);
		}
		for (int i = 1; i <= n; i++) {
			a[i] = nextInt();
			sum[i] = sum[i - 1] + a[i];// 前缀和
			f[i][i] = 0;// 长度为1的区间不需要合并石子，因此是0
		}

		for (int len = 2; len <= n; len++) {
			// 在长度确定的情况下，确定左端点。如果区间的长度（i+len-1）>n了，就不需要在迭代i了
			for (int i = 1; i + len - 1 <= n; i++) {
				// 确定右端点 j - i + 1 = len
				int j = i + len - 1;
				// k不能等于j，因为有k+1
				for (int k = i; k < j; k++) {
					f[i][j] = Math.min(f[i][j], f[i][k] + f[k + 1][j] + sum[j] - sum[i - 1]);
				}
			}
		}
		System.out.println(f[1][n]);
	}

	static BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));

	static StringTokenizer token = new StringTokenizer("");

	static String next() throws IOException {
		while (!token.hasMoreElements()) {
			token = new StringTokenizer(reader.readLine());
		}
		return token.nextToken();
	}

	static int nextInt() throws NumberFormatException, IOException {
		return Integer.parseInt(next());
	}
}
```

### 试题2：涂色

> ![image.png](https://pub-ccd5622eabb54b4a81678003778ddce1.r2.dev/PicGo/99fe3e6b9db70bb032c030a97101c8a16322dbc92c3521869a1818319498ed0f.png)

【思路分析】

> RGBGR共5个单位，可以分成前四个和后一个。那么f[1][5] = min{f[i][k] + f[k+1][j]} 涂色次数计算公式（状态转移公式）
> ![tmp566C.png](https://pub-ccd5622eabb54b4a81678003778ddce1.r2.dev/PicGo/c8d9abbe0322f405d9e485fe77094cc7c46f7c86fa8c1bd95a026f21de519c14.png)
> ![image.png](https://pub-ccd5622eabb54b4a81678003778ddce1.r2.dev/PicGo/7541c905979b9cde55011d0423bd05475cd632fe46fd1d92870d902193975ccd.png)![tmp79C.png](https://pub-ccd5622eabb54b4a81678003778ddce1.r2.dev/PicGo/ff021ce597163f11fa2580f5e55b790ef36b11e76845c713ccbff4ff09778793.png)

```java
import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.StringTokenizer;

public class Main {
	public static void main(String[] args) throws NumberFormatException, IOException {
		String string = reader.readLine();
		int n = string.length();
		char[] a = new char[n + 1];
		for (int i = 0; i < n; i++)
			a[i+1] = string.charAt(i);
		int[][] f = new int[n + 1][n + 1];

		for (int i = 0; i <= n; i++) {
			Arrays.fill(f[i], Integer.MAX_VALUE);
		}
		for (int i = 1; i <= n; i++) {
			f[i][i] = 1;// 一个单位长度的时候，不管要什么颜色，只需要涂色一次。如果是两个单位，两个单位的颜色可能不同
		}

		for (int len = 2; len <= n; len++) {
			// 在长度确定的情况下，确定左端点。如果区间的长度（i+len-1）>n了，就不需要在迭代i了
			for (int i = 1; i + len - 1 <= n; i++) {
				// 确定右端点 j - i + 1 = len
				int j = i + len - 1;
				if (a[i] == a[j]) {
					f[i][j] = Math.min(f[i][j - 1], f[i + 1][j]);
				} else {
					// k不能等于j，因为有k+1
					for (int k = i; k < j; k++) {
						f[i][j] = Math.min(f[i][j], f[i][k] + f[k + 1][j]);
					}
				}
			}
		}
		System.out.println(f[1][n]);
	}

	static BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));

	static StringTokenizer token = new StringTokenizer("");

	static String next() throws IOException {
		while (!token.hasMoreElements()) {
			token = new StringTokenizer(reader.readLine());
		}
		return token.nextToken();
	}

	static int nextInt() throws NumberFormatException, IOException {
		return Integer.parseInt(next());
	}
}
```

【思路分析】

> ![tmp494D.png](https://pub-ccd5622eabb54b4a81678003778ddce1.r2.dev/PicGo/3094f1dcf597afff7a7ad41e275784d093f39c2148b376117cace92f2a627181.png)两种方式使其变为回文串。但是我们不这样做。

> 如果D[·····]D，不知道[·····]是否为回文。那么花费就是：f[l][r] = f[l+1][r-1]
> ![image.png](https://pub-ccd5622eabb54b4a81678003778ddce1.r2.dev/PicGo/166ca84a9f3d9f7eaa37bb33b9df7d075d3785e52166930a9919819d48e44c59.png)![image.png](https://pub-ccd5622eabb54b4a81678003778ddce1.r2.dev/PicGo/b5551608386452a1e49d996ef0416c4ce29e500f9d118aeed77bb94212cab74c.png)

![tmpF9A0.png](https://pub-ccd5622eabb54b4a81678003778ddce1.r2.dev/PicGo/8e1a405c00f56b3c356123644b429fa44364eaa1ddc4a8a4d16c13bb08cb0fbb.png)

```java
import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.StringTokenizer;

public class Main {
	public static void main(String[] args) throws NumberFormatException, IOException {
		int n = nextInt();
		int m = nextInt();
		String string = reader.readLine();
		char[] a = new char[m + 1];
		for (int i = 0; i < m; i++)
			a[i + 1] = string.charAt(i);

		// 字母是按照顺序来的。
		int[] delete = new int[n];
		int[] insert = new int[n];
		int[][] f = new int[m + 1][m + 1];

		for (int i = 0; i < n; i++) {
			char ch = nextChar();
			insert[ch - 'a'] = nextInt();
			delete[ch - 'a'] = nextInt();
		}
		// 不需要Math.min(f[i][j],),也就不需要Integer.MAX_VALUE
		// 原因在后面
//		for (int i = 0; i <= n; i++) {
//			Arrays.fill(f[i], Integer.MAX_VALUE);
//		}
		for (int i = 1; i <= m; i++) {
			f[i][i] = 0;// 字符串为单个字母时，花费为0
		}

		for (int len = 2; len <= m; len++) {
			// 在长度确定的情况下，确定左端点。如果区间的长度（i+len-1）>n了，就不需要在迭代i了
			for (int i = 1; i + len - 1 <= m; i++) {
				// 确定右端点 j - i + 1 = len
				int j = i + len - 1;
				if (a[i] == a[j]) {
					// 添加判断的原因：i = 1, j = 2, f[i+1][j-1]会出错
					if (len == 2) {
						f[i][j] = 0;
					} else {
//						f[i][j] = Math.min(f[i][j], f[i+1][j-1]);
						f[i][j] = f[i + 1][j - 1];
					}
				} else {
					// 之前Math.min(f[i][j],)是因为一个区间可以划分为超过两种的子区间拼凑，
					// 现在每个区间只有两种情况。因此不需要Math.min(f[i][j],)
					f[i][j] = Math.min(f[i + 1][j] + Math.min(insert[a[i] - 'a'], delete[a[i] - 'a']),
							f[i][j - 1] + Math.min(insert[a[j] - 'a'], delete[a[j] - 'a']));
				}
			}
		}
		System.out.println(f[1][m]);
	}

	static BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));

	static StringTokenizer token = new StringTokenizer("");

	static String next() throws IOException {
		while (!token.hasMoreElements()) {
			token = new StringTokenizer(reader.readLine());
		}
		return token.nextToken();
	}

	static int nextInt() throws NumberFormatException, IOException {
		return Integer.parseInt(next());
	}

	static char nextChar() throws IOException {
		return next().charAt(0);
	}
}
```

### 试题1：能量项链

> ![image.png](https://pub-ccd5622eabb54b4a81678003778ddce1.r2.dev/PicGo/dbaa08df61d1c648a3334fd95aaf84a0efcdfb71ef2bae8a9524339671c4886a.png)

【思路分析】

```java
import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.StringTokenizer;

public class Main {
	public static void main(String[] args) throws NumberFormatException, IOException {
		int n = nextInt();
		int[] a = new int[2 * n + 1];
		int[][] f = new int[2 * n + 1][2 * n + 1];

		// 因为求最大，所以注释掉 ,f初始化全为0
//		for (int i = 0; i <= n; i++) {
//			Arrays.fill(f[i], Integer.MAX_VALUE);
//		}
		for (int i = 1; i <= n; i++) {
			a[i] = nextInt();
			// 复制区间
			a[i + n] = a[i];
		}

		for (int len = 2; len <= n; len++) {
			// 在长度确定的情况下，确定左端点。如果区间的长度（i+len-1）>n-1了，就不需要在迭代i了.
			// 2 3 5 10 2 3 5 10 , len=4时，i最大等于4，不能等于5，
			//如果等于5，就重复计算 2 3 5 10了。而且如果等于5 ， j+1最大就为2*n+1了，要越界。 
			for (int i = 1; i + len - 1 <= n * 2 - 1; i++) {
				// 确定右端点 j - i + 1 = len
				int j = i + len - 1;
				// k不能等于j，因为有k+1

				for (int k = i; k < j; k++) {
					//这道题就是这样写a[i] * a[k+1] * a[j+1]，但我也不知道为什么
					f[i][j] = Math.max(f[i][j], f[i][k] + f[k + 1][j] + a[i] * a[k+1] * a[j+1]);
				}
			}
		}
		int ans = 0;
		//遍历长度为n的所有f,当n = 6时， i = 1， f[1][6] ,i = 2, f[2][7], 不能直接f[1][12],因为根本没有12个珠子
		for(int i = 1; i<=n; i++) {
			ans = Math.max(ans, f[i][i+n-1]);
		}
		System.out.println(ans);
	}

	static BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));

	static StringTokenizer token = new StringTokenizer("");

	static String next() throws IOException {
		while (!token.hasMoreElements()) {
			token = new StringTokenizer(reader.readLine());
		}
		return token.nextToken();
	}

	static int nextInt() throws NumberFormatException, IOException {
		return Integer.parseInt(next());
	}
}
```
