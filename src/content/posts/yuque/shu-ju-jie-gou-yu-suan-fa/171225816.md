#### 解法一：朴素解法

【思路分析】

> ![ade64006387f7881d7118f99645b96c7.jpg](https://pub-ccd5622eabb54b4a81678003778ddce1.r2.dev/PicGo/8a583cc4a67b78ba51b170f0ecd8196402ca9231fa9e4ec6c1524ad008de73e5.jpg)
> 右边集合要判断成功，才有右边集合。不然没有右边集合。

```java
import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.StringTokenizer;

public class Main {
	public static void main(String[] args) throws Exception{
		int total = nextInt();
		int n = nextInt();
		//我们要有i-1的迭代，因此需要索引从1开始
		int[] w = new int[n+1];// weight
		int[] v = new int[n+1];// value
		int[][] f = new int[n+1][total+1];
		for(int i = 1; i<=n; i++) {
			w[i] = nextInt();
			v[i] = nextInt();
			
		}
		//先遍历背包，再遍历物品
		for(int i = 1; i<=n ;i++) {
			for(int j = 1; j<=total; j++) {
				//因为是求最值，因此直接赋值左半边的子集.
				
				f[i][j] = f[i-1][j];
				//如果判断失败，那么f[i][j]就没有右边集合， j不能从w[i]开始，因为需要 f[i][j] = f[i-1][j]进行赋值。
				if(j>=w[i]) f[i][j] = Math.max(f[i][j], f[i-1][j-w[i]] + v[i]);
			}
		}
		System.out.println(f[n][total]);
		
	}
	static BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
	static StringTokenizer token = new StringTokenizer("");
	static String next() throws IOException {
		while(!token.hasMoreElements()) {
			token = new StringTokenizer(reader.readLine());
		}
		return token.nextToken();
	}
	static int nextInt() throws NumberFormatException, IOException {
		return Integer.parseInt(next());
	}
}
```

#### 解法二：优化解法

【思路分析】

> ![ade64006387f7881d7118f99645b96c7.jpg](https://pub-ccd5622eabb54b4a81678003778ddce1.r2.dev/PicGo/8a583cc4a67b78ba51b170f0ecd8196402ca9231fa9e4ec6c1524ad008de73e5.jpg)
> 因为f(i,j)只涉及到f(i-1,j) , f(i-1, j-w[i])。没有涉及到f(i-2,) f(i-3)，因此只需要一层循环进行滚动迭代，只使用上一层的结果。j要么用自己，要么用比自己小的数。因此可以只使用一维，这是空间上的优化。滚动数组进行空间优化。

```java
import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.StringTokenizer;

public class Main {
	public static void main(String[] args) throws Exception{
		int total = nextInt();
		int n = nextInt();
		//我们要有i-1的迭代，因此需要索引从1开始
		int[] w = new int[n+1];// weight
		int[] v = new int[n+1];// value
		int[] f = new int[total+1];
		for(int i = 1; i<=n; i++) {
			w[i] = nextInt();
			v[i] = nextInt();
			
		}
		//先遍历背包，再遍历物品
		for(int i = 1; i<=n ;i++) {
			for(int j = total; j>=1; j--) {
				
				f[j] = f[j];
				//当i = 1时，我们初始化了f数组
				//当i = 2时，此时我们使用f[j] = f[j]就相当于 用上一层初始化的结果来给这一层赋值，也就是等价于f[i][j] = f[i-1][j]
				if(j>=w[i]) f[j] = Math.max(f[j], f[j-w[i]] + v[i]);
				//内层循环从大到小
				//当i = 1时，我们初始化了f数组
				//f[j]会先算出来，f[j-w[i]]会后算出来。
				//当i = 2时，我们再次初始化了f数组
				//f[j]会先算出来，f[j-w[i]]会后算出来。正是f[j-w[i]]后算出来，因此目前赋值f[j]时f[j-w[i]]的值是上一层算出来的，f[j-w[i]]仍然是i-1 层的值。
				//上式子就等价为f[i][j] = Math.max(f[i][j], f[i-1][j-w[i]]+v[i])
			}
		}
		System.out.println(f[total]);
		
	}
	static BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
	static StringTokenizer token = new StringTokenizer("");
	static String next() throws IOException {
		while(!token.hasMoreElements()) {
			token = new StringTokenizer(reader.readLine());
		}
		return token.nextToken();
	}
	static int nextInt() throws NumberFormatException, IOException {
		return Integer.parseInt(next());
	}
}
```
