# 反射实现GetTime

### <span style="color: var(--cm-text-color-regular)">GetTime</span>

> <span style="color: var(--cm-text-color-regular)">GetTime获取方法执行时间，静态方法和非静态方法都可以计算时间</span>

```java
/**
 * @author ztyOperation
 * @create 2023-02-19 10:38
 */
public class GetTime {
    public static Object invoke(Class<?> clazz,String methodName,Object... params){//调用类的静态方法，clazz就是该类的class
        try {
            //获取参数们的类型，从而精确找到相应的静态方法
            Class<?>[] paramTypes = new Class<?>[params.length];
            for(int i = 0; i< params.length; i++){
                paramTypes[i] = params[i].getClass();
            }
            Method method = clazz.getMethod(methodName,paramTypes);
            //开始计时
            long startTime = System.currentTimeMillis();
            Object result = method.invoke(null, params);//静态的方法,obj直接为null
            //结束计时
            long endTime = System.currentTimeMillis();
            long time = endTime-startTime;
            if(time>1000){//大于1000毫秒就是大于1秒
                time/=1000;
            }
            System.out.println("耗时: " + time);
            return result;
        } catch (NoSuchMethodException e) {
            throw new RuntimeException(e);
        } catch (IllegalAccessException e) {
            throw new RuntimeException(e);
        } catch (InvocationTargetException e) {
            throw new RuntimeException(e);
        }
    }
    public static Object invoke(Object proxy, String methodName, Object... args) {//调用类的非静态方法，proxy就是该类的实例对象

        Object result = null;
        try {
            //获取参数们的类型，从而精确找到相应的方法
            Class<?>[] paramTypes = new Class<?>[args.length];
            for(int i = 0; i< args.length; i++){
                paramTypes[i] = args[i].getClass();
            }
            Method method = proxy.getClass().getDeclaredMethod(methodName, paramTypes);
            long startTime = System.currentTimeMillis();
            //开始计时
            result = method.invoke(proxy,args);//静态的方法,obj直接为null
            //结束计时
            long endTime = System.currentTimeMillis();
            long time = endTime-startTime;
            System.out.println(methodName+ " waste time: " + time+"ms");
//			if(time>1000){//大于1000毫秒就是大于1秒
//			    time/=1000;
//			    System.out.println("waste time: " + time+"s");
//			}
//			else System.out.println("waste time: " + time+"ms");
        } catch (NoSuchMethodException | SecurityException | IllegalAccessException | InvocationTargetException e) {
            e.printStackTrace();
        }

        return result;
    }
}
```
