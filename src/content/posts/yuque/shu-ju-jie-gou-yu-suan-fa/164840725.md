```java
import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;
import java.util.StringTokenizer;

public class Main {
	static List<Integer>[] list;

	public static void main(String[] args) {
		int n = nextInt();
		list = new ArrayList[n+1];
		// 1~n个结点,不过建议将list[0]也分配空间。防止dfs()有人传入0,0,导致空指针。
		for (int i = 0; i <= n; i++) {
			list[i] = new ArrayList<>();
		}
		// 加入n-1条边，这种方法的好处是，在调用dfs（）时，传入不同的结点值，该结点值就可以作为整颗树的根结点。
		for (int i = 0; i < n - 1; i++) {
			int x = nextInt();
			int y = nextInt();
			// x,y结点相互连接
			list[x].add(y);
			list[y].add(x);
		}
		int[] dep = new int[n+1];
		//因为性质2：直径端点:x,y在任意一个结点作为根结点时，其中一个或两个是最深的叶子结点
		//我们通过方法进行初始化所有结点的深度后，获取一个最深的叶子结点作为x结点
		dfs(1,0,dep);
		int max = 0;
		int x = 0;
		for(int i =1 ; i<=n; i++) {
			if(max < dep[i]) {
				max = dep[i];
				x = i;
			}
		}
		Arrays.fill(dep, 0);
		//这里的0表示虚拟父节点
		dfs(x,0, dep);
		Arrays.sort(dep);
		//最深的深度就是树的直径
		System.out.println(dep[n]);
	}
    //这个方法最神奇的是第一次调用的x是整颗树的根结点，但是后续递归进去就不是整颗树的根结点了。
	// 我们通过该方法，找到curNode作为整颗树的根结点时得到所以子结点的dep,也就是深度
	// 初始dep[fatherNode] = 0，fatherNode开始是虚拟的。因此根结点的深度我们自定义为1
	public static void dfs(int curNode, int fatherNode, int[] dep) {
		dep[curNode] = dep[fatherNode] + 1;

		for (int y : list[curNode]) {
			// 当与curNode连接的结点y是fatherNode，则跳过
			if (y == fatherNode)
				continue;
			dfs(y, curNode, dep);
		}

	}

			}
		}

	}

	static int nextInt() {
		return Integer.parseInt(next());
	}

}
```

```java
/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
 // 看官解
class Solution {
    int ans;
    public int diameterOfBinaryTree(TreeNode root) {
        ans = 1;
        depth(root);
        return ans - 1;
    }
    public int depth(TreeNode node) {
        if (node == null) {
            return 0; // 访问到空节点了，返回0
        }
        int L = depth(node.left); // 左儿子为根的子树的深度
        int R = depth(node.right); // 右儿子为根的子树的深度
        ans = Math.max(ans, L+R+1); // 计算d_node即L+R+1 并更新ans
        return Math.max(L, R) + 1; // 返回该节点为根的子树的深度
    }
}
```

## 4、树的重心

![image.png](https://pub-ccd5622eabb54b4a81678003778ddce1.r2.dev/PicGo/1cbe04164d45baa322c34562713c245101726e39047b88267e36650716f26033.png)

> 1. 删除结点2后，出现3个联通块。联通块的大小是指结点数。{4，3}，{6}，{1，5}，因此联通块最大值为2。这是最小的。
> 2. 删除结点4后，出现2个联通块。联通块的大小是指结点数。{3}，{1，2，6，5}，因此联通块最大值为4。因此不是重心。
> 3. 等价于2作为根结点时，子树{4，3} ，{6}， {1，5}的三个子树大小2，1，2，这时最大子树的大小最小。
> 4. 上述的注意是指，去掉结点2和原先以2为根结点的子树{2，4，3，6}后，剩余部分{1，5}也是结点2的子树（这是变换根结点1-&gt;根结点2后的）。

### 2、求树的重心

![tmpB11D.png](https://pub-ccd5622eabb54b4a81678003778ddce1.r2.dev/PicGo/16e9aba283c6c01bc9c75bb6e26d09e89ce349f8b2b0ee4a6b92eea225ae566a.png)

> 1. sz[x] = 1表示包括子树根结点本身，开始x作为整颗树的根结点时，sz[x]最终会是n.
> 2. sz[x]表示结点x在不修改根结点的原来情况下，作为根结点的子树大小（结点数）。dfs(y,x)相当于初始化了x的子结点y的子树大小。而sz[x] +=sz[y]表示加入子树的结点数（大小）
> 3. mss[x] = max(mss[x], sz[y])：查找x作为子树根结点时，它下面的子树的大小中最大值。
> 4. n - sz[x]表示当让结点x作为整颗树的根结点时（有向边那边没有算上子树），（此时已经不是第一次调用dfs()方法的x结点了，而是递归进去的。因此x是子树的根结点。剩余的一颗没遍历到的子树，因为上面没有涉及到x结点上面的其他结点）剩余部分的大小（结点数）。因为要保证所有子树的大小&lt;=n/2。
> 5. res_add（x）表示不止一个重心。
> 6. 总之，这个代码与重心概念处的 **注意** 是相关的。

![image.png](https://pub-ccd5622eabb54b4a81678003778ddce1.r2.dev/PicGo/f33cd932c0e4a717dab3cd5125185a87974ad649658dd35da89519171208d084.png)

> 输入：
> 8
> 1 2
> 2 4
> 4 3
> 2 6
> 1 5
> 5 8
> 5 7

> 输出:
> [1,2]或者[2,1]

```java
import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;
import java.util.StringTokenizer;

public class Main {
	static List<Integer>[] list;
	static List<Integer> res;//保存重心
	static int[] sz;//sz[x]表示结点x作为子树根结点时，该子树的大小（结点数）
	static int[] mss;//mss[x]表示结点x作为整颗树根结点时，所有子树的最大值（子树中最大结点数）
	static int n = 0;
	public static void main(String[] args) {
		n = nextInt();
		list = new ArrayList[n+1];
		sz = new int[n+1];
		mss = new int[n+1];
		res = new ArrayList<>();
		// 1~n个结点。不过建议将list[0]也分配空间。防止dfs()有人传入0,0,导致空指针。
		for (int i = 0; i <= n; i++) {
			list[i] = new ArrayList<>();
		}
		// 加入n-1条边
		for (int i = 0; i < n - 1; i++) {
			int x = nextInt();
			int y = nextInt();
			// 既然图中没有让x,y结点相互连接，那我们就相互连接，这样求重心，就不用每一个节点带入来计算了。 
			list[x].add(y);
			list[y].add(x);
		}

//		for(int i = 1; i<=n; i++) {
//			
//			dfs(i,0);
//		}
		
		dfs(5,0);
		System.out.println(res.toString());

	}
    //这个方法最神奇的是第一次调用的x是整颗树的根结点，但是后续递归进去就不是整颗树的根结点了。
	// 我们通过该方法，找到x作为整颗树根结点时得到所以子结点的mss和sz,来求得所有重心
	// 初始dep[fatherNode] = 0，fatherNode开始是虚拟的。
	public static void dfs(int x, int fa) {
		sz[x] = 1; mss[x] = 0;

		for (int y : list[x]) {
			// 当与curNode连接的结点y是fatherNode，则跳过
			if (y == fa)
				continue;
            //初始化x结点连接的y结点为根结点的子树的大小。
			dfs(y, x);
			
			
			sz[x] +=sz[y];
			//解决x作为整颗树的根结点时的，那些子树的最大大小
			mss[x] = Math.max(mss[x], sz[y]);
			
		}
		//解决x作为整颗树的根结点时，对应的递归到叶子结点，就会直接跳出for循环。因为只有父节点与其相连。
		//这时，该叶子结点的子树大小（结点数）就是n-sz[x]
        //又或者整颗树的根结点x连接的结点y进行调用dfs()方法时，它就会忽略与父结点x连接的“子树”。这时我们也要进行判断。
        //一句话，第一次调用dfs()的结点x可以不要下面n-sz[x]的比较。但是其他结点需要。
		mss[x] = Math.max(mss[x], n-sz[x]);
		if(mss[x]<=n/2 ) res.add(x);

	}

	static BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));

	static StringTokenizer token = new StringTokenizer("");

	static String next() {
		while (!token.hasMoreTokens()) {
			try {
				token = new StringTokenizer(reader.readLine());
			} catch (IOException e) {
				// TODO Auto-generated catch block
				e.printStackTrace();
			}
		}
		return token.nextToken();
	}

	static int nextInt() {
		return Integer.parseInt(next());
	}

}
```

## 5、倍增LCA

### 1、什么是LCA

![image.png](https://pub-ccd5622eabb54b4a81678003778ddce1.r2.dev/PicGo/ac8d859631f914989dc6ad6101232a388d1fccf32d6a986109ad160150827029.png)

> 1. 如果x = 4, y = 9。dep[4]&lt;dep[9]，则y上跳，变为dep[4] &lt; dep[8] 一直上跳，知道dep[4] == dep[7]，但是x = 4, y = 7(此时)，x,y同时向上跳，x = 2, y = 3 再向上跳，x =1 , y = 1,返回LCA(4,9) = 1
> 2. 如果是一条链，则查找LCA就是O(n)，如果要查找m次，则是O(n*m)，这样的话，将会超时。

### 试题一：最近公共祖先LCA查询

> ![image.png](https://pub-ccd5622eabb54b4a81678003778ddce1.r2.dev/PicGo/2c392e5d7800196f35785edc59e60dffa8d93e0d215fe48df650c839c39b3ce4.png)

```java
import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;
import java.util.StringTokenizer;

public class Main {
	static List<Integer>[] list;
	static int[] dep;//保存结点的深度，如果dfs(x)传入的x不一样，dep的初始化也不一样
	static int[][] f;//f[i][j] 表示结点i向上跳2^j个结点后对应的结点编号
	public static void main(String[] args) {
		int n = nextInt();
        //21或下面所有的20可以使用(int)(Math.log(n)/Math.log(2));来替换。
		f = new int[n+1][21];
		list = new ArrayList[n+1];
		dep = new int[n+1];
		for(int i =0; i<=n; i++) {
			list[i] = new ArrayList<>();
		}
		//存储n-1条边
		for(int i =0; i<n-1; i++) {
			//结点x,y之间有连接
			int x = nextInt();
			int y = nextInt();
			//该题导入边时是从上到下。
			//因此可以少调价list[y].add(x)，后面也不用进行y == fa判断了。当然，加了也不会错
			list[x].add(y);
			list[y].add(x);
		}
		//初始化f[][]数组
		dfs(1,0);
		int p = nextInt();
		while(p-->0) {
			int x = nextInt();
			int y = nextInt();
			System.out.println(lca(x,y));

		}
	}
	//输入两个结点的编号x,y，找到lca(x,y)的最近公共祖先
	public static int lca(int x, int y) {
		//先将结点x转换为dep更深的那个结点。
		if(dep[x] < dep[y]) {
			int t = x;
			x = y;
			y = t;
		}
		//该方法能使dep[x] = dep[y];
		//让结点x向上跳，使用贪心的思想，先跳大的。如果比dep[y]浅了。就逐步减小向上跳的结点个数2^i。
		//例如：隔三层结点，就是跳 2+1.隔10层就是跳8+2
		//争取让x跳后，dep[x]大于dep[y]，f[x][i]结果结点在y结点下方，但是接近dep[y]。
		//如果正序遍历，如果只隔2个结点。你先f[x][2^0]跳一个结点，之后就不满足f[x][2^i]就不能跳了。这时不能保证dep[x] = dep[y]
		for(int i = 20; i>=0; i--) {
			if(dep[f[x][i]] >= dep[y])
				x = f[x][i];
		}
		//可能一次就跳到与y相同的结点了。也就是dep[x]==dep[y]，
		//的同时f[x][i]的结果结点编号与y结点编号相同。
		if(x == y)
			return x;
		//如果x,y只是在同一层，但不是相同结点，那么就要x,y同时向上跳。找到公共结点。
		//因为是找最近的。因此我们的判断是不相等就跳。
		//举例：同时向上跳3层就相等了。那么你向上跳3层及以上，都会时f[x][i] == f[y][i]，但不能保证最近
		//由此，我们跳f[x][1]两层，就差一层相等，此时退出循环
		//举例：同时向上跳4层就相等了。那么你向上跳4层及以上，都会时f[x][i] == f[y][i]，但不能保证最近
		//由此，我们跳f[x][1]两层，然后f[x][0]再跳一层，就差一层相等LCA(x,y)的下方，此时退出循环
		for(int i = 20; i>= 0; i--) {
			if(f[x][i] != f[y][i]) {
				x = f[x][i];
				y = f[y][i];
			}
		}
		
		
		return f[x][0];//或者f[y][0]也行，反正到这里x,y都在LCA(x,y)下面一层
	}
	//最初调用方法，fa是虚拟父节点0，x结点是作为整颗树的根结点。
	//但是递归调用dfs()方法的其他结点就不是作为整颗树的根结点了。
	public static void dfs(int x, int fa) {
		dep[x] = dep[fa] + 1;
		//结点x向上跳2^0个结点，就是父节点fa
		f[x][0] = fa;
		
		//这里最大20，也就是最大向上跳2^20次方个结点，也就是百万个结点
		for(int i = 1; i<=20; i++) {
			//倍增的状态转移。
			//x向上跳2^i个结点的结果结点 = x向上跳2^(i-1)个结点再将结果结点向上跳2^(i-1)个结点的结果结点
			//其实开始时，x作为整颗树的根结点不会向上跳了。
			//dep[x] - 2^j >=1 表示f[x][j]是有效的。我们定义整颗树的根结点的dep为1.我们在代码中没有用到。
			f[x][i] = f[f[x][i-1]][i-1];
		}
		
		for(int y : list[x]) {
			if(y == fa) {
				continue;
			}
```
