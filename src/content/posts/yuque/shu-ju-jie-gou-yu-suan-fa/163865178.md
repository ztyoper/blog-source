# 数位DP

> 数位DP往往都是这样的题型 给定一个闭区间[l,r] 上你求这个区间中满足某种条件的数的总数。所谓数位dp，就是对数位进行dp，也就是个位、十位等。
> 1. [l,r]转换为[0,r] - [0, l-1]
> 2. 状态转移方程：dp[i] = dp[i-1]，i是数字的位数。

> - 最常用的枚举方式是 **控制上界枚举**
> - 控制上界枚举就是要让正在枚举的这个数不能超过上界，
> - 我们常常利用一个bool变量limit来表明该数位前的其他数位是否恰好都处于最大状态。如果目前这个数不是这个位置最大的数字，那么可取的范围为0-9，否则范围将被限制在该数位在上界中的最大值。
>    - [1,223456],当第一位填0时，后面5位可以随便填，此时该数是5位的【后续也应该讨论4位的，3位的，2位的，1位的这些位有一个优势就是不会超区间】；当第一位填1时，后面5位可以随便填，此时该数是6位的；当第一位填2时，后面5位不能随便填，此时该数是6位的，因为可能会超区间，第二位只能填0 1 2 。后面递归讨论，就像第一位一样。
> - 注意， **前导零【以0为开头，如0123】** 要根据题目特殊处理
> 如果上述理解不了可以看例题注释

#### 解法一：现在的数位DP

【思路分析】

> ![4a9807a5a96c7b8e7dede1ab3d94f7f0.jpg](https://pub-ccd5622eabb54b4a81678003778ddce1.r2.dev/PicGo/9f10eff2233fb4d5ceb3ab08dfea99a1a98f60369a73a43656e7915fa511365f.jpg)
> |last - x| &gt;=2的判断需要放在内层循环的下面，不能是上面，例子:345，3和4不构成windy数，但是345只是范围的右边界，小于345的如315它是可以作为windy数的，因此判断需要放在内层循环下面。

```java
import java.util.Scanner;

public class Main {
	static int N = 15;
	static int[][] f = new int[N + 1][10];
	static Scanner scan = new Scanner(System.in);

	public static void main(String[] args) {
		init();
		while (scan.hasNextLine()) {
			String string = scan.nextLine();
			String[] split = string.split(" ");
			int l = Integer.parseInt(split[0]);
			int r = Integer.parseInt(split[1]);
			System.out.println(dfs(r) - dfs(l - 1));
		}

	}

	// 总结：f数组是存储了前导0的满足条件的数的集合（这个集合可能因为前导为0，导致满足条件后，后一位并不能填0~9，前导0时，像windy数就是只能填|k - 0| >=2的数，位数小于N的位数的数首位就不能遍历到1 2），但是是给中间0用的，不是给位数小的数的首位0用的。因为如果前一位是填0，位数小的数的首位判断的f[i][0]并不是将不同位数的情况加完。
	// 只有特殊情况下f[i][0]才能表示位数小于i,且最高位满足条件后可以表示0~9,这时会将"0"这个数算入其中。
    // 只有特殊情况下f[i][0]才能表示位数小于i,且最高位满足条件后可以表示1~9,这时不会将"0"这个数算入其中。
	static void init() {
		// 1、位数为1，f[i,j]表示位数共i个，并且最高位为j（0~9）的满足条件的数的集合。
		for (int i = 0; i <= 9; i++)
			f[1][i] = 1;
		// 2.位数大于1。f[i,j]表示位数共i个，并且最高位为j（0~9）的满足条件的数的集合。j是必须从0开始的，
		// 因为将一个数拆分后，“中间的0”是需要最高位是0，且某长度位数的满足条件的数集合的。记住不是给首位为0准备的。能给首位准备的都是特殊情况。
		for (int i = 2; i < N; i++) {
			for (int j = 0; j <= 9; j++) {
				for (int k = 0; k <= 9; k++) {
					if (abs(j - k) >= 2)
						f[i][j] += f[i - 1][k];
				}
			}
		}
	}

	static int dfs(int n) {
		// 若n等于0，那么nums长度为0，因此要特判一下。0不做为windy数。
		if (n == 0)
			return 0;

		// 拆分数字
		int[] nums = new int[N];
		int len = 0;
		while (n != 0) {
			nums[len++] = n % 10;
			n /= 10;
		}
		int res = 0;
		// 存储前缀信息：上一个数字
		// 设置-2,【1,456】,首位可以取1 2 3,然后在里面选满足条件的 |last - j| >=2
		int last = -2;
		for (int i = len - 1; i >= 0; i--) {
			int x = nums[i];

			// 首位不为0，中间的0可以出现
			// 1、位数相同，最高位不是a_n-i
			int start = i == len - 1 ? 1 : 0;
			for (int j = start; j < x; j++) {
				if (abs(last - j) >= 2)
					res += f[i + 1][j];
			}
			// 判断需要放在内层循环的下面，不能是上面，
			// 例子:345，3和4不构成满足条件的数，但是345只是范围的右边界，小于345的如315它是可以作为满足条件的数的，因此判断需要放在内层循环下面。
			// 必须要有退出命令，如果中间某一个数x不满足条件，就要退出循环，防止进入下一个循环
			if (abs(last - x) < 2)
				break;

			// 2、同数量级（位数相同）,但是最高位是a_n-i,保存该位数信息，进行树状递归
			last = x;

			// 3、当最后一位时，上述for循环填完完 0 ~ a0 中满足的数，最后填a0也就是N本身也是满足windy数的。
			// 前面有判断abs(last - x) < 2
			if (i == 0)
				res++;

		}

		// 3、比原位数少，原位数个数是3，那么就遍历位数个数是1~2且最高位1~9，
		for (int i = 1; i < len; i++) {
			for (int j = 1; j <= 9; j++) {
				res += f[i][j];
			}
		}
		return res;

	}

	static int abs(int a) {
		return a > 0 ? a : -a;
	}
}
```

#### 解法二：以往的数位DP

> 时间复杂度O(lgn)

【思路分析】

> - 相邻两个数可以转化为当前数与前一个数的比较。
> - 数位dp解决【不用记忆化搜索】这个题是在一个区间里找个数所以我们考虑用前缀和的思想只需找出1~m,1~n各自的个数再相减就行了
> - 设dp[i][j]为长度为 **i** 中最高位是 **j** 的windy数的个数
> - 方程dp[i][j]=sum(dp[i-1][k])其中abs(j-k)&gt;=2
> - 枚举数位转移即可

```java
package test;

import java.util.Scanner;
/*前置0看不懂没有关系*/
public class Main {
    static int[][] dp = new int[11][10];
    static Scanner scanner = new Scanner(System.in);

    public static void main(String[] args) {
        int l = scanner.nextInt();
        int r = scanner.nextInt();
        // 表示最高位是j，这里j可以取【0,9】
        for (int j = 0; j <= 9; j++) {
            dp[1][j] = 1;// 数的位数为1位时，因为1、会添加1-9所以就不用j而是1 
        }
        // i表示该数的位数，因为最大为2e9，也就是10位

        // 初始化出dp一位数到10位数所有的结果
        // 该数的位数为i位
        for (int i = 2; i <= 10; i++) {
            // 该数的最高位是j,包含前置0
            for (int j = 0; j <= 9; j++) {
                // 位数为i-1的数的最高位
                for (int k = 0; k <= 9; k++) {
                    // i=3,j = 2时，2xx = 20x + 24x + 25x + 26x + 27x + 28x + 29x = 0x + 4x + 5x + 6x + 7x + 8x + 9x
                    // i=2,j = 4时，4x  = 41 + 42 + 45 + 46 + 47 + 48 + 49 = 1 + 2 + 5 + 6 + 7 + 8 + 9
                    //在这里添加一个判断，后面solve中1、2、就不用添加判断语句
                    if (Math.abs(j - k) >= 2)
                        dp[i][j] += dp[i - 1][k];
                }
            }
        }
        System.out.println(solve(r) - solve(l - 1));
    }

    static int solve(int x) {	
        int[] dig = new int[11];//最高10位数
        int len = 0;
        int ans = 0;
        while(x>0) {
            dig[++len] = x%10;
            x/=10;
        }//右边界x：78359， 被拆分为9 5 3 8 7

        //1、位数小于给定右边界x的位数。如x=78359 所有位数少于5的数：1位，2位，3位，4位
        //1 2 3 4 5 6 7 8 9 1x 2x ... 9x 1xx 2xx .. 9xx 1xxx 2xxx .. 9xxx  
        for(int i = 1; i<len; i++) {
            //因为题目中保证没有前置0，这里最高位的值就从取到1开始
            for(int j = 1; j<=9; j++) {
                ans += dp[i][j];
            }
        }
        //2、位数与右边界x的位数相同 6位。但是最高位小于7,取到1xxxx 2xxxx 3xxxx 4xxxx 5xxxx 6xxxx 
        for(int j = 1; j<dig[len]; j++) {
            ans +=dp[len][j];
        }

        //3、位数与右边界x:78359的位数相同 6位。但是最高位取7,次高位取到0 1 2 3这样后面就可以随便取
        //78359-> 70xxx 71xxx 72xxx 73xxx 74xxx 75xxx ,不超过右边界且满足题目要求
        for(int i = len - 1; i>=1; i--) {
            //j等于8的话，78xxx有可能超出区间，因此不能直接加dp[7][8]
            for(int j = 0; j<=dig[i]-1; j++) {
                //不能取到76和77，所以要有该判断条件
                if(Math.abs(j  - dig[i+1]) >= 2)
                    ans +=dp[i][j];
            }
            //选循环完，再有该判断
            //如果7 8 之间差有2之内，就退出完成for循环，因为不能有78固定开头的情况
            if(Math.abs(dig[i+1] - dig[i]) < 2)break;

            if(i == 1)ans++;//像13这个数字，只能在这个地方添加。如果break没有发生。

        }

        return ans;
    }

}
```

#### 解法三：使用记忆搜索实现数位DP

> 记忆搜索是从dfs递归回溯优化而来。加入了记忆结果，减少递归次数。用他可以实现数位DP

> - dfs(len,last,limit,zero)表示总个数，len表示已经搜到了第len位（从高位到低位）
> - last表示上一位所填的数字，limit表示之前的数字是否已经紧贴上界，zero表示前置是否可以为零。
> - 显然，如果limit=0（之前数字未紧贴上界），那么与前一位数字差的绝对值不小于2的任意数字都可以填到当前位置上，不用考虑是否越界(高位都更小了，低位还管什么？)。否则如果limit=1,那么当前位置上的数不仅雷要保证差的绝对值-2，还要保证数字未超过此位置的上界

【思路分析】

> ![dd9672c98dd78cab49bb9bfaffbd5e45.jpg](https://pub-ccd5622eabb54b4a81678003778ddce1.r2.dev/PicGo/2722c8db6c2337656de175e0f5815dadfd77cf4710404b9c03f6428f3ffafcbb.jpg)

```java

import java.util.Arrays;
import java.util.Scanner;

public class Main {
	static Scanner scanner = new Scanner(System.in);
	static int[] dig = null;
	// len【位数】, last前一个数【因为该题要求当前与前者做比较】,
	// limit【1：限制当前位为该位能取到的上界一方超区间（比如:7834，第一位必须<=7，0：不限制可以[low,9]】，zero【前置是否可以为0
	// --> low是否可以取0】

    // 11是因为最大位数为10,12是因为我要使用11作为输入,11与任何一位数都是相差2之上，
    //2是因为输入1或者0两种数
	static long[][][][] dp = new long[11][12][2][2];
	public static void main(String[] args) {
		int l = scanner.nextInt();
		int r = scanner.nextInt();
		System.out.println(work(r) - work(l - 1));

	}

	public static long work(int x) {
		if(x < 0) return 0;
		if(x == 0 || x == 1)return x;
		// 初始化dp为-1,记忆化搜索的特色
		for (int i = 0; i < 11; i++) {
			for (int j = 0; j < 12; j++) {
				for (int k = 0; k < 2; k++) {
					Arrays.fill(dp[i][j][k], -1);
				}
			}
		}
		int len = 0;
		dig = new int[11];

		while (x > 0) {
			dig[++len] = x % 10;
			x /= 10;
		}
		// 假设x = 7834
		long ans = 0;
		// 1、谈论6位的数
		// 11表示，[0,9]都能与其abs>2,也就是什么数都能填，limit=1,只能填[0,7],zero= 1只能填[1,7].最终第一位只能填[1，7]
		ans = dfs(len, 11, 1, 0);
//		System.out.println("sdas"+ans);

		// 2、讨论5位，4位，3位，2位，1位的数
		for (int i = len - 1; i >= 1; i--) {
			ans += dfs(i, 11, 0, 0);
		}
		return ans;
	}

	public static long dfs(int len, int last, int limit, int zero) {
        //因为长度为1位时，有9种可能。1 2 3 4 5 6 7 8 9
		if (len == 0)
			return 1;// 与数位dp解法，一样都是1。
		// 记忆结果的返回
		if (dp[len][last][limit][zero] != -1)
			return dp[len][last][limit][zero];

		int up = limit == 1 ? dig[len] : 9;
		int low = zero == 1 ? 0 : 1;//可以换成low = zero ^ 1
		long res = 0;
		for (int i = low; i <= up; i++) {
			if (Math.abs(last - i) < 2)
				continue;
			res += dfs(len - 1, i, limit == 1 && (i == up) ? 1 : 0, 1);
		}
		dp[len][last][limit][zero] = res;

		return res;
	}
}
```

#### 解法一：数位DP

【思路分析】

> ![b9f4e663a7c53be7327b677043247fc4.jpg](https://pub-ccd5622eabb54b4a81678003778ddce1.r2.dev/PicGo/fe562ff8eb28d6ab91d4b6ee2b36fcee6d02e0a7ef293ed4224d717c6e303782.jpg)

```java
import java.util.Arrays;
import java.util.Scanner;

public class Main {
	static int N = 20;
	static int p = 7;
	// 总共i位数，最高位是j,各位数字之和模7的余数为a, 该数字模7的余数b , f存储满足条件的数的个数
	static long[][][][] f = new long[N][10][8][8];
	static int[] power7 = new int[N];
	static Scanner scan = new Scanner(System.in);

	public static void main(String[] args) {
		init();

		while (scan.hasNextLine()) {
			String string = scan.nextLine();
			String[] split = string.split(" ");
			long l = Long.parseLong(split[0]);
			long r = Long.parseLong(split[1]);
			if (l == 0 && r == 0)
				break;

			System.out.println(dfs(r) - dfs(l - 1));
		}

	}

	static void init() {
		// 位数为1
		// 0 模任何数都为0，因此可以作为满足条件的数
		for (int i = 0; i <= 9; i++)
				f[1][i][i % p][i % p] = 1;

		// 位数大于1
		long power = 10;
		for (int i = 2; i < N; i++, power *= 10) {
			for (int j = 0; j <= 9; j++) {
				for (int a = 0; a < p; a++) {
					for (int b = 0; b < p; b++) {
						for (int k = 0; k <= 9; k++) {
							f[i][j][a][b] += f[i - 1][k][mod(a - j)][mod(b - j * (power % 7))];
						}
					}
				}
			}
		}
		// power7[2]就是10^2 % 7
		power7[0] = 1;
		for (int i = 1; i < N; i++) {
			power7[i] = power7[i - 1] * 10 % 7;
		}
	}

	static long dfs(long n) {
		if (n == 0)
			return 1;

		int[] nums = new int[N];
		int len = 0;
		while (n != 0) {
			nums[len++] = (int) (n % 10);
			n /= 10;
		}
		long res = 0;

		// 前缀信息：存储前缀和
		long last_a = 0;
		// 前缀信息：存储前面的数的大小，如:1234 ，当i 循环第二次时，我们要存储last_b = 12
		long last_b = 0;

		for (int i = len - 1; i >= 0; i--) {
			int x = nums[i];
			int start = i == len - 1 ? 1 : 0;
			// 因为前置0，不影响位数小于i的数的最高位的取值，因此j可以优化从0开始。不用管前置0
			for (int j = start; j < x; j++) {
//				System.out.println(mod(-last_a));
//				System.out.println(mod(-last_b * power7[i + 1]));
				res += f[i + 1][j][mod(-last_a)][mod(-last_b * power7[i + 1])];
			}
			// 没有退出判断，因为不遍历到最后一个位，我们并不知道是否可以整除

			// 进行树状递归
			last_a += x;
			last_b = last_b * 10 + x;

			// N本身可以被整除, last_a已经被+=x了, last_b 已经变为N了， 
			if (i == 0 && mod(last_a) == 0 && mod(last_b) == 0)
				res++;
		}
		// 位数小于i,且最高为是0~9并且，余数为0的数集合。
		for (int i = 1; i < len; i++) {
			for (int j = 1; j <= 9; j++) {
				// 这个地方填0，因为没有前缀，需要位数为i，且最高位是j，且余数是0
				res += f[i][j][0][0];
			}
		}

		// 0算满足条件的数
		res++;

		return res;
	}

	// mod p 结果必定在[1, p-1]
	static int mod(long x) {
		return (int) ((x % p + p) % p);
	}

}
```

#### 解法二：记忆化搜索

【思路分析】

> 假设该数为x，对应数位之和为s。我们使用dp[len][x%7][s%7]的值表示符合题目要求的数的个数，如果不mod 7的话理论可行。但是x&lt;=10^18，则会爆空间。
> 状态转换公式：
> dp[len][x%7][s%7][limit] -&gt;dp[len-1][(x*10+i)%7][(s+i)%7][limit]
> 前置0没有影响，因为0*10还是0。因此我们可以省略一个维度zero。其实也可以省略limit。这种写法我会写在后面

```java

import java.util.Arrays;
import java.util.Scanner;

public class Main{
	// len【位数】, x与s都是mod 7
	// limit【1：限制当前位为该位能取到的上界一方超区间
	static long[][][][] dp = new long[20][10][10][2];
	//最长取到19位
	static int[] dig = new int[20];
	static Scanner scanner = new Scanner(System.in);
	public static void main(String[] args) {
		
		while(scanner.hasNextLine()) {
			long l = scanner.nextLong();
			long r = scanner.nextLong();
			if(l == 0 && r == 0) return;
			System.out.println(solve(r) - solve(l-1));
		}
	}
	public static long solve(long x) {
		long ans = 0;
		//全部初始化为-1
		for(int i = 0 ;i<20; i++) {
			for(int j = 0; j<10; j++) {
				for(int k = 0; k<10; k++) {
					for(int kk = 0; kk<2; kk++) {
						dp[i][j][k][kk] = -1;
					}
				}
			}
		}
		long tmp = x; 
		int len = 0;
		while(tmp>0) {
			dig[++len] = (int) (tmp % 10);

	public static long dfs(int len, int x, int s, int limit) {

		}
		dp[len][x][s][limit] = res;

	}
```

```java
					res += f[i][k - last];
```

```java
import java.util.Arrays;
import java.util.Scanner;

public class Main {
	static int N = 15;
	static int p;
	static int[][][] f = new int[N][10][101];
	static Scanner scan = new Scanner(System.in);

	public static void main(String[] args) {
		while (scan.hasNextLine()) {
			String string = scan.nextLine();
			String[] split = string.split(" ");
			int l = Integer.parseInt(split[0]);
			int r = Integer.parseInt(split[1]);

		}

	}

	static void init() {
		for (int i = 0; i < N; i++) {
			for (int j = 0; j < 10; j++) {
				Arrays.fill(f[i][j], 0);
			}
		}
		// 位数为1
		// 0 模任何数都为0，因此可以作为满足条件的数
		for (int i = 0; i <= 9; i++)
			f[1][i][i % p] = 1;

		// 位数大于1
		for (int i = 2; i < N; i++) {
			for (int j = 0; j <= 9; j++) {
				for (int k = 0; k < p; k++) {
					for (int x = 0; x <= 9; x++) {
						f[i][j][k] += f[i - 1][x][mod(k - j)];
					}
				}
			}
		}
	}

	static int dfs(int n) {
		if (n == 0)
			return 1;

		int[] nums = new int[N];
		int len = 0;
		while (n != 0) {
			nums[len++] = n % 10;
			n /= 10;
		}
		int res = 0;
		// 前缀信息：存储前缀和
		int last = 0;

		for (int i = len - 1; i >= 0; i--) {
			int x = nums[i];
			int start = i == len - 1 ? 1 : 0;
			// 因为前置0，不影响位数小于i的数的最高位的取值，因此j可以优化从0开始。不用管前置0
			for (int j = start; j < x; j++) {
				res += f[i+1][j][mod(-last)];
			}
			//没有退出判断，因为不遍历到最后一个位，我们并不知道是否可以整除
			
			//进行树状递归
			last += x;

			if(i == 0 && mod(last) == 0)
				res++;
		}
		//位数小于i,且最高为是0~9并且，余数为0的数集合。
		for(int i = 1; i<len; i++) {
			for(int j = 1; j<=9; j++) {
				//这个地方填0，因为没有前缀，需要位数为i，且最高位是j，且余数是0
				res += f[i][j][0];
			}
		}
		
		//加上0
		res++;

		return res;
	}
```

```java
		// 位数小于i,且最高为是0~9并且，余数为0的数集合。
		for (int i = 1; i < len; i++) {
			for (int j = 1; j <= 9; j++) {
				// 这个地方填0，因为没有前缀，需要位数为i，且最高位是j，且余数是0
				if(j != 4)
				res += f[i][j];
			}
		}

		// 加上0
		res++;

		return res;
	}

}
```
