```java
        System.out.println(methodOne());
    }

     * 利用java的大整数api求解
     *
     * @return result
     */
    public static int methodOne() {
```

```java
public class CharacterStatistics {
    public static void main(String[] args) {

    }
    public static void methodOne(){
        Scanner scanner = new Scanner(System.in);
        char[] chs = scanner.nextLine().toCharArray();
        int[] counts = new int[26];
        for (int i = 0; i < chs.length; i++) {
            counts[chs[i] - 'A'] += 1;
        }
        int max = counts[0];
        String result = "A";
        for (int i = 1; i < counts.length; i++) {
            if (counts[i] > max) {
                max = counts[i];
                result = "" + (char) (i + 'A');//该字母出现次数大于之前的，就刷新result

                result += (char) (i + 'A');//该字母出现次数等于之前的，就拼接result
            }
        }
```

### 试题 D: 最少刷题数【<span style="color: rgb(79, 79, 79)">10分</span>】

> 时间限制: 1.0s 内存限制: 512.0MB 本题总分： 10 分

**<span style="color: var(--cm-text-color-regular)">【问题描述】</span>**

> <span style="color: var(--cm-text-color-regular)">小蓝老师教的编程课有N名学生，编号依次是 1…N。第i号学生这学期</span><span style="color: var(--cm-text-color-regular)">刷题的数量是</span>$A_i$<span style="color: var(--cm-text-color-regular)">。</span>
> <span style="color: var(--cm-text-color-regular)">对于每一名学生，请你计算他至少还要再刷多少道题，才能使得全班刷题</span><span style="color: var(--cm-text-color-regular)">比他多的学生数不超过刷题比他少的学生数。</span>

**<span style="color: var(--cm-text-color-regular)">【输入格式】</span>**

> <span style="color: var(--cm-text-color-regular)">第一行包含一个正整数N</span><span style="color: var(--cm-text-color-regular)">第二行包含N个整数：</span>$A_1,A_2,A_3......A_N$

**<span style="color: var(--cm-text-color-regular)">【输出格式】</span>**

> <span style="color: var(--cm-text-color-regular)">输出N个整数，依次表示第 1…N号学生分别至少还要再刷多少道题。</span>

**<span style="color: var(--cm-text-color-regular)">【样例输入】</span>**

> <span style="color: var(--cm-text-color-regular)">5</span><span style="color: var(--cm-text-color-regular)">12 10 15 20 6</span>

**<span style="color: var(--cm-text-color-regular)">【样例输出】</span>**

> <span style="color: var(--cm-text-color-regular)">0 3 0 0 7</span>

**<span style="color: var(--cm-text-color-regular)">【评测用例规模与约定】</span>**

> <span style="color: var(--cm-text-color-regular)">对于 30 %的数据，</span>$1\leq N\leq 1000 ,0\leq A_i\leq 1000$<span style="color: var(--cm-text-color-regular)"></span><span style="color: var(--cm-text-color-regular)">对于 100 %的数据，</span>$1\leq N\leq 100000 ,0\leq A_i\leq 100000$

**<span style="color: rgb(77, 77, 77)">简略说一下思路</span>**

> 大致思路是找到已经有序的刷题数序列的处于中间的值，求刷题数和中值的差值，然后再根据情况调整刷题的数量。
> 比如样例的排序后的序列是`6 10 12 15 20`，处于中间的是`12`，那么左边不是`12`的是`6 10`，右边不是`12`的是`15 20`，由于两边数量是相等的，对于左边的如果仅仅是刷题达到中间值的数量，会导致刷题比他多的学生数超过刷题比他少的学生数。比如`6`和`12`的差值是`6`，假如刷题数为`6`的学生再刷`6`题后，序列就会变成`10 12 12 15 20`**，**可以看到刷题比`12`多的学生`{15 20}`比刷题比`12`少的学生`{10}`多，不满足题意，这个时候只需要再刷一道就可以满足，即刷`6`题边`6+1`为`7`，序列变成`10 12 13 15 20`,这个序列就满足全班刷题比他多的学生数`不超过`刷题比他少的学生数。对于处于中间的`12`，由于左右两边数量相等，所以满足题意，不用再多刷题。对于右边的肯定是满足题意的，所以也不用刷题。
> 再举一个例子，对于一个已经排序的序列`6 12 12 15 20`，处于中间的值是`12`，左边不是`12`的是`6`，右边不是`12`的是`15 20`。对于左边的仅仅刷题达到中间的值的数量，依旧不满足题意，和前一个例子一样需要刷的题数比与中间值的差值多`1`题。对于中间的`12`，由于左边的数量少于右边的数量，所以需要多刷`1`道题才能满足，和前一个例子相比，中间的`12`需要刷`1`道题。对于右边的依旧不需要刷题。
> 其实总的来说就是找到左右两边数量的大小关系：
> 1. 不等于中间值，左边的等于右边的数量，左边的刷题数需要是与中间值的差值加 `1` ，中间的不需要刷题，右边的不需要刷题；
> 2. 不等于中间值，左边的小于右边的数量，左边的刷题数需要是与中间值的差值加 `1` ，中间的需要刷 `1` 题，右边的不需要刷题。
> 3. 不等于中间值，左边的大于右边的数量，左边的刷题数需要是与中间值的差值，中间的不需要刷题，右边的不需要刷题。
> 第一种解法中找中间的值的两边与中值相等的数量其实就是体现出了左右两边的数量关系。
> 比如样例的排序后的序列是`6 10 12 14`，处于中间的是`10 12`**我们选择12作为中值，因为不管是6刷到10还是11都不能满足题意，所以10没有参考价值**，`12`作为中间值，那么左边不是`12`的是`{6 10}`，右边不是`12`的是`{14}`，我们可以发现`6 10 12 14`的结果和`6 10 12 12 14`**是一样的。这样我们就配成奇数个数了。**
> 我们把`获取左右不等于中值的个数`改为`获取左右等于中值的个数`。因为是奇数，左右两边的数的个数是相同的。我们在升序的数组中`获取左右等于中值的个数`。就可以避免找不是中值的数而导致的遍历完整个序列。

```java
import java.util.Scanner;

public class Main {

    public static void main(String[] args) {

        Scanner scanner = new Scanner(System.in);

        int n = scanner.nextInt();

        int[] nums = new int[n];
        int[] temp = new int[n];

        for (int i = 0; i < n; i++) {
            nums[i] = scanner.nextInt();
        }

        // 复制数组
        System.arraycopy(nums, 0, temp, 0, nums.length);

        //快速选择
        int midValue = select(temp, (n + 2) >>> 1);

        int l = 0, r = 0;

        for (int i = 0; i < n; i++) {
            if (nums[i] > midValue) {
                r++;
            } else if (nums[i] < midValue) {
                l++;
            }
        }

        int midOption = l < r ? 1 : 0;

        int option = l <= r ? 1 : 0;

        for (int i = 0; i < n; i++) {
            int count = 0;
            if (nums[i] == midValue) {
                count = midOption;
            } else if (nums[i] < midValue) {
                count = midValue - nums[i] + option;
            }

            if (i != n - 1) {
                System.out.print(count + " ");
            } else {
                System.out.println(count);
            }
        }

    }

    public static int select(int[] arr, int k) {
        return select0(arr, 0, arr.length - 1, k);
    }


    /**
     * 快速选择
     */
    private static int select0(int[] arr, int left, int right, int k) {
        if (left == right) {
            return arr[left];
        }

        int mid = partition(arr, left, right);

        int res = mid - left + 1;

        if (k == res) {
            return arr[mid];
        } else if (k < res) {
            return select0(arr, left, mid - 1, k);
        } else {
            return select0(arr, mid + 1, right, k - res);
        }
    }


    /**
     * 基于随机选取轴的划分
     */
    private static int partition(int[] arr, int l, int r) {


        int pivotIndex = (int) (Math.random() * (r - l + 1)) + l;

        arr[l] = new int[]{arr[pivotIndex], arr[pivotIndex] = arr[l]}[0];

        int pivot = arr[l], left = l, right = r + 1;

        while (left < right) {
            while (arr[++left] < pivot && left < r) {
            }

            while (arr[--right] > pivot) {

            }

            if (left >= right) {
                break;
            }

            arr[left] = new int[]{arr[right], arr[right] = arr[left]}[0];
        }
        arr[l] = arr[right];

        arr[right] = pivot;

        return right;
    }

}
```

#### 解法一：找规律

【思路分析】

> 5！= 1*2*3*4*5 = 120 有一个0。其实就是找因子10,也就是质因子（2*5）。质因子2不管有多少个，2的多少次方都不可能出现末尾有0的。因此最终是找阶乘中5的个数。
> //5！= 1 * 2 * 3 * 4 * 5 = 120
> //10！ = 1 * 2 * 3 * 4 * 5 * ... * 9 * (2 * 5) = 3628800
> //15！= 1 * 2 * 3 * 4 * 5 * ... * 9 * (2 * 5) * ... * 14 * (3 * 5) = 1307674368000
> //25！= 1 * 2 * 3 * 4 * 5 * ... * 9 * (2 * 5) * ... * 14 * (3 * 5) ... * (4 * 5) * 24 * (5 * 5) = xxxxxxxxxx000000
> 我们发现，10！中质因子5的个数刚好是 10 / 5
> 我们发现，15！中质因子5的个数刚好是 15 / 5
> 我们发现，25！中质因子5的个数会多出2个，因为5*5=25。使用25 / 5只能得到5个，但是 x/=5循环后。会再加1个。这样就刚刚好6个质因子5。
> 方法cal()传入5时，我们5！中质因子5的个数是1。因此当K = 1时，r = 5满足。

```java
import java.util.Scanner;
// 1:无需package
// 2: 类名必须Main, 不可修改

public class Main {
   public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        long K = sc.nextLong(); //K必须是long类型，不然会报段错误
        long l = 1l; //l在最左边
        long r = Long.MAX_VALUE - 1; //r在最右边，刚刚好-1就不超时。
        while (l < r) {
            long mid = (l + r) / 2; //二分法
            if (calc(mid)>=K) {
                r = mid; //r为阶乘的末尾
            }else {
                l = mid + 1; //左指针右移
            }
        }
        //找到的末尾0的个数大于等于K的第一个数r
        if (calc(r) != K) { //如果遍历完毕最后求r阶乘末尾的0个数不等于K的话就返回-1
            System.out.println(-1);
        }else {
            System.out.println(r); //找到r 输出
        }
        sc.close();
    }
    static long calc(long x) {
        long res = 0;
         //求阶乘末尾0的个数其实就是 求阶乘因子中5的个数
        //5！= 1 * 2 * 3 * 4 * 5 = 120
        //10！ = 1 * 2 * 3 * 4 * 5 * ... * 9 * (2 * 5) = 3628800
        //15！= 1 * 2 * 3 * 4 * 5 * ... * 9 * (2 * 5) * ... * 14 * (3 * 5) = 1307674368000
        //25！= 1 * 2 * 3 * 4 * 5 * ... * 9 * (2 * 5) * ... * 14 * (3 * 5) ... * 24 * (5 * 5) = xxxxxxxxxx0000
        while (x != 0) {
            res = res + x / 5; //求5的个数，如果x为17时，while()循环没有结束，这里不用担心x取到17
            x /= 5; 
        }
        return res;
    }
}
```

```java
import java.util.*;
public class Main {
    public static void main(String[] args) {
        Scanner scanner=new Scanner(System.in);
        int N=scanner.nextInt();
        int M=scanner.nextInt();
        int[][] a =new int[N][M];
        int sq=0;
        for (int i = 0; i <N ; i++)
        {
            for (int j = 0; j <M ; j++) {
                a[i][j]=scanner.nextInt();
            }
        }
        int limit = scanner.nextInt();
        for (int i = 0; i <N ; i++)
        {
            for (int j = 0; j <M ; j++) {
                for (int k = i; k <N ; k++) {
                    for (int l = j; l < M; l++) {
                        int f = f(a, i, j,k,l);
                        if(f<=limit) sq=Math.max((k-i+1)*(l-j+1),sq);
                        else break;
                    }
                }
            }
        }
        System.out.println(sq);
    }
    public static int f(int a[][],int n,int m,int k,int l){
        int min=Integer.MAX_VALUE;
        int max =Integer.MIN_VALUE;
        for(int i=n;i<=k;i++){
            for (int j = m; j <=l ; j++) {
                max=Math.max(max,a[i][j]);
                min=Math.min(min,a[i][j]);
            }
        }
        return max-min;
    }
}
```

#### 解法二：快慢指针+单调队列

【思路分析】

> 看注释
> 遍历所有的上下界情况，O(n^2)。
> 我们这里使用单调队列来存储最大值和最小值。
> 当快指针r运行到最大边界的时候，再往右一步就是不满足题意。这时我们要右移慢指针l来使子矩阵再次满足题意。但是移动慢指针`l`后我们不知道新的较小的子矩阵的max和min更新为多少。因此使用单调队列来存储最大值和最小值，如果`l`移动，就出队首元素（maxq保存的最大值或者minq保存的最小值），之后的首元素就是新的最大值或者最小值。
> 当然，拿示例举例：up = 0, down = 2, l = 0, r= 2，此时`l`将右移，如果移除的那列不包含目前最大值或者最小值，最大值是第三列的9。我们直接进行移除首元素会导致当前矩阵的最大值元素9会被移除。因此我们需要添加判断。如果移除的那列`l`&gt;= 最大值或者最小值对应的**列的下标**。就可以出队。否则不能。
> 这道例题告诉我们，快慢指针不仅可以用作求最小值。还可以求最大值（这题求最大子矩阵）。；两个单调队列可以分别用来保存最大值和最小值。其实好像单调作用也就是这个，哈哈。

```java
import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.Arrays;
import java.util.Deque;
import java.util.LinkedList;
import java.util.StringTokenizer;

public class Main {
	static int m;
	static int n;
	static int[][] nums;

	public static void main(String[] args) throws IOException {
		// 使用单调队列确定上下界，k为上下界（滑动窗口的）的大小
		// 使用双重for遍历所有的滑动窗口。
		m = nextInt();// 行数
		n = nextInt();// 列数
		nums = new int[m][n];
		for (int i = 0; i < m; i++) {
			for (int j = 0; j < n; j++) {
				// 当数字为两位时，去掉空格。在分为char字符串，这个时候是每位数字作为一个字符，当然出错
				//nums[i] = read.readLine().replace(" ", "").toCharArray();
				
				// next().charAt(0) - '0' 如果该为数是两位或者多位就会出错
				nums[i][j] = nextInt();
			}

		}
		int limit = nextInt();
		long ans = 0;// 表示矩阵的大小
		for (int up = 0; up < m; up++) {
			for (int down = up; down < m; down++) {
				// maxq队列的第一个元素是子矩阵的最大值。
				// minq队列的第一个元素是子矩阵的最小值。
				Deque<int[]> maxq = new LinkedList<>();
				Deque<int[]> minq = new LinkedList<>();
				int l = 0, r = 0;
				while (r < n) {
					// newMax([0]该列的下标,[1]该列的最大值)
					int[] newMax = new int[2];
					// newMin([0]该列的下标,[1]该列的最小值)
					int[] newMin = new int[2];

					int max = Integer.MIN_VALUE;
					int min = Integer.MAX_VALUE;

					for (int k = up; k <= down; k++) {
						max = Math.max(max, nums[k][r] - '0');
						min = Math.min(min, nums[k][r] - '0');
					}
					newMax[0] = r;
					newMax[1] = max;
					newMin[0] = r;
					newMin[1] = min;
					// 当慢指针l++后，curMax会改变。
					while (!maxq.isEmpty() && maxq.getLast()[1] <= newMax[1])
						maxq.removeLast();
					// 保存最大值
					maxq.addLast(newMax);
					// 当慢指针l++后，curMin会改变。
					while (!minq.isEmpty() && minq.getLast()[1] >= newMin[1])
						minq.removeLast();
					// 保存最大值
					minq.addLast(newMin);

					int[] curMax = maxq.getFirst();
					int[] curMin = minq.getFirst();
					// 当l == r时，如果在排除当前列的对应的结点，此时就会使maxq或minq为空,getFirst()就会爆异常
					while (curMax[1] - curMin[1] > limit && l < r) {
						// l目前表示要去除的那一列
						// 去除队列首结点是前l列的，包括第l列。
						if (!maxq.isEmpty() && curMax[0] <= l)
							maxq.removeFirst();
						// 去除队列首结点是前l列的，包括第l列。
						if (!minq.isEmpty() && curMin[0] <= l)
							minq.removeFirst();

						curMax = maxq.peekFirst();
						curMin = minq.getFirst();
						l++;

					} // while结束时三种情况：一是：l==r但是仍然>limit。这个时候我们只针对r++，让下次while循环结束它。
						// 二是：满足条件<=limit并且l<r，将该变小后的矩阵的大小比较ans就行。
						// 三是：满足条件<=limit并且l==r，子矩阵只有一列了。r++后继续循环就行。
					if (curMax[1] - curMin[1] > limit) {
						r++;
						continue;
					}

					// 保存最大子矩阵大小
					ans = Math.max(ans, (down - up + 1) * (r - l + 1));
					r++;
				}
			}

		}
		System.out.println(ans);

	}

	static BufferedReader read = new BufferedReader(new InputStreamReader(System.in));
	static StringTokenizer token = new StringTokenizer("");

	static String next() {
		while (!token.hasMoreTokens()) {
			try {
				token = new StringTokenizer(read.readLine());
			} catch (IOException e) {
				throw new RuntimeException(e);
			}
		}
		return token.nextToken();
	}

	static int nextInt() {
		return Integer.parseInt(next());
	}

}
```

#### 解法一：动态DP

【思路分析】

> 看注释
> [https://blog.csdn.net/m0_73889210/article/details/134581400](https://blog.csdn.net/m0_73889210/article/details/134581400)

```java
/*
方法：DP
技巧：如何判断区间[i,j]是否可以组成一段连续的自然数？
    只需区间最大值 - 区间最小值 == j - i (区间长度)即可

思路：这里设两层循环，一层j表示左端点，第二层i表示右端点。
    因为是连续的所以在所取的[l，r]范围中寻找最大值，最小值。
    然后相减，最后和r-l作比较即可。因为是1~N的排列。
    {3,2,1} j = 1, i = 3, max = 3 , min = 1
 */
import java.util.Scanner;
public class Main {
    static int mod = (int)1e9 + 7;
    public static void main(String[] args) {
        Scanner scan=new Scanner(System.in);
        int n=scan.nextInt();
        int[] a=new int[n+1];
        for (int i=1;i<=n;i++) a[i]=scan.nextInt();
        //dp[i]：以a[i]结尾的切分合法数组的方法数量
        int[] dp=new int[n+1];
        dp[0]=1;//初始化
        for (int i=1;i<=n;i++){
            int max=Integer.MIN_VALUE,min=Integer.MAX_VALUE;
            /**
            在第i次循环中我们考虑的部分是：第1个元素到第i个元素
            而第j个元素到第i个元素是我们切分后的最后一个集合，记为A
            max 是A中的最大值，min是A中的最小值
            当i-j = max-min 时，说明A中的元素是连续的，此时第一个元素到第j-1个元素共有dp[j-1]种切法，与A拼接到最后之后，共有dp[j-1]种
            必须判断A是连续的，不然不能拼接。
            j继续减小寻找下一个集合A（元素连续的集合），将这些切法均累加到dp[i]中
            dp[4] = dp[3] + dp[2] + dp[1] + dp[0] ，我们定义dp[0]=1
            下面这个推导看链接。
            共4个数 += 前三位数的种数【如果第四位数构成的集合A满足连续】
            共4个数 += 前两位数的种数【如果第三，四位数构成的集合A满足连续】
            共4个数 += 前一位数的种数【如果第二，三，四位数个数构成的集合A满足连续】
            共4个数 += dp[0]就是加1，因为不切分就是一种情况
          */
            for (int j=i;j>0;j--){
                max =Math.max(max,a[j]);
                min =Math.min(min,a[j]);
                if (max-min==i-j) dp[i]=(dp[i]%mod+dp[j-1]%mod)%mod;

            }
        }
        System.out.println(dp[n]);
    }
}
```

#### 解法二：记忆化搜索

【思路分析】

> 动态DP的变式。看懂解法一就能看懂记忆化搜索的解法。其实解法二要更好理解一点，因为i不用从

```java
import java.util.*;
// 1:无需package
// 2: 类名必须Main, 不可修改

public class Main {

    private static final int MOD = 1000000007;
    static int[] dp;

    public static void main(String[] args) {
        Scanner scan = new Scanner(System.in);
        //在此输入您的代码...

        int n = scan.nextInt();
        dp = new int[n + 1];
        Arrays.fill(dp, -1);
        int[] nums = new int[n+1];
        for (int i = 1; i <= n; i++) {
            nums[i] = scan.nextInt();
        }

        System.out.println(dfs(n, nums));

        scan.close();
    }

    static int dfs(int i, int[] nums) {
        //记忆化搜索放到第一位可以少进过几个判断语句。因为后期都是直接使用记忆化的值。而不是递归到1或者0
        if (dp[i] != -1) return dp[i];
        //dp[2] = dp[1] + dp[0], dp[1]表示只有一个数。dp[0]我们定义它为1，这样就不用单独添加不切分的情况了。
        //dp[1] = dp[0] 
        if (i == 1) return 1;//不写这句判断也行
        if (i == 0) return  1;
        int res = 0;
        int max = Integer.MIN_VALUE, min = Integer.MAX_VALUE;
        for (int j = i; j > 0; j--) {
            max = Math.max(max, nums[j]);
            min = Math.min(min, nums[j]);
            if (max - min == i - j)
                res = (res + dfs(j-1, nums)) % MOD;
        }
        //不用因为return dp[i] = res赋值了。
        // dp[i] = res;
        return dp[i] = res;
    }
}
```
