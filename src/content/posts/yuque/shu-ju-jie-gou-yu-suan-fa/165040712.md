```java

import java.util.Scanner;

/**
 * @author ztyOperation
 * @create 2022-02-27-21:07 快速排序 不稳定 时间 最好O(nlog2n) 最坏n^2 空间O(nlog2n)
 */
public class Main {
	public static void main(String[] args) {
		Scanner scan = new Scanner(System.in);
		int n = scan.nextInt();
		int[] a = new int[n];
		for (int i = 0; i < n; i++) {
			a[i] = scan.nextInt();

		}
		quick_sort(a, 0, n - 1);
		for (int i = 0; i < n; i++) {
			System.out.print(a[i] + " ");
		}

	}

	public static void quick_sort(int[] q, int l, int r) {
		if (l >= r)
			return;
        // 可以理解为随便找一个基准点，然后将区间变为左边是全部小于等于基准点，右边是全部大于等于基准点
		int x = q[l], i = l - 1, j = r + 1;
		while(i<j) {
			do i++; while(q[i] < x);
			do j--; while(q[j] > x);
			if(i<j) swap(q, i, j);
		}
		quick_sort(q, l, j);
		quick_sort(q, j+1, r);
	}
	private static void swap(int[] q, int i , int j) {
		int t = q[i];
		q[i] = q[j];
		q[j] = t;
	}

}
```

```java
import java.util.Arrays;

/**
 * @author ztyOperation
 * @create 2022-02-27-21:07
 * 快速排序 不稳定 时间 最好O(nlog2n) 最坏n^2 空间O(nlog2n)
 */
public class Main {
    public static void sort(int[] a, int low, int high) {
        int i = low;
        int j = high;//初始high == a.length-1

        int pivot = a[low];
        while (i != j) {
            while (i < j && a[j] >= pivot) {//刚好等于时，需要进入循环继续下去
                j--;
            }
            if (i < j) {
                a[i] = a[j];
            }

            while (i < j && a[i] <= pivot) {
                i++;
            }
            if (i < j) {
                a[j] = a[i];
            }

        }
        a[j] =  a[i] = pivot;

        if (i > 1 + low) {//i最小为2，所以前面至少有0,1两个数
            sort(a, low, i - 1);
        }
        if (j < high - 1) {  //j最大为high-2,所以若high == 8，则j最大为6, 后面有,7,8两个数
            sort(a, i + 1, high);
        }

    }

    public static void main(String[] args) {
        Scanner scan = new Scanner(System.in);
		int n = scan.nextInt();
		int[] a = new int[n];
		for (int i = 0; i < n; i++) {
			a[i] = scan.nextInt();

		}
		sort(a, 0, n - 1);
		for (int i = 0; i < n; i++) {
			System.out.print(a[i] + " ");
		}
    }
}
```

```java
public class MergeSort {
    public static void main(String[] args) {
        int[] a = new int[]{6, 4, 3, 7, 9, 0, 6, 3, };
        System.out.println(Arrays.toString(a));
        int[] temp = new int[a.length];
        sort(a,0,a.length-1,temp);
        System.out.println(Arrays.toString(a));

    }

    public static void sort(int[] arr, int left, int right, int[] temp) {
        int mid = (left + right) / 2;
        if (left < right) {
            //将左边的分开
            sort(arr,left,mid,temp );
            //将右边的分开
            sort(arr,mid+1,right,temp);

            merge(arr,left,mid,right,temp);

        }
    }
    //对有序序列的合并部分

    public static void merge(int[] arr, int left, int mid, int right, int[] temp) {
        int i = left;//表示左边有序序列的初始索引
        int j = mid + 1;//表示左边有序序列的初始索引
        int t = 0; //temp[] 的索引
        //排序合并
        while (i <= mid && j <= right) {
            if (arr[i] <= arr[j]) {
                temp[t] = arr[i];
                i++;
                t++;
            } else {
                temp[t] = arr[j];
                j++;
                t++;
            }
        }

        //左序列或右序列剩余部分
        while (i <= mid) {

            temp[t] = arr[i];
            i++;
            t++;
        }
        while (j <= right) {
            temp[t] = arr[j];
            t++;
            j++;
        }

        //将temp[] 填充到arr[]中去
        t = 0;
        int tempLeft = left;
        System.out.println("left = " + left + "right = " + right);
        System.out.println("temp = "+Arrays.toString(temp));

        while (tempLeft <= right) {
            arr[tempLeft] = temp[t];
            tempLeft++;
            t++;
        }
        System.out.println("arr = "+ Arrays.toString(arr));

    }
}
```

## 3.堆排序

```java
/**
 * @author ztyOperation
 * @create 2022-05-02-10:26
 * 堆排序 不稳定 时间O(nlog2n) 空间O(1)
 */
public class HeapSort {
    public static void main(String[] args) {
        int[] arr = {4, 2, 6, 8, 1, 3, 5};
//        int[] arr = {4,6,8,6,9};
        HeapSort.sort(arr);
        System.out.println(Arrays.toString(arr));

    }

    public static void sort(int[] arr) {
        //得到大顶堆，但是并没有保证，左子节点 绝对小于 右子节点，因此要后续调整
        //最初i为右下角的非叶子结点
        // 1).将无序序列构建成一个堆，根据升序降序需求选择大顶堆或小顶堆;
        for (int i = arr.length / 2 - 1; i >= 0; i--) {//从下到上,从右到左
            adjust(arr, i, arr.length);
        }
        /*
        将根节点与最后一个结点互换，并重新进行大顶堆的调整
        2).将堆顶元素与未尾元素交换，将最大元素"沉"到数组未端;
        3).重新调整结构，使其满足堆定义，然后继续交换堆顶元素与当前未尾元素，反复执行调整+交换步骤，直到整个序列有序。
        */
        for (int j = arr.length - 1; j > 0; j--) {
            int temp = arr[0];
            arr[0] = arr[j];
            arr[j] = temp;
            adjust(arr, 0, j);//从上到下，再一次微调成大顶堆，也就是将下一个最大值放在 堆顶。
        }
    }

    /**
     * 功能：对以i为对应的非叶子结点的树调整成大顶堆
     *
     * @param arr    表示待调整的数组
     * @param i      表示非叶子结点的索引
     * @param length 表示对多少个元素进行调整（每得到一个大顶堆，会将根节点与最后一个结点互换），length在逐渐减少
     */
    public static void adjust(int[] arr, int i, int length) {
        int temp = arr[i];//保存i结点的数值，防止数据丢失
        //k指向i结点的左子节点
        for (int k = 2 * i + 1; k < length; k = k * 2 + 1) {
            //将k指针指向i左右子节点中的较大值
            if (k + 1 < length && arr[k] < arr[k + 1]) {//k+1为右孩子的下标
                k++;
            }
            if (arr[k] > temp) {//比较的是最初的值,temp小于arr[k]会被传下去做为下一个循环的父节点
                arr[i] = arr[k];//把最大的值放到双亲结点拿那去
                //替换后，需要继续 for 比较替换下去的 temp 值是否仍然满足左右子节点小于父节点。
                i = k;//i指向k,保存被替换的值的索引，最后 arr[i] = temp 需要用到。
            } else {
                break;//替换下去的值仍然大于左右子节点，就不用调整了（我们是自下而上的调整），或者直接是没有进行交换，退出循环，因为不需要在进行调整
            }
        }//for结束
        arr[i] = temp;//若有交换，i此时已经指向原先自身的子节点上;若没有交换，i还是本身,用本身的值给本身赋值没问题
    }
}
```

## 4. 基数排序（不能排负数，使用数组桶）

```java

import java.util.Arrays;

/**
 * @author ztyOperation
 * @create 2022-04-25-17:32
 * 只能排序非负数
 * 基数排序 稳定 平均时间 O(nxk) 空间O(n+k)
 */
public class RadixSort {
    public static void main(String[] args) {
        int[] arr = new int[80000];
        for(int i =0 ;i< 80000; i++){
            arr[i] = (int)(Math.random() * 80000);
        }
        long start = System.currentTimeMillis();
        RadixSort.sort(arr);
        long end = System.currentTimeMillis();
        System.out.println("花费时间： "+ (end-start));
    }

    public static void sort(int[] arr){
        //找到最大数的位数多少
        int max = arr[0];
        for(int i = 1 ; i< arr.length; i++){
            if(max<arr[i]){
                max = arr[i];
            }
        }

        int maxLength = (max + "").length();

        for (int i = 0,  n = 1; i<maxLength; i++, n *= 10) {
            //10 0-9共10个桶 桶的大小为arr.length 防止数据溢出
            int[][] bucket = new int[10][arr.length];

            //计算每个桶中储存的数据个数
            int[] bucketElementCount = new int[10];

            //装数据
            for(int j = 0; j<arr.length; j++){
                int digitOfElement = arr[j] / n % 10;
                bucket[digitOfElement][bucketElementCount[digitOfElement]] = arr[j];
                bucketElementCount[digitOfElement] ++;
            }

            int index = 0;//表示arr[]的索引
            //覆盖arr[]
            for(int k = 0; k<10; k++){
                //说明该桶有数据
                if(bucketElementCount[k] != 0){
                    for(int l = 0; l<bucketElementCount[k]; l++){
                        arr[index++] = bucket[k][l];
                    }
                }
                bucketElementCount[k] = 0; //注意！！,一定要置0每个桶装的数据个数，一遍下一轮的装桶
            }
            System.out.println("第" + (i+1) + "轮排序 "+ Arrays.toString(arr));
        }


    }

}
```

## 6.桶排序

```java
public class BucketSort {
    public void bucketSort(int[] arr, int bucketSize) {
        if (arr.length <= 1) {
            return;
        }

        // Step 1: 找出最大值和最小值
        int minValue = arr[0];
        int maxValue = arr[0];
        for (int i = 1; i < arr.length; i++) {
            if (arr[i] < minValue) {
                minValue = arr[i];
            }
            if (arr[i] > maxValue) {
                maxValue = arr[i];
            }
        }

        // Step 2: 创建桶，桶的数量由数据范围决定
        // 例如，如果你选择 bucketSize 为 10，那么每个桶将容纳 10 个数。即一个桶包含从某个范围 minValue 到 minValue + bucketSize - 1 之间的所有数。最终保证范围里内的数都能被装入。
        // 极端点，每个数都使用一个桶（每个桶只装一个元素，其实也能可能是多个相同大小的元素装在一个桶内），这样桶就不需要桶内排序了。
        int bucketCount = (maxValue - minValue) / bucketSize + 1;
        List<List<Integer>> buckets = new ArrayList<>(bucketCount);

        for (int i = 0; i < bucketCount; i++) {
            buckets.add(new ArrayList<>());
        }

        // Step 3: 将元素分配到桶中
        for (int num : arr) {
            int index = (num - minValue) / bucketSize;
            buckets.get(index).add(num);
        }

        // Step 4: 对每个桶内的元素进行排序
        int currentIndex = 0;
        for (List<Integer> bucket : buckets) {
            // 使用计数排序来替代排序桶内的元素
            int[] count = new int[maxValue - minValue + 1];  // 计数排序数组
            for (int num : bucket) {
                count[num - minValue]++;
            }

            for (int i = 0; i < count.length; i++) {
                while (count[i]-- > 0) {
                    arr[currentIndex++] = i + minValue;
                }
            }
        }
    }
}
```

> 对于桶排序，每个桶中有多个元素的话，需要将桶中元素进行排序（使用冒泡排序，保持稳定但是可能导致桶排序最坏 O(n^2)）

## 10.希尔排序

```java

import java.util.Arrays;

/**
 * @author ztyOperation
 * @create 2022-04-24-20:23
 * 希尔排序 不稳定 时间O(nlog2n) 空间O(1)
 */
public class ShellSort {
    public static void main(String[] args) {
        int[] a = new int[]{4, 2, 5, 6, 8, 9, 6, 0,4};
        ShellSort.sort(a);
    }

    /*
    分组，每组都进行插入排序
     */
    public static void sort(int[] arr) {
        int count = 0;
        for (int gap = arr.length / 2; gap > 0; gap /= 2) {//gap为步长

            for (int i = gap; i < arr.length; i++) {

                int data = arr[i];
                int j = i - gap;
                while (j>= 0 && data < arr[j]) {
                    arr[j+gap] = arr[j];
                    j -= gap;
                }
                arr[j] = data;
            }
                System.out.println("第" + (++count) + "轮排序： " + Arrays.toString(arr));
        }
    }
}
```
