import java.io.*;
import java.math.BigInteger;
import java.util.*;

public class Main {
    static Scanner scan = new Scanner(new BufferedInputStream(System.in));
    static TreeMap<Integer, Integer> map;

    static final long p = (long) 1e9 + 7;

    public static void main(String[] args) throws IOException {
        //区间长度
        long n = nextLong();
        int m = nextInt();
        map = new TreeMap<>();
        for (int i = 0; i < m; i++) {
            int num = nextInt();
            int value = nextInt();
            map.put(value, num);
        }
        int q = nextInt();
        for (int i = 0; i < q; i++) {

            long l = nextLong();
            long r = nextLong();
            Set<Integer> values = map.keySet();
            long num = 1;
            long tt = 1;//前一个区间的长度
            for (int value : values) {
                Integer numbers = map.get(value);
                //当前区间的长度【tt, tt+ numbers - 1】
                long myR = tt + numbers - 1;
                //下面三种情况
                if (l <= tt && myR <= r){
                    num = num * qmi(value, numbers) % p;
//                    num *= Math.pow(value, numbers);
                }
                else if(tt <= l && myR >= r){
                    num = num * qmi(value, r - l + 1) % p;
                }
                else if (tt <= l && myR >= l){
                    num = num * qmi(value, myR - l + 1) % p ;
//                    num *= Math.pow(value, myR - l + 1);
                }
                else if (tt <= r && myR >= r){
                    num = num * qmi(value, r - tt + 1) % p;
//                    num *= Math.pow(value, r - tt + 1);
                }
                else if (myR < r);
                else if (tt > l);
                tt = myR + 1;
                if(tt > r)break;

            }
            System.out.println(get(num) % p );
        }
//        //一个数可以被分解为p1^k1 + p2 ^ k2 + ... Pn ^ kn, p1, p2, ,, pn都是质数。
    }

    //查找一个数的约数的个数
    static long get(long n) {
        List<Integer> list = new ArrayList<>();
        for (int i = 2; i * i <= n; i++) {
            if (n % i == 0) {
                int cnt = 0;
                while (n % i == 0) {
                    cnt++;
                    n /= i;
                }
                list.add(cnt+1);
            }
        }
        long res = 1;
        if (n != 1) list.add(2);
        for (int i = 0; i < list.size(); i++) {
            res = res * (list.get(i)) % p;
        }
        return res;
    }

    static long qmi(long a, long b){
        long ans = 1L;
        while(b != 0){
            if((b & 1) == 1){
                ans = ans * a % p;
            }
            b >>=1;
            a = a * a % p ;
        }
        return ans;

    }
    static BigInteger qmi(BigInteger a, int b){
            return a.modPow(BigInteger.valueOf(b), BigInteger.valueOf(p));

    }



    static StreamTokenizer token = new StreamTokenizer(new BufferedReader(new InputStreamReader(new BufferedInputStream(System.in))));

    static int nextInt() throws IOException {
        token.nextToken();
        return (int) token.nval;
    }

    static long nextLong() throws IOException {
        token.nextToken();
        return (long) token.nval;
    }
}
