**【答案提交】**

> 这是一道结果填空的题，你只需要算出结果后提交即可。本题的结果提交答案时只填写这个字符串，填写多余的内容将无法得分。

**解法：**

```java

public class OverLapTime {
        public static void main(String[] args) {
                LocalTime start = LocalTime.of(6,13,22);
                LocalTime end = LocalTime.of(14, 36, 20);
                int count =0 ;
                for(;start.compareTo(end) <=0; start = start.plusSeconds(1) ){
                        if(start.getSecond() == start.getMinute() && start.getSecond() != 0){
                                count ++;
                        }
                }
                System.out.println(count);
        }
}
```

> 1. 题目中分针和秒针都是匀速运动。假如从00:00开始。01:00整时，没有重合。当秒针来到01秒时。分针已经运动一段距离了。因此最终重合点在01后面一点点。这一点点是秒针从整点到重合点所花费的时间。因此第一个重合点在01到02之间，靠近01。
> 我们在代码中仅仅体现出01：01重合一次。
> 而真实情况是【01，02】之间重合。
> 2. 当时间来到了59:00。当秒针来到59秒时，分针已经运动到接近00的位置。因此最终重合点在00前面一点点。因为秒针从整点到重合点所花费的时间接近1s。因此最后一个重合点在59到00之间，靠00。
> 我们在代码中仅仅体现出59：59重合一次。
> 而真实情况是【59，00】之间重合。也可以看作整点重合一次。
> 3. **1小时。只有59次重合。**
> 4. 如果开始时间是整点开始的话，我们的代码需要加一次初始的整点的重合。

```java
import java.io.PrintWriter;
import java.io.IOException;
import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.io.StreamTokenizer;
import java.util.PriorityQueue;
import java.util.Queue;

public class Main {

	public static void main(String[] args) { new Main().run(); }
	
	char[][] desktop;
	
	int n, m;
	
	void run() {
		desktop = new char[n = nextInt()][m = nextInt()];
		PrintWriter out = new PrintWriter(System.out);
		Queue<Window> queue = new PriorityQueue();
		Window[] windows = new Window[100001];
		for (int k = nextInt(); k-- > 0;) {
			String opt = next();
			int PID = nextInt();
			if (opt.equals("close")) windows[PID] = null;
			else {
				switch (opt) {
					case "new": 
						windows[PID] = 
							new Window(nextInt(), nextInt(), nextInt(), nextInt());
						break;
					case "move":
						windows[PID].
							move(nextInt(), nextInt());
						break;
					case "resize":
						windows[PID].
							resize(nextInt(), nextInt());
				}
				windows[PID].z = k;
			}
		}
		for (int i = 1; i <= 100000; ++i)
			if (windows[i] != null) queue.offer(windows[i]);
		while (queue.size() > 0) {
			Window now = queue.poll();
			int x1 = now.top, x2 = now.top + now.height - 1,
				y1 = now.left, y2 = now.left + now.width - 1;
			for (int i = x1 + 1; i < x2; ++i)
				for (int j = y1 + 1; j < y2; ++j) paint(i, j, ' ');
			for (int j = y1; j <= y2; ++j) {
				paint(x1, j, j > y1 && j < y2 ? '-' : '+');
				paint(x2, j, j > y1 && j < y2 ? '-' : '+');
			}
			for (int i = x1 + 1; i < x2; ++i) {
				paint(i, y1, '|');
				paint(i, y2, '|');
			}
		}
		for (int i = 0; i < n; ++i) {
			for (int j = 0; j < m; ++j)
				out.write(desktop[i][j] == '\0' ? '.' : desktop[i][j]);
			out.println();
		}
		out.flush();
	}
	
	void paint(int x, int y, char color) { if (x >= 0 && x < n && y >= 0 && y < m && desktop[x][y] == '\0') desktop[x][y] = color; }
	
	class Window implements Comparable<Window> {
		
		int z, top, left, height, width;
		
		Window(int top, int left, int height, int width) {
			this.resize(height, width);
			this.move(top, left);
		}
		
		void move(int vertical, int horizontal) {
			this.top += vertical;
			this.left += horizontal;
		}
		
		void resize(int height, int width) {
			this.height = height;
			this.width = width;
		}

		@Override
		public int compareTo(Window win) { return Integer.compare(this.z, win.z); }
	}
	
	StreamTokenizer in = new StreamTokenizer(new BufferedReader(new InputStreamReader(System.in)));
	
	StreamTokenizer nextToken() {
		try {
			in.nextToken();
		} catch (IOException e) {
			e.printStackTrace();
		}
		return in;
	}
	
	int nextInt() { return (int)nextToken().nval; }
	
	String next() { return nextToken().sval; }
}
```

### 试题 E: 迷宫【编程题】

> ![image.png](https://pub-ccd5622eabb54b4a81678003778ddce1.r2.dev/PicGo/d146cefe7199b23baec73ae1a6d16f13a7e4531ed7401e4a2179775c84e9c6d3.png)

#### 解法一：自己的解法

【思路分析】

> 使用BFS求终点到每一个节点的最短距离。使用map存储传送门。

```java
import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.HashMap;
import java.util.LinkedList;
import java.util.Map;
import java.util.Queue;
import java.util.StringTokenizer;

public class Main {
	static Map<Integer, ArrayList<int[]>> map = null;
	// 迷宫大小
	static int n;
	// 传送门个数
	static int m;
	// 我们使用邻接矩阵来表示各个节点
	static int[][] st;

	static int N = 2010;

	// x不动，y动；y不动，x动
	static int[] dx = { 0, 0, 1, -1 };
	static int[] dy = { 1, -1, 0, 0 };

	public static void main(String[] args) {
		// 迷宫大小
		n = nextInt();
		// 传送门个数
		m = nextInt();
		// 我们使用邻接矩阵来表示各个节点
		st = new int[N][N];

		// 使用BFS求（n,n）到每一个节点的最短距离
		Queue<int[]> q = new LinkedList<>();

		// 保存传送门相连的两个结点的信息, int[]表示可能有多个节点与之相连
		map = new HashMap<>();

		// 统计节点n,n到每一个节点的最短距离
		// 节点的数量 2 * n + 1
		int[] dis = new int[N * N];

		for (int i = 0; i < m; i++) {
			int x1 = nextInt();
			int y1 = nextInt();
			int x2 = nextInt();
			int y2 = nextInt();
			add(x1, y1, x2, y2);
			add(x2, y2, x1, y1);

		}
		// 表示所有节点都没有被访问过
		Arrays.fill(dis, -1);

		long res = 0;

		q.add(new int[] { n, n });
		// 大家都平移，n个，本来从1开始，应该是（n-1）* n + n
		dis[n * n + n] = 0;
		while (q.size() > 0) {
			int[] u = q.poll();
			int x = u[0];
			int y = u[1];
			// 找出通过传送门与u相连的节点
			if (map.containsKey(x * n + y)) {
				for (int[] v : map.get(x * n + y)) {
					if (dis[v[0] * n + v[1]] == -1) {
						dis[v[0] * n + v[1]] = dis[x * n + y] + 1;
						res += dis[v[0] * n + v[1]];
						q.add(v);
					}
				}
			}
			// 与u相连的节点
			for (int i = 0; i < 4; i++) {
				int newX = dx[i] + x;
				int newY = dy[i] + y;
				if (newX >= 1 && newX <= n && newY >= 1 && newY <= n && dis[newX * n + newY] == -1) {
					q.add(new int[] { newX, newY });
					dis[newX * n + newY] = dis[x * n + y] + 1;
					res += dis[newX * n + newY];
				}
			}
		}
		System.out.printf("%.2f\n", res * 1.0 / (n * n));
	}

	static void add(int x1, int y1, int x2, int y2) {
		if (!map.containsKey(x1 * n + y1))
			map.put(x1 * n + y1, new ArrayList<int[]>());
		map.get(x1 * n + y1).add(new int[] { x2, y2 });

	}

	static BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));

	static StringTokenizer token = new StringTokenizer("");

	static String next() {
		while (!token.hasMoreTokens()) {
			try {
				token = new StringTokenizer(reader.readLine());
			} catch (IOException e) {
```

```java
import java.io.BufferedInputStream;
import java.io.BufferedReader;
import java.io.BufferedWriter;
import java.io.IOException;
import java.io.InputStreamReader;
import java.io.OutputStreamWriter;
import java.io.PrintWriter;
import java.io.StreamTokenizer;

public class Main {
	static int n;
	static int[] a; // weight
	static int[] b; // value
	static long[][][] f;

	static class Point {
		int x;
		int y;

		Point(int x, int y) {
			this.x = x;
			this.y = y;
		}
	}

	static Point[] p;

	public static void main(String[] args) throws IOException {
		n = nextInt();
		f = new long[n + 1][n + 1][n + 1];
		p = new Point[n + 1];
		for (int i = 1; i <= n; i++) {
			p[i] = new Point(nextInt(), nextInt());
		}
		long res = 0;
		// 插多少个洞 至少插3个洞，区间DP的固定写法.
		for (int k = 3; k <= n - 2; k++) {
  			// i为左端点
			for (int i = 1; i + k -1 <= n; i++) {
				// 不要循环直接 int j = i + k -1 是区间连续的要求，这题可以在更长的区间中，选择k个洞进行插入，不要求连续插洞
				for (int j = i + k - 1; j <= n; j++) {
					for (int h = i; h < j; h++) {
						f[i][j][k] = max(f[i][j][k], f[i][h][k - 1]+area(p[i],p[h],p[j]));
						res = max(f[i][j][k], res);
					}
				}
			}
		}
        
		out.println(res);
		out.flush();
	}

	static long area(Point a, Point b, Point c) {
        //x , y 都是10^6,因此需要转为long类型.
		//(long)(c.x - a.x)再进行乘法就不会爆
		//(long)((c.x - a.x) * (b.y - a.y))先进行乘法再强转会爆掉，因为默认是int存储
		return abs( (long)(c.x - a.x) * (b.y - a.y) -  (long)(b.x - a.x) * (c.y - a.y));
	}

    static long abs(long a){
        return  a > 0? a : -a;
    }

	static StreamTokenizer token = new StreamTokenizer(
			new BufferedReader(new InputStreamReader(new BufferedInputStream(System.in))));

	static double nextDouble() throws IOException {
		token.nextToken();
		return token.nval;
	}

	static int nextInt() throws IOException {
		token.nextToken();
		return (int) token.nval;
	}

	static PrintWriter out = new PrintWriter(new BufferedWriter(new OutputStreamWriter(System.out)));

	static long max(long a, long b) {
		return a > b ? a : b;
	}
}
```

```java
import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.io.StreamTokenizer;
import java.io.IOException;

/**
 * 如果有七个点，选五个点，全部遍历都需要dp[][][][][]，所以直接给一个[][]表示i-j个点之间，再来一个[]表示取多少个点
 */
public class Main {

    public static void main(String[] args) { new Main().run(); }

    void run() {
        int n = nextInt();
        int[] x = new int[n], y = new int[n];
        long[][][] dp = new long[n][n][n];
        long res = 0;
        for (int i = 0; i < n; ++i) {
            x[i] = nextInt();
            y[i] = nextInt();
        }
        //k = 2表示总共i到j中选择3个点： i= 0 j =2 ，k = 2 表示 2 1 0 三个点
        for(int k=2;k<n-2;k++){
            for(int i=0;i<n;i++){
                for(int j=i+k;j<n;j++){
                    for(int h=i;h<=j;h++){
                        dp[i][j][k] = max(dp[i][j][k],dp[i][h][k-1]+
                                abs(((long)(x[j]-x[i])*(y[h]-y[i])-(long)(y[j]-y[i])*(x[h]-x[i]))));
                        res = max(dp[i][j][k],res);
                    }
                }
            }
        }

        System.out.println(res);
    }

    long max(long arg1, long arg2) { return arg1 > arg2 ? arg1 : arg2; }

    long abs(long arg) { return arg > 0 ? arg : -arg; }

    StreamTokenizer in = new StreamTokenizer(new BufferedReader(new InputStreamReader(System.in)));

    int nextInt() {
        try {
            in.nextToken();
        } catch (IOException e) {
            e.printStackTrace();
        }
        return (int)in.nval;
    }
}
```

**解法一：使用构造法**

> 构造出所有包含2022这个子串的数字，并且使用set集合防止，2022022,20222022这样的数字重复出现。

```java
import java.util.HashSet;
import java.util.Scanner;
import java.util.Set;

public class Main {
	public static void main(String[] args) {
		Set<Integer> set = new HashSet<>();
		Scanner sc = new Scanner(System.in);
		int l = sc.nextInt();
		int r = sc.nextInt();
		for (int i = 0; i < 10; i++) {
			for (int j = 0; j < 10; j++) {
				for (int k = 0; k < 10; k++) {
					for (int m = 0; m < 10; m++) {
						for (int n = 0; n < 10; n++) {
							// xxxxx2022
							int num1 = i * 100000000 + j * 10000000 + k * 1000000 + m * 100000 + n * 10000 + 2022;
							// xxxx2022x
							int num2 = i * 100000000 + j * 10000000 + k * 1000000 + m * 100000 + 20220 +  n * 1;
							// xxx2022xx
							int num3 = i * 100000000 + j * 10000000 + k * 1000000 + 202200 +m * 10 + n * 1;
							// xx2022xxx
							int num4 = i * 100000000 + j * 10000000 + 2022000 + k * 100 + m * 10 + n * 1;
							// x2022xxxx
							int num5 = i * 100000000 + 20220000 + j * 1000 + k * 100 + m * 10 + n * 1;
							// 2022xxxxx
							int num6 = 202200000 + i * 10000 + j * 1000 + k * 100 + m * 10 + n * 1;

							set.add(num1);
							set.add(num2);
							set.add(num3);
							set.add(num4);
							set.add(num5);
							set.add(num6);
						}
					}
				}
			}
		}
		long sum = 0;
		for (int num : set) {
			if (num >= l && num <= r)
				sum += num;
		}
		System.out.println(sum);
	}

}
```
