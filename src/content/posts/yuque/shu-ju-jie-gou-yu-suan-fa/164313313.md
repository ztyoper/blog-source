### 试题一: 全排列

【思路分析】

> 使用递归回溯即可解决，示意图上面有。

```java
import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

public class Main {
	static Scanner scanner = new Scanner(System.in);
	static int n  = 0;//n等于3时，就是1 2 3 的全排列
	//保存答案 [123] [132] [213] [231] [312] [321]
	static List<ArrayList<Integer>> list = new ArrayList<>();
	public static void main(String[] args) {
		n = scanner.nextInt();
		ArrayList<Integer> res = new ArrayList<>();
		int[] v = new int[n+1];
		dfs(0,v,res);
		for(ArrayList<Integer> result : list) {
			System.out.println(result.toString());
			
		}
	}
	public static void dfs(int cnt, int[] v, ArrayList<Integer> res) {
		if(cnt == n) {
			list.add(new ArrayList<>(res));
			return ;
		}
		for(int i = 1; i<=n; i++) {
			//判断条件,该数已经加入过
			if(v[i] == 1)
				continue;
			//递归
			res.add(i);
			v[i] = 1;
			dfs(cnt+1,v,res);//cnt+1，这样就不用cnt--了。
			//回溯
			res.remove(res.size()-1);//去除末尾添加的那个
			v[i] = 0;
		}
	}
}
```

### 试题二: 开心

解法一：

【思路分析】

> 1234 2：n=1234 k = 2
> 1_2_3_4然后递归调用dfs,替换_为+然后获得结果。但是这样有8个测试点运行超时。

```java
import java.util.Scanner;

public class Main {
	static Scanner scanner = new Scanner(System.in);
	static long max = Long.MIN_VALUE;// 保存最大开心值， 可以保存19位数位
	static long min = Long.MAX_VALUE;// 保存最小开心值
	static long k = 0;
	static long n = 0;// 保存对象数字
	static long res = 0;
	static int len = 0;
	public static void main(String[] args) {
		// string = 1234
		String string = scanner.next();
		k = scanner.nextLong();
		n = Long.valueOf(string);
		// 构造表达式
		// 先填入首字母
		StringBuilder str = new StringBuilder(string.charAt(0)+"");
		
		//达到这种效果：1_2_3_4
		for (int i = 1; i < string.length(); i++) {
			str.append('_');// 用做占位符
			str.append(string.charAt(i));
		}
		len = str.toString().toCharArray().length;//7位

//		System.out.println(str);
		dfs(0,str);
//		System.out.println(max);
//		System.out.println(min);
		System.out.println(max-min);
		scanner.close();
	}

	public static void dfs(int cnt, StringBuilder str) {
		// 当达到k个+号时返回赋值最大值，最小值。结束递归
		if (cnt == k) {
			String[] strings = str.toString().replace("_","").split("\\+");// 需要添加转移字符
			res = 0;// 这一步不能省去
			for (String s : strings)
				res += Long.valueOf(s);
			max = Math.max(max, res);
			min = Math.min(min, res);
			return;// 结束递归

		}
		for (int i = cnt*2+1; i < len; i+=2) {
			//需要判断当前要替换的位置是否为“+”
            //如果没有该判断，就会存在1_2_3+4进入，cnt = 2的情况
			if (str.charAt(i) != '_')
				continue;
			str.replace(i, i+1, "+");//替换为插入加号
			// System.out.println(str.toString());
			// 递归
			dfs(++cnt, str);
			// 回溯
			cnt--;
			str.replace(i,i+1,"_");// 删除“+”
		}

	}
}
```

解法二：

【思路分析】

> 看注释

```java
import java.util.Scanner;

public class Main {

    static Long max=Long.MIN_VALUE,min=Long.MAX_VALUE;
    public static void main(String[] args) {
        // TODO Auto-generated method stub
        Scanner sc = new Scanner(System.in);
        char[] n;
        int k;
        n = sc.next().toCharArray();
        k=sc.nextInt();
        StringBuilder str = new StringBuilder();
        dfs(0,n,k,str);
        System.out.println(max-min);

    }
    //u代表加入的数字个数；k代表未插入的+数量 
    static void dfs(int u,char[] n,int k,StringBuilder str) {
        //已加入所有数字
        if(u==n.length) {
            if(k==0) {
                String[] s = str.toString().split("\\+");
                //计算表达式结果
                long res=0;
                for(String x:s) {
                    res+=Long.valueOf(x);
                }
                //求出最大和最小值
                max = Math.max(max, res);
                min = Math.min(min, res);
            }
            return ;
        }
        //加入当前数字
        str.append(n[u]);
        //不插入＋时的递归调用
        dfs(u+1,n,k,str);
        //回溯
        str.deleteCharAt(str.length()-1);
        //有+未插入且不为最后一位数
        if(k>0 && u<n.length-1) {
            //插入+时的递归调用
            str.append(n[u]);
            str.append('+');
            dfs(u+1, n, k-1, str);
            //回溯，删除+
            str.deleteCharAt(str.length()-1);
            //回溯，删除数字
            str.deleteCharAt(str.length()-1);
        }
    }

}
```
