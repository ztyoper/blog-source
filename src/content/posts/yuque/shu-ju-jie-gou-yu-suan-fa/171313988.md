### 试题1：帮派弟位

> ![image.png](https://pub-ccd5622eabb54b4a81678003778ddce1.r2.dev/PicGo/d86a3aeff87cb63c498b26fbb6c8a0fb0c6addb57b5986c8211ffbf592404c8d.png)

【思路分析】

> 注释即思路

```java
import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;
import java.util.StringTokenizer;

public class Main {
	static List<Integer>[] list;
	static int[] dp;
	public static void main(String[] args) {
		//共n人
		int n = nextInt();
		//小明的序号
		int m = nextInt();
		//找到根结点, f[i] > 0 表示它是一个结点的子结点。不可能是根节点。
		int[] f = new int[n+1];
		list = new ArrayList[n+1];
		for(int i = 0; i<=n; i++)
			list[i] = new ArrayList<>();
		for(int i = 1; i<=n-1; i++) {
			int l = nextInt();
			int r = nextInt();
			list[r].add(l);
			f[l]++;
		}
		//因为是树的结构，因此只可能有一个root
		int root = 0;
		for(int i = 1; i<=n; i++) {
			if(f[i] == 0) {
				root = i;
				break;
			}
		}
		
		//我们要知道小明这个序号m是排第几，我们需要知道每一个序号作为子树根结点时，子树的大小（手下帮众的数量）。
		dp = new int[n+1];
		dfs(root);
		//当前小明排在第一位
		int ans = 1;
		for(int i = 1; i<m; i++) {
			//帮众数大于小明的帮众数，或者相等时序号又大于小明的m
			if(dp[i] >= dp[m]) {
				ans++;
			}
		}
		for(int i = m+1; i<=n; i++) {
			if(dp[i] > dp[m]) {
				ans++;
			}
		}
		System.out.println(ans);
	}

	static void dfs(int x) {
		//为了方便，我们不可能将dp[x]= 0，就只好设置为1
//		dp[x] = 1;
		for(int y : list[x]) {
			dfs(y);
			dp[x] += dp[y] + 1;
		}
	}
	static BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));

	static StringTokenizer token = new StringTokenizer("");

	static String next() {
		while (!token.hasMoreTokens()) {
			try {
				token = new StringTokenizer(reader.readLine());
			} catch (IOException e) {
				// TODO Auto-generated catch block
				e.printStackTrace();
			}
		}
		return token.nextToken();
	}

	static int nextInt() {
		return Integer.parseInt(next());
	}

}
```

```java
import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.LinkedList;
import java.util.List;
import java.util.Queue;
import java.util.StringTokenizer;

public class Main {
	static List<Integer>[] list;

	public static void main(String[] args) {
		// 总共有n个城市，并且我们的目的地也是城市n
		int n = nextInt();
		// 总共有m条无向边
		int m = nextInt();
		list = new ArrayList[n + 1];
		for (int i = 0; i <= n; i++) {
			list[i] = new ArrayList<>();
		}
		for (int i = 1; i <= m; i++) {
			int x = nextInt();
			int y = nextInt();
			list[x].add(y);
			list[y].add(x);
		}
		
		long mod = (long)1e9 + 7;
		// 我们使用dis[i]表示城市i到城市1的最短距离。城市1是我们的起点城市
		// 求得每一个城市到城市1的最短距离， dis初次被赋值时，dis就是对应节点的最短距离，
        //想像上图一层一层地被访问，第一次被访问到就是该节点到节点1的最短距离。
		long[] dis = new long[n + 1];
		dis[1] = 0;// 城市1到城市1距离为0
		Arrays.fill(dis,-1);
		// 我们使用dp[i]表示城市i到城市1的方案数。
		long[] dp = new long[n + 1];
		//城市1到城市1的方案数只有1种。
		dp[1] = 1;
		Queue<Integer> q = new LinkedList<>();
		q.add(1);
		while(q.size()>0) {
			int u = q.poll();
			for(int v : list[u]) {
				if(dis[v] == -1) {
					dp[v] = dp[u];
					//每条边权重都是1
					dis[v] = dis[u] + 1;
					q.add(v);
				}else {
                    //统计所有最短路径数量,也就初次访问过的终点假设dis[n] = 8
                    //另一条路径再走1到节点n，如果刚好这个距离dis[u] + 1 == dis[v]最短距离,我们就将这条路径的方案数dp[u]加上
					if(dis[v] == dis[u] + 1) {
						dp[v] +=dp[u];
						dp[v] %=mod;
					}
				}
			}
		}
		System.out.println(dp[n]);
	}

	static BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));

	static StringTokenizer token = new StringTokenizer("");

	static String next() {
		while (!token.hasMoreTokens()) {
			try {
				token = new StringTokenizer(reader.readLine());
			} catch (IOException e) {
				// TODO Auto-generated catch block
				e.printStackTrace();
			}
		}
		return token.nextToken();
	}

	static int nextInt() {
		return Integer.parseInt(next());
	}

}
```
