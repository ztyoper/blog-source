#### 优化写法

> 控制树的高度。

```java
import java.io.*;
import java.util.*;

/*
输入：
6 6
0 1
1 2
1 3
2 4
2 5
3 4
 */
public class Main {
    //邻接表实现存储边
    static List<Integer>[] list;

    //当前结点i的根结点是j, parent[i] = j
    static int[] parent;

    //控制树的高度。rank[i] = j, 表示节点i所在的树的高度为j
    static int[] rank;

    public static void main(String[] args) throws IOException {
        //结点的个数
        int n = nextInt();
        //边的个数
        int m = nextInt();
        rank = new int[n];
        parent = new int[n + 1];
        Arrays.fill(parent, -1);

        list = new ArrayList[n + 1];
        for (int i = 0; i <= n; i++) list[i] = new ArrayList<>();

        for (int i = 0; i < m; i++) {
            int x = nextInt();
            int y = nextInt();
            //无向边
            list[x].add(y);
            list[y].add(x);
        }

        //判断图中是否有环
        Queue<Integer> q = new LinkedList<>();

        //先遍历结点1
        dfs(1, -1);
        System.out.println("\n图中不存在环");
    }

    static void dfs(int x, int pa) {
        for (int y : list[x]) {
            if (y != pa) {
                if (!union(y, x)) {
                    System.out.println("图中存在环");
                    System.exit(0);
                }
                dfs(y, x);
            }

        }
    }

    /*
    合并成一个集合的操作
     */
    static boolean union(int x, int y) {
        int x_root = find_root(x);
        int y_root = find_root(y);
        if (x_root != y_root) {

            if (rank[x] > rank[y]) {
                //将y_root结点作为x_root结点的子结点
                parent[y_root] = x_root;
            } else if (rank[y] > rank[x]) {
                //将x_root结点作为y_root结点的子结点
                parent[x_root] = y_root;
            } else {//高度相同
                //将x_root结点作为y_root结点的子结点
                parent[x_root] = y_root;
                rank[y]++;
            }
            //合并成功，说明没有环
            return true;
        }
        //合并失败，说明存在环
        return false;
    }

    /*
    找到x结点的根结点是谁.每一个结点首先的parent都是-1，标志着都是独立结点
     */
    static int find_root(int x) {
        while (parent[x] != -1) x = parent[x];
        return x;
    }

    static int nextInt() throws IOException {
        token.nextToken();
        return (int) token.nval;
    }
}
```

```java
import java.io.*;
import java.util.*;


public class Main {

    //当前结点i的根结点是j, parent[i] = j
    static int[] parent;

    //控制树的高度。rank[i] = j, 表示节点i所在的树的高度为j
    static int[] rank;

    public static void main(String[] args) throws IOException {
        //结点的个数
        int n = nextInt();
        //操作个数
        int m = nextInt();
        rank = new int[n + 1];
        parent = new int[n + 1];
        Arrays.fill(parent, -1);

        for (int i = 0; i < m; i++) {
            int opt = nextInt();
            int x = nextInt();
            int y = nextInt();
            switch (opt) {
                case 1:
                    union(x, y);
                    break;
                case 2:
                    int x_root = find_root(x);
                    int y_root = find_root(y);
                    if (x_root != y_root) System.out.println("NO");
                    else System.out.println("YES");
                    break;
            }
        }
    }

    /*
    合并成一个集合的操作
     */

        int x_root = find_root(x);
        int y_root = find_root(y);
        if (x_root != y_root) {
            if (rank[x] > rank[y]) {
                //将y_root结点作为x_root结点的子结点
                parent[y_root] = x_root;
            } else if (rank[y] > rank[x]) {
```
