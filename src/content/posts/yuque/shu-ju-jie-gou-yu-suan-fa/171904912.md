# 迭代深搜IDA*

> ID(iterate deep)：迭代深搜
> A*：估价函数，用于剪枝的。

### 试题1：糖果

【思路分析】

> ![image.png](https://pub-ccd5622eabb54b4a81678003778ddce1.r2.dev/PicGo/feb306a2b62f7add8db3c1ae81e1fea924165f70826bb39cf63aa95cf362549a.png)我们发现，只需要买第2包和第5包就可以覆盖所有的糖果种类。其实这种题是重复覆盖类型（需要使用Dancing Links 跳动链表来解决。但是我们这里使用`迭代深搜+估价函数`IDA*来解决）力度就是2^20 = 10^5
> 迭代深搜（按照每列进行搜索，也就是限制特定层数的广度搜素）：我们先只搜0层，行就退出循环，并返回0包。只搜1层，行就退出循环，并返回1包。只搜2层，行就退出循环，并返回2包。只搜3层，行就退出循环，并返回3包....只搜20层，行就退出循环，并返回20包，这还不行的话，就返回-1,因为遍历完所有列都无法覆盖20种糖果。
> 估价函数：只搜10层，行就退出循环，并返回10包-&gt;搜到第5层时，我们调用估价函数h()，得到至少还要搜索6层才能覆盖全部种类，这时5 + 6 &gt; 10 ， 就提前退出搜素。

#### 解法一：朴素解法

```java
import java.io.BufferedInputStream;
import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.io.StreamTokenizer;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;

public class Main {
	static int n;
	static int m;
	static int k;
	//pack[i] = 10101 第i包有哪些糖果：第i行有哪些为1
	static int[] pack;
	//candy[2] = {1,2,4,5,6},第2列有对应1，2，4，5，6行是1.
	static List<Integer>[] candy;
	public static void main(String[] args) throws IOException {
		n = nextInt();
		m = nextInt();
		k = nextInt();
		//我们从索引0开始,因为设计到位运算
		pack = new int[n];
		candy = new ArrayList[m];
		boolean[] st = new boolean[n];
		for(int i = 0; i<m; i++)candy[i] = new ArrayList<>();
		for(int i = 0; i<n; i++) {
			//不能忘
			Arrays.fill(st, false);
			for(int j = 0; j<k; j++) {
				int t = nextInt();
				//有些行2 1 1 ，一包3颗糖，一个2种类，两个1种类。我们要对其去重，加排序。
				st[t - 1] = true;
			}
			for(int j = 0; j<m; j++) {
				if(st[j]) {
					pack[i] += 1 << j;
					//不会出现重复，因为j在变化
					candy[j].add(i);
				}
			}
		}
		
		int t = 0; 
		//迭代深搜
		for(t = 0; t<=m; t++) {
			if(dfs(t,0))break;
		}
		System.out.println(t > m ? -1:t);
		
	}
	/**
	 * 
	 * @param u 迭代深搜的层数
	 * @param state 当前状态
	 * @return
	 */
	static boolean dfs(int u , int state) {
		if(u == 0) {
			//全为1就返回true
			if(state == (1 << m) - 1) return true;
			return false;
		}
		int c = -1;
		//选择一列，该列的1最少，后面遍历candy列表时，会更快。
		for(int i = 0; i<m; i++) {
			//state >> i & 1 就是看第i位是否是1，若不是1的话，进入判断
			if((state >> i & 1) == 0) {
				if(c == -1 || candy[c].size() > candy[i].size())
					c = i;
			}
		}
		//p代表行.
		//看图，t = 2时，开始c = 3, 图中第4列
		for(int p : candy[c]) {
			//一次只买一包，因此需要进入下一层迭代
			if(dfs(u-1, state | pack[p]))return true;
		}
		return false;
	}

	static StreamTokenizer token = new StreamTokenizer(
			new BufferedReader(new InputStreamReader(new BufferedInputStream(System.in))));

	static int nextInt() throws IOException {
		token.nextToken();
		return (int) token.nval;
	}

}
```

#### 解法二：优化一

```java
import java.io.BufferedInputStream;
import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.io.StreamTokenizer;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;

public class Main {
	
	
	static int n;
	static int m;
	static int k;
	// pack[i] = 10101 第i包有哪些糖果：第i行有哪些为1
	static int[] pack;
	// candy[2] = {1,2,4,5,6},第2列有对应1，2，4，5，6行是1.
	static List<Integer>[] candy;

	public static void main(String[] args) throws IOException {
		n = nextInt();
		m = nextInt();
		k = nextInt();
		// 我们从索引0开始,因为设计到位运算
		pack = new int[n];
		candy = new ArrayList[m];
		boolean[] st = new boolean[n];
		for (int i = 0; i < m; i++)
			candy[i] = new ArrayList<>();
		for (int i = 0; i < n; i++) {
			// 不能忘
			Arrays.fill(st, false);
			for (int j = 0; j < k; j++) {
				int t = nextInt();
				// 有些行2 1 1 ，一包3颗糖，一个2种类，两个1种类。我们要对其去重，加排序。
				st[t - 1] = true;
			}
			for (int j = 0; j < m; j++) {
				if (st[j]) {
					pack[i] += 1 << j;
					// 不会出现重复，因为j在变化
					candy[j].add(i);
				}
			}
		}

		int t = 0;
		// 迭代深搜
		for (t = 0; t <= m; t++) {
			if (dfs(t, 0))
				break;
		}
		System.out.println(t > m ? -1 : t);

	}
	
	/**
	 * 估价函数
	 * 使用h(0)来算这道题，肯定是错的，因为for(int p : candy[i])是买了多包，但res只加1
	 * @param state
	 * @return
	 */
	static int h(int state) {
		int res = 0;
		for (int i = 0; i < m; i++) {
			if ((state >> i & 1) == 0) {
				res++;
				//比如：第2列，我将1，2，4，5，6包都选入，但我只算1包。最终这个“至少”res包坑定是比较小的。
				//如果我这么捡大便宜的情况下 res仍然大于剩下迭代层数，那么就可以不用继续迭代了，直接退出吧。
				//
				for (int p : candy[i]) {
					state |= pack[p];
				}
			}
		}
		return res;
	}

	/**
	 * 
	 * @param u     迭代深搜的层数
	 * @param state 当前状态
	 * @return
	 */
	static boolean dfs(int u, int state) {
		// u < h(state)看h函数注释
		if (u == 0 || u < h(state)) {
			// 全为1就返回true
			if (state == (1 << m) - 1)
				return true;
			return false;
		}
		int c = -1;
		// 选择一列，该列的1最少，后面遍历candy列表时，会更快。
		for (int i = 0; i < m; i++) {
			// state >> i & 1 就是看第i位是否是1，若不是1的话，进入判断
			//c最终就是state中还不为1的列，且该列中的candy还是最少的。
			if ((state >> i & 1) == 0) {
				if (c == -1 || candy[c].size() > candy[i].size())
					c = i;
			}
		}
		// p代表行.
		// 看图，t = 2时，开始c = 3, 图中第4列
		for (int p : candy[c]) {
			// 一次只买一包，因此需要进入下一层迭代
			if (dfs(u - 1, state | pack[p]))
				return true;
		}
		return false;
	}



	static StreamTokenizer token = new StreamTokenizer(
			new BufferedReader(new InputStreamReader(new BufferedInputStream(System.in))));

	static int nextInt() throws IOException {
		token.nextToken();
		return (int) token.nval;
	}

}
```

#### 解法三：优化二

> acwing上，发现会超时，没有优化一快，唉。。。

```java
import java.io.BufferedInputStream;
import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.io.StreamTokenizer;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;

public class Main {

	static int n;
	static int m;
	static int k;
	// pack[i] = 10101 第i包有哪些糖果：第i行有哪些为1
	static int[] pack;
	// candy[2] = {1,2,4,5,6},第2列有对应1，2，4，5，6行是1.
	static List<Integer>[] candy;

	// 用作优化二，lowbit(10100100) = 100, x & -x， f[100] = 2 f[1000] = 3, 没错就是存储1的位置（列数）
	static int[] f;

	public static void main(String[] args) throws IOException {
		n = nextInt();
		m = nextInt();
		k = nextInt();
		// 我们从索引0开始,因为设计到位运算
		pack = new int[n];
		candy = new ArrayList[m];
		f = new int[(1<<(m-1))+1];
		boolean[] st = new boolean[n];
		for (int i = 0; i < m; i++)
			f[1 << i] = i;
		for (int i = 0; i < m; i++)
			candy[i] = new ArrayList<>();
		for (int i = 0; i < n; i++) {
			// 不能忘
			Arrays.fill(st, false);
			for (int j = 0; j < k; j++) {
				int t = nextInt();
				// 有些行2 1 1 ，一包3颗糖，一个2种类，两个1种类。我们要对其去重，加排序。
				st[t - 1] = true;
			}
			for (int j = 0; j < m; j++) {
				if (st[j]) {
					pack[i] += 1 << j;
					// 不会出现重复，因为j在变化
					candy[j].add(i);
				}
			}
		}

		int t = 0;
		// 迭代深搜
		for (t = 0; t <= m; t++) {
			if (dfs(t, 0))
				break;
		}
		System.out.println(t > m ? -1 : t);

	}

	/**
	 * 估价函数 使用h(0)来算这道题，肯定是错的，因为for(int p : candy[i])是买了多包，但res只加1
	 * 
	 * @param state
	 * @return
	 */
	static int h(int state) {
		int res = 0;
		for (int s = (1 << m) - 1 - state; s != 0; s -= s & -s) {
			int i = f[s & -s];
			res++;
			// 比如：第2列，我将1，2，4，5，6包都选入，但我只算1包。最终这个“至少”res包坑定是比较小的。
			// 如果我这么捡大便宜的情况下 res仍然大于剩下迭代层数，那么就可以不用继续迭代了，直接退出吧。
			//
			for (int p : candy[i]) {
				// 之前是state |= pack[p]，将state中1的个数变多，现在是将s中0的个数变多。
				s &= ~pack[p];
			}
		}
		return res;

	}

	/**
	 * 
	 * @param u     迭代深搜的层数
	 * @param state 当前状态
	 * @return
	 */
	static boolean dfs(int u, int state) {
		// u < h(state)看h函数注释
		if (u == 0 || u < h(state)) {
			// 全为1就返回true
			if (state == (1 << m) - 1)
				return true;
			return false;
		}
		int c = -1;
		// 选择一列，该列的1最少，后面遍历candy列表时，会更快。
		// s为state的反数，如：state = 1101, s就等于 0010，这样使用f[lowbit(s)]找到的就是state中0的位置了。
		// 相比于之前固定循环m（20）次，这时会越循环，循环次数越少
		for (int s = (1 << m) - 1 - state; s != 0; s -= s & -s) {
			int i = f[s & -s];
			// c最终就是state中还不为1的列，且该列中的candy还是最少的。
			if ((state >> i & 1) == 0) {
				if (c == -1 || candy[c].size() > candy[i].size())
					c = i;
			}
		}

		// p代表行.
		// 看图，t = 2时，开始c = 3, 图中第4列
		for (int p : candy[c]) {
			// 一次只买一包，因此需要进入下一层迭代
			if (dfs(u - 1, state | pack[p]))
				return true;
		}
		return false;
	}

	static StreamTokenizer token = new StreamTokenizer(
			new BufferedReader(new InputStreamReader(new BufferedInputStream(System.in))));

	static int nextInt() throws IOException {
		token.nextToken();
		return (int) token.nval;
	}

}
```
