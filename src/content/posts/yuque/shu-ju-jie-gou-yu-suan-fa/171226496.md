### 试题1：求完全背包问题

#### 解法一：朴素算法

【思路分析】

> ![0189685bc03a98f115aaf7418509cea5.jpg](https://pub-ccd5622eabb54b4a81678003778ddce1.r2.dev/PicGo/272da7f5a1e3f66b3cc7d4f30d63bfa682a7d5f3d4d8ca0df5075c7a7af23d3c.jpg)
> 如果f(i,j-w[i])，j&lt;w[i]的话，相当于一个都装不下，也就没有右边的所有子集。

```java
import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.StringTokenizer;

public class Main {
	public static void main(String[] args) throws Exception{
		int total = nextInt();
		int n = nextInt();
		//我们要有i-1的迭代，因此需要索引从1开始
		int[] w = new int[n+1];// weight
		int[] v = new int[n+1];// value
		int[][] f = new int[n+1][total+1];
		for(int i = 1; i<=n; i++) {
			w[i] = nextInt();
			v[i] = nextInt();
			
		}
		//先遍历背包，再遍历物品
		for(int i = 1; i<=n ;i++) {
			for(int j = 1; j<=total; j++) {
				//因为是求最值，因此直接赋值左半边的子集.
				
				f[i][j] = f[i-1][j];
				//如果判断失败，那么f[i][j]就没有右边所有的子集合
				if(j>=w[i]) f[i][j] = Math.max(f[i][j], f[i][j-w[i]] + v[i]);
			}
		}
		System.out.println(f[n][total]);
		
	}
	static BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
	static StringTokenizer token = new StringTokenizer("");
	static String next() throws IOException {
		while(!token.hasMoreElements()) {
			token = new StringTokenizer(reader.readLine());
		}
		return token.nextToken();
	}
	static int nextInt() throws NumberFormatException, IOException {
		return Integer.parseInt(next());
	}
}
```

#### 解法二：优化算法

```java
import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.StringTokenizer;

public class Main {
	public static void main(String[] args) throws Exception{
		int total = nextInt();
		int n = nextInt();
		//我们要有i-1的迭代，因此需要索引从1开始
		int[] w = new int[n+1];// weight
		int[] v = new int[n+1];// value
		int[] f = new int[total+1];
		for(int i = 1; i<=n; i++) {
			w[i] = nextInt();
			v[i] = nextInt();
			
		}
		//先遍历背包，再遍历物品
		for(int i = 1; i<=n ;i++) {
			for(int j = 1; j<=total; j++) {
				
				f[j] = f[j];
				//当i = 1时，我们初始化了f数组
				//当i = 2时，此时我们使用f[j] = f[j]就相当于 用上一层初始化的结果来给这一层赋值，也就是等价于f[i][j] = f[i-1][j]
				if(j>=w[i]) f[j] = Math.max(f[j], f[j-w[i]] + v[i]);
				//当i等于某一个值时，也就是同一层时
				//内层循环从小到大，因此f[j-w[i]]会先算出来，f[j]会后算出来。正是f[j-w[i]]先算出来，因此目前赋值f[j]时f[j-w[i]]的值是当前一层算出来的，f[j-w[i]]是 i 层的值。
				//上式子就等价为f[i][j] = Math.max(f[i][j], f[i][j-w[i]]+v[i])
			}
		}
		System.out.println(f[total]);
		
	}
	static BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
	static StringTokenizer token = new StringTokenizer("");
	static String next() throws IOException {
		while(!token.hasMoreElements()) {
			token = new StringTokenizer(reader.readLine());
		}
		return token.nextToken();
	}
	static int nextInt() throws NumberFormatException, IOException {
		return Integer.parseInt(next());
	}
}
```
