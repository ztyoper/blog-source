# 第六届校园千峰杯

### 试题一：斐波那契数列【编程题】

> 时间限制: 400ms 内存限制:<span style="color: var(--color)">64 MB</span>本题总分： 20 分

**<span style="color: var(--cm-text-color-regular)">【问题描述】</span>**

> 斐波那契数列:1 1 2 3 5 8 13 ...... 求指定项的值

**<span style="color: var(--cm-text-color-regular)">【输入格式】</span>**

> <span style="color: rgb(64, 64, 64)">第一行输入一个正整数T，表示测试组数。（1≤T≤1000）</span><span style="color: rgb(64, 64, 64)">接下来n行每行输入一个整数n，表示第n项。（1≤n≤10000）</span>

**<span style="color: var(--cm-text-color-regular)">【输出格式】</span>**

> <span style="color: rgb(64, 64, 64)">每行输出一个整数表示第n项的值</span>

**<span style="color: var(--cm-text-color-regular)">【样例输入】</span>**

> <span style="color: rgb(64, 64, 64)">2</span>
> <span style="color: rgb(64, 64, 64)">2</span>
> <span style="color: rgb(64, 64, 64)">5</span>

**<span style="color: var(--cm-text-color-regular)">【样例输出】</span>**

> 1
> 5

**<span style="color: var(--cm-text-color-regular)">【评测用例规模与约定】</span>**

> <span style="color: var(--cm-text-color-regular)">对于 100 %的评测用例，</span>$1 \le T \le 1000 ,1 \le n \le 10000$

**<span style="color: var(--cm-text-color-regular)">解法：</span>**

```cpp
#include <iostream>
#include <cmath>
#include <string.h>

#include <chrono>
#include <time.h>
using namespace std;
//解决题目要求
void methodOne()
{
    long long numA[1000] = {0};
    long long numB[1000] = {0};
    long long curNum[1000] = {0};
    int m = 1000;
    int n = 3;
    int bits = 8;
    numA[0] = 1;
    numB[0] = 1;
    long long tempNum = pow(10, bits);
    cout << tempNum << endl;
    cin >> n;
    if (n <= 0)
    {
        cout << 0 << endl;
    }
    else if (n == 1 || n == 2)
    {
        cout << 1 << endl;
    }
    else
    {
        for (int i = 3; i <= n; i++)
            {
                //该数组每一个int都能保存10^8次方大小的数。超过1e8则进一位
                curNum[0] = (numA[0] + numB[0]) % tempNum;
                for (int j = 1; j < m; j++)
                    {
                        // cout << ((numA[j - 1] + numB[j - 1]) / tempNum) << endl;
                        curNum[j] = ((numA[j - 1] + numB[j - 1]) / tempNum + numA[j] + numB[j]) % tempNum;
                    }
                for (int j = 0; j < m; j++)
                    {
                        numA[j] = numB[j];
                        numB[j] = curNum[j];
                    }
            }
        for (int i = m - 1; i >= 0; i--)
            {
                if (curNum[i] != 0)
                {
                    cout << curNum[i];
                }
            }
    }
}
//大数相加
void methodTwo()
{

    char s[10000], ss[10000]; // 这里的范围我们设大一些，最多就放10000个数（10的10000方）
    int a[10000], b[10000];   // 这里用来进行数学运算，并将最后的结果储存在a[10000]中
    int len;
    scanf("%s", s); // 输入要计算的2个数字
    scanf("%s", ss);
    int len1 = strlen(s);  //"strlen"是一个计算字符串长度的函数
    int len2 = strlen(ss); // 将输入的两个字符串的长度赋值给len1,len2
    if (len1 > len2)
        len = len1; // 将len赋值为len1,len2中大的那个
    else
        len = len2;
    memset(a, 0, sizeof(a)); // 这是用内存设置函数置零字符串
    memset(b, 0, sizeof(b)); // 这是用内存设置函数置零字符串
    // 两个for循环是将输入的两个字符串倒过来，再将字符串里的字符转换为数字赋值给a,b整型数组
    for (int i = len1 - 1; i >= 0; i--)
        a[len1 - i - 1] = s[i] - '0';
    for (int i = len2 - 1; i >= 0; i--)
        b[len2 - i - 1] = ss[i] - '0';
    // 这里将上面逆序的数字进行相加再进行进位
    for (int i = 0; i < len; i++)
        {
            a[i] = a[i] + b[i];    // 运算
            a[i + 1] += a[i] / 10; // 如有进位，在后一位上加上
            a[i] = a[i] % 10;      // 原来那一位减掉进位了的
        }
    if (a[len] != 0)                   // 如果有进位就多显示一位
        len++;                         // 位数就加一
    while (a[len - 1] == 0 && len > 1) // while去零法，将多余是0的位数去掉
        len--;
    for (int i = len - 1; i >= 0; i--) // 再逆序输出结果
        printf("%d", a[i]);
    printf("\n");
}



int main()
{
    clock_t start = clock();     //clock()返回单位是毫秒
    //计时程序
    methodOne();
    clock_t end = clock();
    cout << "花费了" << (double)(end - start) / CLOCKS_PER_SEC << "秒" << endl;  //返用秒为单位可以用CLOCKS_PER_SEC
    cout << "花费了" << (double)(end - start) << "毫秒" << endl;
    // sleep(10);  //让正在执行的线程等待10毫秒
}
```
