# 记忆化搜索

### 试题一: 斐波那契数列

> 题目大意：
> 斐波那契数列的定义为F(n)=F(n-1)+F(n-2),同时F1=1,F2=1.请你输出数列的第n个数的对1e9+7取模的值。

【思路分析】

> 使用递归搜索，不过相对于传统的递归到dp[1] 和dp[2],我们使用记忆化搜索的方式。当我们求dp10时，首先深度递归一条链，将所有链上的节点都赋初值之后，后续递归就可以利用这些初值。
> ![fcc1d0ae8d78a99e5d0b0233a4d3a51f.jpg](https://pub-ccd5622eabb54b4a81678003778ddce1.r2.dev/PicGo/12c8d1607231ef326679986c85b7095dcf4414527ad281384469e359099480e0.jpg)

```cpp
import java.util.Arrays;
import java.util.Scanner;

public class Main {
	static Scanner scanner = new Scanner(System.in);
	static long[] dp;
	static long mod = (long)1e9 + 7;
	public static void main(String[] args) {
		int n  = scanner.nextInt();
		dp = new long[n+1];
		Arrays.fill(dp,-1);
		dp[1] = dp[2] = 1;
		System.out.println(dfs(n));
	}
	public static long dfs(int n ) {
		if(dp[n] != -1) {
			return dp[n];
		}
		long ans = dfs(n-1) + dfs(n-2);
		ans%=mod;
		dp[n] = ans;
		return dp[n];
	}

}
```

```cpp
import java.math.BigInteger;
import java.util.Arrays;
import java.util.Scanner;

public class Main {
	static Scanner scanner = new Scanner(System.in);
	static BigInteger[] dp;
	static long mod = (long)1e9 + 7;
	public static void main(String[] args) {
		int n  = scanner.nextInt();
		dp = new BigInteger[n+1];
		for(int i = 0 ;i<=n; i++) {
			dp[i] = BigInteger.valueOf(-1);
		}
		dp[1] = dp[2] = BigInteger.valueOf(1);
		System.out.println(dfs(n));
	}
	public static BigInteger dfs(int n ) {
		if(dp[n].compareTo(BigInteger.valueOf(-1)) != 0) {
			return dp[n];
		}
		BigInteger ans = dfs(n-1).add(dfs(n-2));
		ans.mod(BigInteger.valueOf(mod));
		dp[n] = ans;
		return dp[n];
	}

}
```

### 试题二: 红与黑

![image.png](https://pub-ccd5622eabb54b4a81678003778ddce1.r2.dev/PicGo/94485fb024b0b0c648d73ae99452f9d17d1b36d2f3c95a34c507871a2a2a66d3.png)

【输入样例】

> 9 6
> ....#.
> .....#
> ......
> ......
> ......
> ......
> ......
> #@...#
> .#..#.

【输出样例】

> 45

【思路分析】

> 使用递归，当我们深度递归出一条路径之后，将路径上所有点都置为红色。证明此地砖已经被采过了。然后后续递归就会避免重复踩这些地砖。

```java
import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

public class Main {
	static Scanner scanner = new Scanner(System.in);
	static int ans = 0;// 保存答案

	public static void main(String[] args) {
		int n = scanner.nextInt();
		int m = scanner.nextInt();
		char[][] c = new char[n][m];
		for (int i = 0; i < n; i++) {
			c[i] = scanner.next().toCharArray();// 将一行的字符串变为字符数组。使用next是因为没有空格
		}
		for (int i = 0; i < n; i++) {
			for (int j = 0; j < m; j++) {
				if (c[i][j] == '@') {
					dfs(i, j, c);
				}
			}
		}
		System.out.println(ans);

	}

	public static void dfs(int x, int y , char[][] c) {
		ans++;
		c[x][y] = '#';//变为红色地砖
		int[] dx = {1,-1,0,0};
		int[] dy = {0,0, 1,-1};
		for(int i = 0; i<4; i++) {
			int x1 = x + dx[i];
			int y1 = y + dy[i];
            //c[x1][y1] != '.'是记忆化的体现
			if(x1<0 || x1>=c.length || y1 < 0 || y1>=c[0].length || c[x1][y1] != '.')continue;
			dfs(x1,y1,c);
		}
	}
}
```

### 试题三: 混沌之地5

> ![image.png](https://pub-ccd5622eabb54b4a81678003778ddce1.r2.dev/PicGo/371e6ea5709af3008cc33a26b2c93831c71af1c70921a8a35f6c01ef18fd68d2.png)

【思路分析】

> 思路：用记忆化数组dp[i][j]表示(i,j)是否能被搜索到。在搜索的过程中，我们要维护当前状态是否使用了喷气背包。flag=1表示没使用，flag=0表示使用过了。最后判断dp[c][d]是否等于1即可。

```cpp
import java.math.BigInteger;
import java.util.Arrays;
import java.util.Scanner;

public class Main {
	static Scanner scanner = new Scanner(System.in);
	static int[][] f = null;
	static int[][] dp = null;
	static int n ; 
	static int m;
	static int k;
	public static void main(String[] args) {
		 n = scanner.nextInt();
		 m = scanner.nextInt();
		 k = scanner.nextInt();
		int a = scanner.nextInt() - 1;
		int b = scanner.nextInt() - 1;
		int c = scanner.nextInt() - 1;
		int d = scanner.nextInt() - 1;
		boolean flag = true;//是否还能使用背包
		f = new int[n][m];
		dp = new int[n][m];// 如果深度搜索能到达，则置为非-1
//		for (int i = 0; i < n; i++) {
//			Arrays.fill(dp[i], -1);
//		}
		for (int i = 0; i < n; i++) {
			for (int j = 0; j < m; j++) {
				f[i][j] = scanner.nextInt();
				dp[i][j] = -1;//将Arrays.fill转换为这一步
			}
		}
		dfs(a,b,flag);
		if (dp[c][d] == 1) {
			System.out.println("Yes");
		} else {
			System.out.println("No");
		}
	}
	//
	//int[][] f可以是static的，因为地图不会改变。int[][] dp可以是static的。因为当两种路径都能到达一个点(x0,y0)，dp[x0][y0] = 1,两种走法不会相互影响。
	//而且一直走法将该点置为1后，另一种走法就可以不走这个点了，因为前一种走法已经遍历出走过改点后的情况了。
	public static void dfs(int x, int y, boolean flag) {
		dp[x][y] = 1;
		int[] dx = { 1, -1, 0, 0 };
		int[] dy = { 0, 0, 1, -1 };
		for (int i = 0; i < 4; i++) {
			int x1 = x + dx[i];
			int y1 = y + dy[i];
            //dp[x1][y1] == 1是记忆化的体现
			if (x1 < 0 || x1 >= f.length || y1 < 0 || y1 >= f[0].length || dp[x1][y1] == 1)
				continue;

            //我们从四个方向选中一个方向后。有两种选择。：能走就走（继承flag），不能走（看看使用背包能走不）
			//我们使用深度搜索进行所有路径的搜索，flag必须不能是static的，必须是私有的
			if(f[x][y] > f[x1][y1]) {
				dfs(x1,y1,flag);//flag 一开始是true
			}else {
				//使用背包
				if(flag && k+f[x][y] > f[x1][y1]) {
					/*
					 * 有for循环的情况下，不能flag = false 赋值然后形参传入 
					 * 因为flag = false后，后面for循环的其他路径还可以使用背包，
					 * flag必须为true
					 */
					dfs(x1,y1,false);
				}
			}
		}
	}
}
```

### 试题四: 地宫取宝

> ![image.png](https://pub-ccd5622eabb54b4a81678003778ddce1.r2.dev/PicGo/9811f02dedfc1305cb520beb432817ebc63e6f90803e69a930d3e9bae119bfde.png)

【思路分析】

> 定义记忆化数组dp[i][j][max][count]表示位置i,j拿了count个宝贝，最大价值为max的方案数。按照题意dfs搜索即可

```java
import java.math.BigInteger;
import java.util.Arrays;
import java.util.Scanner;

public class Main {
	static Scanner scanner = new Scanner(System.in);
	static int[][] f = null;
	static long mod = (long) 1e9 + 7;
	// dp[x][y][max手中最大价值的宝物][手中目前有多少宝物] = ?多少中方案
	static long[][][][] dp;
	static int k;
	static int m;
	static int n;

	public static void main(String[] args) {
		n = scanner.nextInt();
		m = scanner.nextInt();
		k = scanner.nextInt();

		f = new int[n][m];
		for (int i = 0; i < n; i++) {
			for (int j = 0; j < m; j++) {
				f[i][j] = scanner.nextInt();
			}
		}
		dp = new long[n][m][13][k+1];// 13表示能填入最大是价值12的宝物
		for (int i = 0; i < n; i++) {
			for (int j = 0; j < m; j++) {
				for (int l = 0; l < 13; l++) {// 这一层是
					Arrays.fill(dp[i][j][l], -1);
				}
			}
		}
		//max初始值必须为-1，因为宝物的价值可以为0，而宝物的价值大于手中价值才能拿起。
		long res = dfs(0, 0, -1, 0);
		System.out.println(res);
	}

	public static long dfs(int x, int y, int max, int cnt) {
		if (x >= n || y >= m || cnt > k)
			// 越界，返回0方案
			return 0;
		// 记忆化搜索的体现
		//max初始值为-1，因此必须max+1,当然这样也让dp中东西的最大价值变为max+1，
		//然而不影响，因为我们比较时还是用的Max，而且不需要输出最大价值
		if (dp[x][y][max+1][cnt] != -1)
			return dp[x][y][max+1][cnt];
		

		if (x == n - 1 && y == m - 1 && cnt == k - 1 && f[x][y] > max)
			// 当到达出口方格时，出口方格上的宝物在此时需要拿起。这时，一种方案诞生。
			return 1;
		if (x == n - 1 && y == m - 1 && cnt == k)
			// 当到达出口方格时，出口方格上的宝物在此时不能拿起。这时，一种方案诞生。
			return 1;

		// 当前方格宝物的价值大于最大价值
        //相当于就是递归遍历所有情况
		long res = 0;
		if (f[x][y] > max) {
			res += dfs(x + 1, y, f[x][y], cnt + 1);// 向右走，并拿起宝物
			res += dfs(x, y + 1, f[x][y], cnt + 1);// 向下走，并拿起宝物
            //最大价值传承下去，就算是-1，但上述操作也有max+1保证非负
			res += dfs(x + 1, y, max, cnt);// 向右走，不拿宝物
			res += dfs(x, y + 1, max, cnt);// 向下走，不拿宝物
		} else {
			res += dfs(x + 1, y, max, cnt);// 向右走，不拿宝物,因为也不能拿
			res += dfs(x, y + 1, max, cnt);// 向下走，不拿宝物，因为也不能拿
		}
		res %= mod;
		dp[x][y][max+1][cnt] = res;
		return res;

	}
}
```
