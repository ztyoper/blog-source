# KMP匹配算法

### 试题一: KMP实现

【思路分析】

> KMP是通过已经匹配成功的那部分子串中的最大相同前后缀长度进行移动的。注意：找的是 最长 前后缀**子**串，如果将本身算入的话，子串匹配的时候就不移动了。

> ![image.png](https://pub-ccd5622eabb54b4a81678003778ddce1.r2.dev/PicGo/263fab26747fe629773c9e65f4203811bb8713fde471eba7c1f74dc35016a68a.png)
> 上述为next数组推导过程，其实j可以=t.length,也就是j&lt;t.length不用减1。这样整个串也有最大前后缀了。getNextVal 方法会出现越界问题。因为 j++ 后里面还有一个 if 判断比较。就越界了。getNext 就不会。

```java
package thread_test;

import java.util.Arrays;

/**
 * @author ztyOperation
 * @create 2023-04-29 22:54
 * (1)kmpTest是将当前字母作为了前后缀的一部分，如abcabcm ，next[c] = 3 而kmpTest2 next[c] = 2 next[m] = 3
 * (2)readme.md -->i是不往前移动的，最长前后缀是保正结果不错的原因。不是改进速度。是为了防止出现漏的情况，因为你最长前后缀都可以继续比对下去，那如果舍弃最长情况，直接将j拉到前面来，这样会漏掉结果
 * 但如果是aaaaa这样的情况，再这样找最长前后缀会耽误时间，你可以自己推一下，所以有了改进
 */
public class Main {
    public static void main(String[] args) {
//        String str = "AAAAABAAAAAC";
//        String sub = "AAAAAC";
//        String str = "abcabeabcabcmn";
//        String sub = "abcabcmn";
        String str = "abaabaeabaabamn";
        String sub = "abaabamn";
        int[] next = getNext(sub);
        int result = KMP(str, sub, 0);
        System.out.println(Arrays.toString(next));
        System.out.println(result);
    }

    public static int KMP(String str, String sub, int pos){
        if (str == null || sub == null) {
            return -1;
        }
        if ("".equals(str) || "".equals(sub)) {
            return -1;
        }
        char[] arrStr = str.toCharArray();
        char[] arrSub = sub.toCharArray();
        int lenStr = str.length();
        int lenSub = sub.length();
        int i = pos, j = 0;
        int[] next = getNext(sub);
        while (i<lenStr && j < lenSub) {//kmp最优处就是i不往回移动,只有j移动
            //j=-1表示sub第一个字符都不匹配。这时i++就是从下一个字符开始匹配。前面i-1项不管了。
            if (j == -1 || arrStr[i] == arrSub[j]) {
                i++;
                j++;
            }else{
                j = next[j];
            }
        }//while
        if (j == lenSub) {//完全匹配
            return i - j;
        }
        return -1;
    }

    /**
     * 获取nextVal,改良版
     * @param sub
     * @return
     */
    public static int[] getNextVal(String sub){
        int[] next = new int[sub.length()];
        char[] arr = sub.toCharArray();
        next[0] = -1;
        int j= 0;
        int k = -1;
        while(j<sub.length() - 1){// 因为j要++,也就是对于整个字符串也要获取next[tt.length]因此要注意长度
            if(k == -1 || arr[j] == arr[k]){
                j++;k++;
                if(arr[j] == arr[k]){
                    next[j] = next[k];
                }else{
                    next[j] = k;
                }
            }else{
                k = next[k];
            }
        }
        return next;
    }

    /**
     * 获取next
     * @param sub
     * @return
     */
    public static int[] getNext(String sub){
        int[] next = new int[sub.length() + 1];
        char[] arr = sub.toCharArray();
        next[0] = -1;
        int i = 0;
        int k = -1;
        // next[1] 必然等于 0。 判断条件必须要有 -1 ，不然外面 源串 与 目标串 之间的 比较逻辑会出现问题。 j 无法等于 -1 。
        while(i<sub.length()){
            if(k == -1 || arr[i] == arr[k]){
                i++;k++;
                next[i] = k;
            }else{
                k = next[k];
            }
        }
        return next;
    }
}
```

### 试题二: 斤斤计较的小Z

> ![image.png](https://pub-ccd5622eabb54b4a81678003778ddce1.r2.dev/PicGo/8bbb926572b47eef57105addc8fe335996db64ad326c12d38547948833de2f0d.png)

#### 解法一：KMP算法

**【思路分析】**

> 如果不是用kmp，时间复杂度则为O(n*m) = 10^12，使用kmp 时间复杂度则为O(n+m) = 10^6

```java
import java.util.Scanner;

public class Main {
	static Scanner scanner = new Scanner(System.in);
	static int[] next = null;
	static int cnt = 0;
	public static void main(String[] args) {
		String t = scanner.nextLine();
		String s = scanner.nextLine();
		int i = 0;
		int lenT = t.length();
		int lenS = s.length();
		next = new int[lenT + 2]; // 保证不超就行，你加多少都行
		getNext(t);
		KMP(s, t);
		System.out.println(cnt);
		scanner.close();
	}

	public static int KMP(String s, String t) {
		//AAALAA , AA 应该是3次,但是这道题不考虑也行.我们下面是考虑了的。也能AC
		if (s == null || t == null)
			return -1;
		if ("".equals(s) || "".equals(t))
			return -1;
		char[] ss = s.toCharArray();
		char[] tt = t.toCharArray();
		int lenS = s.length();
		int lenT = t.length();
		int i = 0, j = 0;
		while (i < lenS && j < lenT) {
			if (j == -1 || ss[i] == tt[j]) {
				i++;
				j++;
			} else {
				j = next[j];// 只移动t字符串
			}
			// 每匹配成功一次就加一
			if (j == lenT) {
				cnt++;
				j = 0;
                i = i - next[lenT];//相当于减去整个字符串t的最大前后缀长度。
                //如s = LQCLQCLQ , t = LQCLQ时，next[lenT] = next[5] = 2
                //i = 5 - 2
			}
		}
		

		return -1;// 没有找到
	}

	public static void getNext(String t) {
		next[0] = -1;
		int k = -1, j = 0;
		char[] tt = t.toCharArray();
		while (j < tt.length) {// 因为j要++,也就是对于整个字符串也要获取next[tt.length]因此要注意长度
			if (k == -1 || tt[k] == tt[j]) {// k == -1必须在前面，不然要负数越界
				k++;
				j++;
				//AAA保证next[1]不是-1，我们必须要使用原始版
				next[j] = k;
				// 这是改良版KMP
//				if (tt[k] != tt[j]) {
//					next[j] = k;
//				} else {
//					next[j] = next[k];
//				}
			} else {
				k = next[k];
			}
		}
	}
	
	
}
```

#### 解法三：字符串Hash

**【思路分析】**

> 使用字符串Hash来解决

```java

import java.lang.annotation.Target;
import java.util.Scanner;

public class Main {
	static Scanner scanner = new Scanner(System.in);
	static int[] next = null;
	static int cnt = 0;

	public static void main(String[] args) {
		char[] t = scanner.nextLine().toCharArray();
		char[] s = scanner.nextLine().toCharArray();
		int lenT = t.length;
		int lenS = s.length;
		long[] h = new long[lenS + 1];
		long[] b = new long[lenS + 1];
		long base = 2;// prime
		long target = 0;// target hash
		for (int i = 0; i < lenT; i++) {
			// 只要res不爆掉就行
			target = target * base + (t[i] - 'A' + 1);
		}
		b[0] = 1;
		for (int i = 1; i <= lenS; i++) {
			h[i] = h[i - 1] * base + s[i - 1] - 'A' + 1;
			b[i] = b[i - 1] * base;
		}
		//计算结果
		int count = 0; 
        for (int l = 0, r = l + lenT; r <= lenS; l++, r++) { //LR
        	//使用双指针，确保h[r]与h[l]始终相差3位
        	//b[lenT] 对于base ,我们只需要计算计算 base 的目标字串 n 长度次幂就行，都不用设置数组，是一个固定值
            long sum = h[r] - h[l] * b[lenT];
            if (sum == target) count++;
        }
        
		System.out.println(count);
	}
}
```
