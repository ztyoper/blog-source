### 试题1：

> ![image.png](https://pub-ccd5622eabb54b4a81678003778ddce1.r2.dev/PicGo/7c4dfe6fe1b35d75a719e940c2169325551eecf2e88f5bed0ea037a357b04b38.png)

【思路分析】

> ![tmp2F9E.png](https://pub-ccd5622eabb54b4a81678003778ddce1.r2.dev/PicGo/7b1b699319708e48cb09fe0cdce078e7304d4b65822ba98405f816c3e4c37e75.png)
> 上面应该是(k - f[i-1] ) / k

```java
import java.io.IOException;
import java.util.Scanner;

public class Main {
	static Scanner scanner = new Scanner(System.in);
	public static void main(String[] args) throws NumberFormatException, IOException {
		int n = scanner.nextInt();
		int k = scanner.nextInt();
		double[] f = new double[n+1];
		f[1] = 1;
		for(int i = 2; i<=n ;i++) {
			f[i] = f[i-1] + (k-f[i-1]) / k;
		}
		System.out.printf("%.6f\n", f[n]);
	}
	
}
```

```java
import java.io.IOException;
import java.util.Scanner;

public class Main {
	static Scanner scanner = new Scanner(System.in);
	public static void main(String[] args) throws NumberFormatException, IOException {
		int n = scanner.nextInt();
		int k = scanner.nextInt();
		double[] f = new double[n+1];
		f[1] = 1;
		for(int i = 2; i<=k ;i++) {
			f[i] = f[i-1] + k * 1.0 / (k - i + 1);
		}
		System.out.printf("%.6f\n", f[n]);
	}
	
}
```
