### 1、添加购物车

#### 1、Controller

```java
@RestController
@Slf4j
@Api(tags = "C端购物车管理")
@RequestMapping("/user/shoppingCart")
public class ShoppingCartController {

    @Autowired
    private ShoppingCartService shoppingCartService;

    /**
     * 添加购物菜品（套餐）到购物车中,传入的数据
     *
     * @param shoppingCartDTO
     * @return
     */
    @PostMapping("/add")
    @ApiOperation("添加购物菜品（套餐）到购物车中")
    public Result save(@RequestBody ShoppingCartDTO shoppingCartDTO) {
        log.info("添加购物菜品（套餐）到购物车中, 菜品（套餐）信息：{}", shoppingCartDTO);

        shoppingCartService.save(shoppingCartDTO);

        return Result.success();

    }
}
```

#### 2、Service

```java
@Service
public class ShoppingCartServiceImpl implements ShoppingCartService {

    @Autowired
    private DishMapper dishMapper;

    @Autowired
    private SetmealMapper setmealMapper;
    @Autowired
    private ShoppingCartMapper shoppingCartMapper;

    /**
     * 添加购物菜品（套餐）到购物车中
     *
     * @param shoppingCartDTO
     */
    @Override
    public void save(ShoppingCartDTO shoppingCartDTO) {
        //判断用户的购物车中是否存在该菜品(套餐)
        ShoppingCart shoppingCart = new ShoppingCart();
        BeanUtils.copyProperties(shoppingCartDTO, shoppingCart);
        Long userId = BaseContext.getCurrentId();
        shoppingCart.setUserId(userId);

        //根据条件获取指定菜品(套餐)， 其实这里只会根据userId和dishId获取到一个值,因此后面是list.get(0)
        List<ShoppingCart> list = shoppingCartMapper.list(shoppingCart);
        //如果购物车中存在该菜品（套餐）则数量加一
        if (list != null && list.size() > 0) {
            //list中只可能有一个成员，userId , dishId , setmealId dishFlavor不同就是不同的两个成员。
            shoppingCart = list.get(0);
            shoppingCart.setNumber(shoppingCart.getNumber() + 1);
            // 修改购物车中该菜品（套餐）的数量
            shoppingCartMapper.update(shoppingCart);
            return;
        }
        Long dishId = shoppingCartDTO.getDishId();
        Long setmealId = shoppingCartDTO.getSetmealId();
        //如果购物车中不存在该菜品（套餐）则添加到购物车中
        //如果添加的是菜品
        if (dishId != null) {
            DishVO dishVO = dishMapper.getByIdWithCategoryName(dishId);
            String name = dishVO.getName();
            String image = dishVO.getImage();
            BigDecimal price = dishVO.getPrice();

            shoppingCart.setDishFlavor(shoppingCartDTO.getDishFlavor());
            shoppingCart.setName(name);
            shoppingCart.setImage(image);
            shoppingCart.setAmount(price);
            //如果添加的是套餐
        } else if (setmealId != null) {
            SetmealVO setmealVO = setmealMapper.getBySetmealId(setmealId);
            String name = setmealVO.getName();
            String image = setmealVO.getImage();
            BigDecimal price = setmealVO.getPrice();

            shoppingCart.setName(name);
            shoppingCart.setImage(image);
            shoppingCart.setAmount(price);
        }
        shoppingCart.setCreateTime(LocalDateTime.now());
        shoppingCart.setNumber(1);
        shoppingCart.setUserId(userId);

        // 插入到购物车中
        shoppingCartMapper.insert(shoppingCart);

    }
}
```

#### 3、Mapper

```java
@Mapper
public interface ShoppingCartMapper {
    /**
     * 获取当前用户购物车内的商品信息
     * @return
     */
    List<ShoppingCart> list(ShoppingCart shoppingCart);
    /**
     * 插入到购物车中
     * @param shoppingCart
     */
    @Insert("insert into shopping_cart (name, image, user_id, dish_id, setmeal_id, dish_flavor, amount, create_time)" +
            "values" +
            "(#{name}, #{image}, #{userId}, #{dishId}, #{setmealId}, #{dishFlavor}, #{amount}, #{createTime})")
    void insert(ShoppingCart shoppingCart);

    /**
     * 修改购物车中该菜品（套餐）的数量
     * @param shoppingCart
     */
    void update(ShoppingCart shoppingCart);
}
```
