##### 3、AliOssUtil

```java
@Data
@AllArgsConstructor
@Slf4j
public class AliOssUtil {

    private String endpoint;
    private String accessKeyId;
    private String accessKeySecret;
    private String bucketName;

    /**
     * 文件上传
     *
     * @param bytes
     * @param objectName
     * @return
     */
    public String upload(byte[] bytes, String objectName) {

        // 创建OSSClient实例。
        OSS ossClient = new OSSClientBuilder().build(endpoint, accessKeyId, accessKeySecret);

        try {
            // 创建PutObject请求。
            ossClient.putObject(bucketName, objectName, new ByteArrayInputStream(bytes));
        } catch (OSSException oe) {
            System.out.println("Caught an OSSException, which means your request made it to OSS, "
                    + "but was rejected with an error response for some reason.");
            System.out.println("Error Message:" + oe.getErrorMessage());
            System.out.println("Error Code:" + oe.getErrorCode());
            System.out.println("Request ID:" + oe.getRequestId());
            System.out.println("Host ID:" + oe.getHostId());
        } catch (ClientException ce) {
            System.out.println("Caught an ClientException, which means the client encountered "
                    + "a serious internal problem while trying to communicate with OSS, "
                    + "such as not being able to access the network.");
            System.out.println("Error Message:" + ce.getMessage());
        } finally {
            if (ossClient != null) {
                ossClient.shutdown();
            }
        }

        //文件访问路径规则 https://BucketName.Endpoint/ObjectName
        StringBuilder stringBuilder = new StringBuilder("https://");
        stringBuilder
                .append(bucketName)
                .append(".")
                .append(endpoint)
                .append("/")
                .append(objectName);

        log.info("文件上传到:{}", stringBuilder.toString());

        return stringBuilder.toString();
    }
}
```

> 这时固定写法，作用是将文件传输到对应的阿里云Oss上，并返回存储的文件的url地址。

```java
@RestController
@Slf4j
@Api(tags = "通用接口")
@RequestMapping("/admin/common")
public class CommonController {

    @Autowired
    private AliOssUtil aliOssUtil;
    /**
     * 文件上传到本地阿里云OSS
     *
     * @param file
     * @return
     */
    @ApiOperation("文件上传到本地阿里云OSS")
    @PostMapping("/upload")
    public Result<String> uploadFileToAliOss(MultipartFile file) {
        try {
            log.info("文件上传到本地阿里云OSS....");
            //获取上传的文件的文件名
            String fileName = file.getOriginalFilename();
            //处理文件重名问题
            String postfixName = fileName.substring(fileName.lastIndexOf("."));
            fileName = UUID.randomUUID().toString() + postfixName;

            String url = aliOssUtil.upload(file.getBytes(), fileName);

            return Result.success(url);
        } catch (IOException e) {
            //给后端看的
            log.error("文件上传失败：{}", e.getMessage());
        }
        //给前端看的
        return Result.error(MessageConstant.UPLOAD_FAILED);
    }
}
```

> 为了让浏览器能够正常获取到图片，你应该确保图片可以通过 HTTP 或 HTTPS 协议访问到。而不是使用 file:// 协议，因为浏览器通常不允许通过本地文件路径直接访问资源。

```java
/**
     * 文件上传到本地
     *
     * @return
     */
    @ApiOperation("文件上传到本地")
    @PostMapping("/upload")
    public Result<String> uploadFileToLocal(@RequestParam("file") MultipartFile photo) throws IOException {
        //获取上传的文件的文件名
        String fileName = photo.getOriginalFilename();
        //处理文件重名问题
        String hzName = fileName.substring(fileName.lastIndexOf("."));
        fileName = UUID.randomUUID().toString() + hzName;
        //获取服务器中photo目录的路径
        String folderPath = new File("files").getAbsolutePath();

        File file = new File(folderPath);

        if (!file.exists()) {
            file.mkdir();
        }
        Path filePath = Paths.get(folderPath, fileName);
        String finalPath = filePath.toString();
        //实现上传功能
        photo.transferTo(new File(finalPath));

//        String replace = finalPath.replace('a' , '\/');
        return Result.success(finalPath);
    }
```

### 3、新增菜品提交接口

> 新增菜品提交接口(要有回显功能，数据库保存存储图片的url,提交成功，需要回显图片)

#### 1、Controller

```java
@Api(tags = "菜品管理")
@RestController
@Slf4j
@RequestMapping("/admin/dish")
public class DishController {

    @Autowired
    private DishService dishService;

    /**
     * 添加菜品
     * @param dishDTO
     * @return
     */
    @PostMapping
    public Result save(@RequestBody  DishDTO dishDTO){
        log.info("添加菜品：{}",dishDTO);
        dishService.save(dishDTO);
        return Result.success();
    }
}
```

#### 2、Service

```java
@Service
public class DishServiceImpl implements DishService {
    @Autowired
    DishMapper dishMapper;

    @Autowired
    DishFlavorMapper dishFlavorMapper;
    /**
     * 添加菜品
     * @param dishDTO
     * @return
     */

    // 像这个要操作两张表的数据，一定不能忘了事物注解，
    // 而且入口程序上@EnableTransactionManagement //开启注解方式的事务管理
    @Transactional
    public void save(DishDTO dishDTO) {
       //将该菜品添加到菜品表中
        Dish dish = new Dish();
        //我们不需要添加dishDTO中的List<Flavor> flavors属性
        BeanUtils.copyProperties(dishDTO, dish);
        dishMapper.insert(dish);

        //获取insert语句生成的主键值
        //useGeneratedKeys="true" keyProperty="id"
        //要使用生成的id,存储在id属性中。这样就不用去数据库查保存的这条Dish的id了。
        Long dishId = dish.getId();

        List<DishFlavor> flavors = dishDTO.getFlavors();
        if(flavors != null && flavors.size() > 0){
            //给这几个flavor的dishId赋值
            flavors.forEach(dishFlavor -> dishFlavor.setDishId(dishId));
        }
        //将该菜品对应的几种口味添加到口味表中
        dishFlavorMapper.insertBatch(flavors);
    }
}
```


```java

        <foreach collection="flavors" item="f" separator=",">
            (#{f.dishId}, #{f.name}, #{f.value})
        </foreach>
    </insert>
```
