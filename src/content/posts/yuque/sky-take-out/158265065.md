# 缓存菜品

## 1、问题说明

![tmpA008.png](https://pub-ccd5622eabb54b4a81678003778ddce1.r2.dev/PicGo/ccac0787cba87c740ea53e26967e3d2970f6b147ee88c39ba195d87bc6eaa34f.png)

> 我们获取不同分类的菜品列表是向服务端发送请求。每次切换分类都要重新发一次请求给服务端来获取菜品列表。如果人多。那么就卡。我们就利用缓存来解决这一问题。

## 2、实现思路

![tmpD.png](https://pub-ccd5622eabb54b4a81678003778ddce1.r2.dev/PicGo/fe0773eb2348be3a8057b0b63a9ae98c681211c469da2fa9ddfee04f83e52904.png)

## 3、代码开发

### 1、实现缓存

#### 1、Controller

```java
@RestController("userDishController")
@RequestMapping("/user/dish")
@Slf4j
@Api(tags = "C端-菜品浏览接口")
public class DishController {
    @Autowired
    private DishService dishService;

    @Autowired
    private RedisTemplate<Object, Object> redisTemplate;

    /**
     * 根据分类id查询菜品
     *
     * @param categoryId
     * @return
     */
    @GetMapping("/list")
    @ApiOperation("根据分类id查询菜品")
    public Result<List<DishVO>> list(Long categoryId) {
        Dish dish = new Dish();
        dish.setCategoryId(categoryId);
        dish.setStatus(StatusConstant.ENABLE);//查询起售中的菜品

        //构造redis中的Key,规则：dish_分类id
        String key = "dish_"+categoryId;

        //判断redis中是否有菜品的缓存
        List<DishVO> list = (List<DishVO>) redisTemplate.opsForValue().get(key);
        //如果有，那么直接取缓存中的数据
        if(list != null && list.size() > 0){
            return Result.success(list);
        }
        //如果没有，那么查询，再缓存到redis中
        list = dishService.listWithFlavor(dish);

        redisTemplate.opsForValue().set(key,list);

        return Result.success(list);
    }

}
```

#### 2、RedisConfiguration

```java
@Configuration
@Slf4j
public class RedisConfiguration {

    @Bean
    public RedisTemplate redisTemplate(RedisConnectionFactory redisConnectionFactory){
        log.info("开始创建redis模板类...");
        RedisTemplate<Object, Object> redisTemplate = new RedisTemplate<>();
        //设置redis的连接工厂对象
        redisTemplate.setConnectionFactory(redisConnectionFactory);
        //设置key的序列化器，默认为JdkSerializationRedisSerializer
        redisTemplate.setKeySerializer(new StringRedisSerializer());
        //尚硅谷那个json转换器不行。
        Jackson2JsonRedisSerializer<List> valueSerializer = new Jackson2JsonRedisSerializer<>(List.class);

        //设置value的序列化方式
        //这样可以处理LocalDateTime类型的对象
        JacksonObjectMapper objectMapper = new JacksonObjectMapper();
        valueSerializer.setObjectMapper(objectMapper);
        redisTemplate.setValueSerializer(valueSerializer);

        return redisTemplate;

    }
}
```

> 发现一个bug,就是反序列化value得到店铺状态时会报错，因为不是List.class而是Integer.class。所以对于Value最好使用默认的序列化器。如果真想将某个分类的多有菜品都已json格式存储到redis中，而又不影响店铺状态，可以使用两个redistemplate来进行数据缓存。

### 2、清除缓存

```java
@Api(tags = "菜品管理")
@RestController
@Slf4j
@RequestMapping("/admin/dish")
public class DishController {


    @Autowired
    private DishService dishService;


    @Autowired
    private RedisTemplate<Object, Object> redisTemplate;

    /**
     * 添加菜品
     * @param dishDTO
     * @return
     */
    @PostMapping
    @ApiOperation("新增菜品")
    public Result save(@RequestBody  DishDTO dishDTO){
        log.info("添加菜品：{}",dishDTO);
        dishService.save(dishDTO);

        //清除当前分类的菜品的缓存
        //新增不需要进行清除操作，因为新增后默认是禁用，不会出现在微信列表。启用时清除就行了。
//        Long categoryId = dishDTO.getCategoryId();
//        cleanRedisCache("dish_"+categoryId);

        return Result.success();
    }

    /**
     * 分页查询菜品
     * @param dishPageQueryDTO
     * @return
     */
    @GetMapping("/page")
    @ApiOperation("菜品分页查询")
    public Result<PageResult> pageQuery(DishPageQueryDTO dishPageQueryDTO){
        log.info("进入菜品分页查询:{}", dishPageQueryDTO);
        PageResult result = dishService.pageQuery(dishPageQueryDTO);
        return Result.success(result);
    }

    /**
     * 批量删除菜品
     * @param dishIds
     * @return
     */
    @DeleteMapping
    @ApiOperation("删除菜品")
    public Result delete(@RequestParam("ids") List<Long> dishIds){//接收1,2,3
        log.info("开始删除菜品：{}", dishIds);
        dishService.deleteBatch(dishIds);

        //清除所有缓存
        //删除时不需要清除操作。因为删除前必须将菜品禁用，禁用后就已经不在微信列表中了。

//        cleanRedisCache("dish_*");

        return Result.success();
    }

    /**
     * 根据dishId查询菜品
     * @return
     */
    @GetMapping("/{id}")
    @ApiOperation("根据dishId查询菜品")
    public Result<DishVO> getByIdWithFlavors(@PathVariable("id") Long  dishId){
        log.info("根据dishId查询菜品:{}", dishId);
        DishVO dishVO  = dishService.getByIdWithFlavors(dishId);
        return Result.success(dishVO);
    }

    /**
     * 修改菜品
     * @param dishDTO
     * @return
     */
    @PutMapping
    @ApiOperation("修改菜品")
    public Result update(@RequestBody  DishDTO dishDTO){
        log.info("修改菜品:{}", dishDTO);
        dishService.update(dishDTO);

        //清除所有缓存
        //因为修改，可能会修改所属的分类，这样就要涉及多个分类。我们直接清除全部缓存
        cleanRedisCache("dish_*");

        return Result.success();
    }

    /**
     * 根据categoryId查询菜品不要带上口味
     * @return
     */
    @GetMapping("/list")
    @ApiOperation("根据dishId查询菜品不要带上口味")
    public Result<List<Dish>> getByCategoryIdWithoutFlavors(Long categoryId){
        log.info("根据dishId查询菜品不要带上口味:{}", categoryId);
        List<Dish> dishes  = dishService.getByCategoryIdWithoutFlavors(categoryId);
        return Result.success(dishes);
    }

    /**
     * 启售，停售菜品
     * @param status
     * @param id
     * @return
     */
    @PostMapping("/status/{status}")
    @ApiOperation("启售，停售菜品")
    public Result startOrStopDish(@PathVariable Integer status , Long id ){
        log.info("启售，停售菜品:{},{}",status , id);
        dishService.startOrStopDish(status, id);

        //根据dishId找到对应的分类i，然后精确清除该分类的菜品缓存
        Long categoryId =  dishService.getCategoryIdByDishId(id);

        cleanRedisCache("dish_" + categoryId);

        return Result.success();
    }

    private void cleanRedisCache(String pattern){
        //查询缓存
        Set<Object> keys = redisTemplate.keys(pattern);
        //删除缓存
        redisTemplate.delete(keys);
    }
}
```
