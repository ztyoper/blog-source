#include <stdio.h>
#include <cuda_runtime.h>
//#include "./tools/common.cuh"

__global__ void addFromGPU(int *A, int *B, int *C, const int WIDTH, const int HEIGHT)
{
    int col = blockIdx.x * blockDim.x + threadIdx.x; // 矩阵中的位置col
    int row = blockIdx.y * blockDim.y + threadIdx.y; // 矩阵中的位置row
    int id = row * WIDTH + col;
    if(row<HEIGHT && col<WIDTH){
    C[id] = A[id] + B[id];
    }
    //printf("row=%d\tcol=%d\tresult=%d\n",row,col,C[id]);
}

int main(void)
{
    float elapsedTime;
    cudaEvent_t start, stop;

    // 初始化 CUDA 事件
    cudaEventCreate(&start);
    cudaEventCreate(&stop);

    // 记录开始事件
    cudaEventRecord(start, 0);

    // 在这里插入您的 CUDA 代码
    // 例如，您可以在这里调用一个 CUDA 核函数

    // 1、设置GPU设备
    //setGPU();

    // 2、分配主机内存和设备内存，并初始化
    // 设置元素数量
    int col = 2048;
    int row = 1024;
    int byteCount = col * row * sizeof(int); // 字节数

    // （1）分配主机内存，并初始化
    int *host_A, *host_B, *host_C;
    host_A = (int *)malloc(byteCount);
    host_B = (int *)malloc(byteCount);
    host_C = (int *)malloc(byteCount);

    if (host_A != NULL && host_B != NULL && host_C != NULL)
    {
        memset(host_C, 0, byteCount);
        // 初始化矩阵
        for (int i = 0; i < row*col; i++)
        {
            
                host_A[i] = 0; // 学号倒数第1位
                host_B[i] = 1; // 学号倒数第2位
      
        }
    }
    else
    {
        printf("主机内存分配失败\n");
        exit(-1);
    }
    // （2）分配设备内存，并初始化
    int *device_A, *device_B, *device_C;
    // 以行为单位分配内存
    cudaMalloc((void **)&device_A, byteCount);
    cudaMalloc((void **)&device_B, byteCount);
    cudaMalloc((void **)&device_C, byteCount);
    if (device_A != NULL && device_B != NULL && device_C != NULL)
    {
        cudaMemset(device_A, 0, byteCount); // 设备内存初始化为0
        cudaMemset(device_B, 0, byteCount);
        cudaMemset(device_C, 0, byteCount);
    }
    else
    {
        printf("设备内存分配失败\n");
        free(host_A);
        free(host_B);
        free(host_C);
        exit(-1);
    }

    // 4、数据从主机复制到设备
    cudaMemcpy(device_A, host_A, byteCount, cudaMemcpyHostToDevice);
    cudaMemcpy(device_B, host_B, byteCount, cudaMemcpyHostToDevice);
    cudaMemcpy(device_C, host_C, byteCount, cudaMemcpyHostToDevice);

    // 5、调用核函数在设备中进行计算
    dim3 block_size(16, 16, 1);
    dim3 grid_size((col - 1) / 16 + 1, (row - 1) / 16 + 1,1);
    //dim3 grid_size(128,64,1);
    // 调用核函数
    addFromGPU<<<grid_size, block_size>>>(device_A, device_B, device_C, col, row);
    // cudaDeviceSynchronize();

    

    // 6、将计算得到的数据从设备传给主机
    cudaMemcpy(host_C, device_C, byteCount, cudaMemcpyDeviceToHost); // cudaMemcpy自带cudaDeviceSynchronize，所以可以不需要上面一行cudaDeviceSynchronize();


    for (int i = 0; i < row; i++) // 打印
    {
        for (int j = 0; j < col; j++)
        {
            printf("%d\t",host_C[i*col + j]);
        }
        printf("\n");
    }


    // 7、释放主机与设备内存
    free(host_A);
    free(host_B);
    free(host_C);
    cudaFree(device_A);
    cudaFree(device_B);
    cudaFree(device_C);

    // 记录结束事件
    cudaEventRecord(stop, 0);

    // 同步事件，确保所有任务完成
    cudaEventSynchronize(stop);

    // 计算经过的时间
    cudaEventElapsedTime(&elapsedTime, start, stop);
    
    // 销毁事件
    cudaEventDestroy(start);
    cudaEventDestroy(stop);

    cudaDeviceReset();//这句话只能在最后，不然会出问题

    printf("Elapsed time: %3.1f ms\n", elapsedTime);

    return 0;
}

__global__ void addFromGPU(size_t *A, size_t *B, size_t *C,  size_t WIDTH,  size_t  HEIGHT, size_t COUNT)
{

    size_t col = blockIdx.x * blockDim.x + threadIdx.x; // 结果矩阵中的位置col
    size_t row = blockIdx.y * blockDim.y + threadIdx.y; // 结果矩阵中的位置row
    size_t id = row * WIDTH + col;
    if (row < HEIGHT && col < WIDTH)//heightB = 511，row最大510
    {
        size_t pValue = 0;
        for (int k = 0; k < COUNT; k++) // WIDTH水平宽度，两个矩阵相乘，满足矩阵乘法法则
        {
            pValue += A[row * COUNT + k] * B[k * WIDTH + col];
        }
        C[id] = pValue;
    }
}
void printResult(size_t *host_C, int row, int col)
{
    for (int i = 0; i < row; i++) // 打印
    {
        for (int j = 0; j < col; j++)
        {
            printf("%ld\t", host_C[i * col + j]);
        }
        
        printf("\n");
    }
}

> CPU与GPU的加法没有难度
> CPU与GPU的乘法需要注意的是：
> - 分配GPU线程时，是对两个矩阵相乘的结果矩阵进行分配线程,grid_size ,block_size都按照结果矩阵来决定。
> - 当前代码是可以实现两个矩阵相乘，不管是A*B还是B*A
> - 使用一维数组来代替二维数组，二维数组的GPU显存分配太难了，不会，就只能使用一维的来代替。
