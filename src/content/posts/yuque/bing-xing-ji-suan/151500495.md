# 6.并行程序开发

### 6.1 n体问题的两种解决方法

#### 6.1.1 问题描述

> n个天体知道空间中的`初始位置向量和速度向量和对应的质量`，求`某一时刻`各个天体的`位置向量和速度向量`

$$
F_{qk} = -\frac{Gm_qm_k}{|s_q(t)-s_k(t)|^3}(s_q(t)-s_k(t))\\
k粒子对q粒子的作用力
$$

$F_{q} =\sum_{k=0\ k!=q}^{n-1}{F_{qk}}= -Gm_q\sum_{k=0\ k!=q}^{n-1}\frac{m_k}{|s_q(t)-s_k(t)|^3}(s_q(t)-s_k(t))$

$$
F_{q} = m_q*a_q\\

a_q = s_q^{''}二阶导数
$$

$s_{q}(t)^{''} = -G\sum_{k=0\ k!=q}^{n-1}\frac{m_k}{|s_q(t)-s_k(t)|^3}(s_q(t)-s_k(t))$

> 总结：根据天体相互作用力，获取q天体收到的合力
> 然后$F = ma$得到加速度，从而得到某一时刻的位置向量和速度向量

#### 6.1.2 两个串行代码

```c
伪代码
Get input data:
for 每一个时间步{
    if(这个时间步结束)输出想要的粒子的位置和速度
    for 每一个粒子{
        算出合力
    }
    for 每一个粒子{
        算出粒子的位置和速度
    }
}
```

```c
算出合力
for q粒子{
    for k粒子 != q粒子{
        x_diff = pos[q][X] - pos[k][X];
        y_diff = pos[q][Y] - pos[k][Y];
        dist = sqrt(x_dff*x_diff + y_diff*y_diff);
        dist_cube = dist*dist*dist;
        //因为作用力符号是符号，求和就是-=
        forces[q][X] -=G*masses[q]*masses[k]/dist_cube * x_diff;//x方向分量
        forces[q][Y] -=G*masses[q]*masses[k]/dist_cube * x_diff;//y方向分量

    }
}
```

```c
算出合力
for q粒子{
    for k粒子 > q粒子{
        //q = 1 ,k =2 ,3, ... n
        //q = 2, k =3, 4, ... n 不用算F21,因为F21 = -F12,力是相互的
        x_diff = pos[q][X] - pos[k][X];
        y_diff = pos[q][Y] - pos[k][Y];
        dist = sqrt(x_dff*x_diff + y_diff*y_diff);
        dist_cube = dist*dist*dist;
        //因为作用力符号是符号，求和就是‘-=’
        forces_qk[X] =-G*masses[q]*masses[k]/dist_cube * x_diff;//x方向分量
        //k粒子对q粒子的作用力
        forces_qk[Y] =-G*masses[q]*masses[k]/dist_cube * x_diff;//y方向分量

        forces[q][X] +=force_qk[X];
        forces[q][X] +=force_qk[Y];
        forces[k][X] -=force_qk[X];
        forces[k][X] -=force_qk[Y];
    }
}
```

```c
算出粒子的位置和速度
已经知道了当前时间所有粒子分别收到的合力F。
根据F= ma可以只当当前时间片的加速度。
delta_t趋近0时，速度变化量就是0，位置就是s = s + delta_t*vel
delta_t趋近0时，加速度变化量就是定值a = f/m，速度就是vel = vel + delta_t*f/m
for q粒子{
    pos[q][X] +=delta_t*vel[q][X];
    pos[q][Y] +=delta_t*vel[q][Y];

    vel[q][X] +=delta_t/masses[q]*forces[q][X];
    vel[q][Y] +=delta_t/masses[q]*forces[q][Y];
}
```

#### 6.1.3 用openmp实现基本算法代码：

```c
伪代码
Get input data:
for 每一个时间步{
    if(这个时间步结束)输出想要的粒子的位置和速度
#pragma omp parallel for
    for 每一个粒子{
        算出合力
    }
    同步一下
#pragma omp parallel for
    for 每一个粒子{
        算出粒子的位置和速度
    }
}
```

> `外层循环`存在`循环依赖`，不能并行。
> 两个`内存循环`都是按照独立的粒子进行迭代，最多对其他粒子的质量，位置有数据依赖，因此可以并行。

```c
伪代码
Get input data:
#pragma omp parallel 
for 每一个时间步{
#pragma omp single
    if(这个时间步结束)输出想要的粒子的位置和速度
#pragma omp for
    for 每一个粒子{
        算出合力
    }
    同步一下
#pragma omp for
    for 每一个粒子{
        算出粒子的位置和速度
    }
}
```

> 只动态生成一次多线程。第二个`#pragma omp for`前有一个隐式的线程同步。
> `#pragma omp single`：只让任意一个线程进行输出操作。`single`指令后也有隐含的同步，但这个同步在这里没有作用，因为没做修改，只做输出。其他线程会等一个线程输出完毕才会向下执行。可以用`nowait`子句取消默认的路障，但这里没有性能提升，因为两个`#pragma omp for`后面都存在路障，取消路障不会多执行多少语句。

> 15行到18行会有竞争条件（例如线程0：q=1,k=2 线程1：q=2,k=3时），解决方法有：
> 1. 使用 `#pragma omp critical{}` ,但性能会下降很多。
> 2. 使用互斥量，要比临界区好点，但不多。
> 3. 添加本地变量


```c
算出合力（阶段二）
for q粒子{
	forces[q][X] = forces[q][Y] = 0.0;

        forces[q][X] +=local_forces[q][X];
```

> 简化算法中求作用力使用`循环划分`（负载更加均衡，因为粒子q越大，所做的工作越少），其他循环使用`块划分`，最终能取得做好的性能。

#### 6.1.5 用openmp实现简化算法最终代码

```c
#include <stdio.h>
#include <stdlib.h>
#include <omp.h>
#include <math.h>
#define G 3.5
typedef struct _plant
{
    double m;
    double s[2];
    double v[2];
} Plant;
void Get_input(int *n_p, Plant *plants)
{
    int i, n;
    printf("请输入总天体数n：");
    scanf("%d", n_p);
    n = *n_p;
    plants = (Plant *)malloc(n * sizeof(Plant));

    for (i = 0; i < n; i++)
    {
        printf("请输入第%d个天体的质量m：", i);
        scanf("%lf", &plants[i].m);
        printf("请输入第%d个天体的位置向量s[0],s[1](,分割)：", i);
        scanf("%lf,%lf", &plants[i].s[0], &plants[i].s[1]);
        printf("请输入第%d个天体的速度向量v[0],v[1](,分割)：", i);
        scanf("%lf,%lf", &plants[i].v[0], &plants[i].v[1]);
    }

    printf("n = %d\n", n);
    printf("天体信息：\n");
    for (i = 0; i < n; i++)
    {
        printf("第%d个天体的质量m：%lf\t", i, plants[i].m);

        printf("位置向量s：(%lf, %lf)\t", plants[i].s[0], plants[i].s[1]);

        printf("速度向量v：(%lf, %lf)\n", plants[i].v[0], plants[i].v[1]);
    }
}
void Print_result(int n, Plant *plants)
{
    int i;
    printf("天体信息：\n");
    for (i = 0; i < n; i++)
    {
        printf("第%d个天体的质量m：%lf\t", i, plants[i].m);

        printf("位置向量s：(%lf, %lf)\t", plants[i].s[0], plants[i].s[1]);

        printf("速度向量v：(%lf, %lf)\n", plants[i].v[0], plants[i].v[1]);
    }
}
int main(int argc, char *argv[])
{
    int n;
    Plant *plants = NULL;
    // Get_input(&n, plants);
    n = 3;
    plants = (Plant *)malloc(n * sizeof(Plant));
    plants[0].m = 1;
    plants[1].m = 2;
    plants[2].m = 3;
    plants[0].s[0] = 1.0;
    plants[0].s[1] = 2.0;
    plants[1].s[0] = 2;
    plants[1].s[1] = 3;
    plants[2].s[0] = 3;
    plants[2].s[1] = 4;

    plants[0].v[0] = 1;
    plants[0].v[1] = 2;
    plants[1].v[0] = 2;
    plants[1].v[1] = 3;
    plants[2].v[0] = 3;
    plants[2].v[1] = 4;

    int thread_count = strtol(argv[1], NULL, 10); // 4

    int delta_t, q, k;
    double x_diff, y_diff;
    double dist, dist_cube;
    double local_forces[thread_count][n][2];
    double forces_qk[2];

    double forces[n][2];
    int thread;
#pragma omp parallel default(none) shared(plants, n, forces, thread_count, delta_t) private(q, k, x_diff, y_diff, dist, dist_cube, forces_qk, local_forces, thread)
    for (delta_t = 1; delta_t < 1000; delta_t++)
    {
#pragma omp single
        {
            // 1.符合所求时间步，输出结果
            if (delta_t == 1)
            {
                printf("时间步为%d时：\n", delta_t);
                Print_result(n, plants);
            }
        }

        // 循环划分
#pragma omp for schedule(static, 1)
        // default(none) shared(plants,G,n) private(q,k,x_diff,y_diff,dist,dist_cube,forces_qk,local_forces)
        // 2.1求合作用力第一阶段
        for (q = 0; q < n; q++)
        {
            for (k = q + 1; k < n; k++)
            {
                // q = 1 ,k =2 ,3, ... n
                // q = 2, k =3, 4, ... n 不用算F21,因为F21 = -F12,力是相互的
                x_diff = plants[q].s[0] - plants[k].s[0];
                y_diff = plants[q].s[1] - plants[k].s[1];
                dist = sqrt(x_diff * x_diff + y_diff * y_diff);
                dist_cube = dist * dist * dist;
                // 因为作用力符号是符号，求和就是‘-=’
                forces_qk[0] = -G * plants[q].m * plants[k].m / dist_cube * x_diff; // x方向分量
                // k粒子对q粒子的作用力
                forces_qk[1] = -G * plants[q].m * plants[k].m / dist_cube * x_diff; // y方向分量

                int my_rank = omp_get_thread_num();
                local_forces[my_rank][q][0] += forces_qk[0];
                local_forces[my_rank][q][1] += forces_qk[1];
                local_forces[my_rank][k][0] -= forces_qk[0];
                local_forces[my_rank][k][1] -= forces_qk[1];
            }
        }

#pragma omp for
        // default(none) shared(n,forces,thread_count) private(q,thread)
        // 2.2求合作用力第二阶段
        for (q = 0; q < n; q++)
        {
            forces[q][0] = forces[q][1] = 0.0;
            for (thread = 0; thread < thread_count; thread++)
            {
                forces[q][0] += local_forces[thread][q][0];
                forces[q][1] += local_forces[thread][q][1];
            }
        }

#pragma omp for
        // default(none) shared(plants,delta_t,forces,n) private(q)
        // 3.求位置向量和速度向量
        for (q = 0; q < n; q++)
        {
            plants[q].s[0] += delta_t * plants[q].v[0];
            plants[q].s[1] += delta_t * plants[q].v[1];

            plants[q].v[0] += delta_t / plants[q].m * forces[q][0];
            plants[q].v[1] += delta_t / plants[q].m * forces[q][1];
        }
    } // 时间步
    return 0;
}
```

> 1. `#pragma omp for` 后面不能有 `shared` 和 `private`
> 2. 不能忘记 `#pragma omp single` 的作用块 `{}`
