# 3.MPI进行分布式内存编程

### 3.1 打印Hello Wolrd

> [https://easyhpc.net/problem/free-coding](https://easyhpc.net/problem/free-coding)

> 1. mpicc -g -Wall -o mpi_hello mpi_hello.c
>    1. -g : 打开调试（产生调试信息）
>    2. -Wall : 打开所有警告
>    3. -o: 设置生成文件名
>    4. 注意：a 和 b是选项，没有参数
> 2. mpiexec -n 4 ./mpi_hello 去执行
>    1. -n 是进程数
> 3. `mpicc` 是 `包装脚本（wrapper script）` ,可以简化编译器的运行,事实上windows上并没有mpicc ，是使用 `g++ 文件名.cpp -o 文件名 -I mpi的Include路径 -L mpi的Lib/x64路径 -lmsmpi` 来代替,就像pthread一样。

```c
#include<stdlib.h>
#include<stdio.h>
#include<mpi.h>
#include<string.h>
const int MAX_STRING = 100;

int main(void){
    char greeting[MAX_STRING];
    int comm_sz;//进程的数量
    int my_rank;//当前进程号，进城号是从0开始的

    MPI_Init(NULL,NULL);
    MPI_Comm_size(MPI_COMM_WORLD, &comm_sz);
    MPI_Comm_rank(MPI_COMM_WORLD, &my_rank);

    if(my_rank  != 0){
        //输出到字符串greeting中
        sprintf(greeting, "Greetings from process %d of %d!",my_rank, comm_sz);
        MPI_Send(greeting,strlen(greeting) + 1,MPI_CHAR,0,0,MPI_COMM_WORLD);

    }else{
        printf("--->Greetring from process %d of %d!-<--\n",my_rank, comm_sz);
        for(int i = 1 ; i < comm_sz; i++){//进程2，3先到，则在缓冲区等待
            MPI_Recv(greeting, MAX_STRING, MPI_CHAR, i,0,MPI_COMM_WORLD, MPI_STATUS_IGNORE);
            printf("%s\n",greeting);
        }
    }
    MPI_Finalize();

}
```

> 1. 通讯子中的每一个进程都进行当前程序，使用的是分布式内存，只能通过消息传递交互。
> 2. 要加 `#include<mpi.h>`
> 3. `mpiexec -n 4 ./mpi_hello` : 4是进程数
> 4. 在MPI程序中，一个 `通信子` 是一组进程的集合，还集合中的进程之间可以相互发送信息。MPI程序启动后，MPI创建的所有进程组成的通信子称为MPI_COMM_WORLD
> 5. 许多并行程序使用 `单程序多数据流（SPMD）` 的方法，通过根据不同的进程号转移到不同的 `分支语句` ，使得运行一个程序就能获得运行多个不同程序的效果。用完MPI后，记得调用MPI_Finalize()；结束程序。

### 3.2 用MPI来实现梯形积分法

![board](https://pub-ccd5622eabb54b4a81678003778ddce1.r2.dev/PicGo/27706e5926532fcb5aced1a3240aed09370f62d706ad8dbc30fab131911903d0.jpg)

$$
梯形面积S = \frac{h}{2}\left[f(x_i) + f(x_{i+1})\right] \\
梯形面积和 = h[f(x_0)/2+f(x_1)+...+f(x_{n-1})+f(x_n)/2]
$$

#### 1.串行程序

```c

h = (b - a) / n;
approx = (f(a) + f(b)) / 2.0;
for(int i = 1; i<= n-1; i++){
    x_i = a + i*h;
    approx +=f(x_i);
}
approx = h*approx;
```

#### 2.并行代码

```c
#include <stdio.h>
#include <stdlib.h>
#include <mpi.h>
#include <math.h>
// 梯形函数
double f(double a)
{
    return a * a;
}
// 梯形部分面积求和
double Trap(double local_a, double local_b, double local_n, double h, int my_rank)
{
    //printf("process = %d , %lf,%lf,%lf,%lf\n", my_rank, local_a, local_b, local_n, h);

    double approx = (f(local_a) + f(local_b)) / 2.0;
    double x_i = 0;
    for (int i = 1; i <= local_n - 1; i++)
    {
        x_i = local_a + i * h;
        approx += f(x_i);
    }
    approx = h * approx;
    //printf("process = %d,approx = %lf\n", my_rank , approx);
    return approx;
}
int main(void)
{
    int my_rank, comm_sz, n = 4, local_n;
    double a = 0.0, b = 4.0, h, local_a, local_b;
    double local_int, total_int;

    MPI_Init(NULL, NULL);
    MPI_Comm_rank(MPI_COMM_WORLD, &my_rank);
    MPI_Comm_size(MPI_COMM_WORLD, &comm_sz);

    h = (b - a) / n;
    local_n = n / comm_sz;
    local_a = a + my_rank * local_n * h;
    local_b = local_a + local_n * h;

    local_int = Trap(local_a, local_b, local_n, h, my_rank);
    // printf("from process  = %d, %lf\n", my_rank, local_int);
    if (my_rank != 0)
    {
        MPI_Send(&local_int, 1, MPI_DOUBLE, 0, 0, MPI_COMM_WORLD);
    }
    else
    {
        total_int = local_int;
        for (int i = 1; i < comm_sz; i++)
        {
            MPI_Recv(&local_int, 1, MPI_DOUBLE, i, 0, MPI_COMM_WORLD, MPI_STATUS_IGNORE);
            printf("from process  = %d, %lf\n", i, local_int);
            total_int += local_int; // 这里使用local_int来接收数据，少定义一个变量
        }
    }
    if (my_rank == 0)
    {
        printf("梯形从%lf到%lf总共分成了%d分，每个线程处理%d份，最终估计结果为：%.15e\n",
               a, b, n, local_n, total_int); //%e是指数输出，5.2e就是长度为5，小数点精确度为2
    }
    MPI_Finalize();
    return 0;
}
```

> 使用MPI_Reduce规约函数进行树形规约，相比第二个版本，不在是每一个进程计算完local_int后发给0号进程，让0号进程相加。而是每个进程会按照树形结构累加local_int。这样效率提升很大。

```c

//集合通信
MPI_Reduce(
    _In_range_(!=, recvbuf) _In_opt_ const void* sendbuf,		//区间数据对应的指针

    _In_range_(>=, 0) int count,								//计算的维度，一般是1
    _In_ MPI_Datatype datatype,									//数据的类型
    _In_ MPI_Op op,												//计算的操作符
    _mpi_coll_rank_(root) int root,						//规约到哪一个进程上，一般是0进程
    _In_ MPI_Comm comm											//通信子
    );


MPI_Bcast(
    _Pre_opt_valid_ void* buffer,       // 数据的指针
    _In_range_(>=, 0) int count,		//当前数据类型的个数
    _In_ MPI_Datatype datatype,			//数据的类型
    _mpi_coll_rank_(root) int root,		//起源进程，一般是0，表示0号进程发起广播
    _In_ MPI_Comm comm					//通信子
    );
//scanf只能0号进程调用，其他进程不能调用，这样规定是为了防止输入错误
```

归约操作符

| MPI_MAX | 求最大值 | MPI_LOR | 逻辑或 |
| --- | --- | --- | --- |
| MPI_MIN | 求最小值 | MPI_BOR | 按位或 |
| MPI_SUM | 求累加和 | MPI_LXOR | 逻辑异或 |
| MPI_PROD | 求累乘积 | MPI_BXOR | 按位异或 |
| MPI_LAND | 逻辑与 | MPI_MAXLOC | 求最大值和最大值所在的位置 |
| MPI_BAND | 按位与 | MPI_MINLOC | 求最小值和最小值所在的位置 |

> 1. MPI_MAX时，global_max初始化为最小
> 2. MPI_MIN时，global_min初始化为最小
> 3. MPI_LAND时，result初始化为1
> 4. MPI_BAND时，result初始化为OxFFFFFFFF
> 5. MPI_LOR时，result初始化为0
> 6. MPI_LXOR时，result初始化为0

### 3.3 向量相加

$$
\begin{align}
&x = (x_0 ,x_1, x_2,...x_{n-1})\\
&y = (y_0,y_1, y_2,...y_{n-1})\\
&x+y = (x_0+y_0, x_1+y_1....)
\end{align}
$$

#### 1.串行代码

```c
//串行代码
void Vector_sum(double x[], double y[] , double z[],int n ){
    int i;
    for(i = 0; i<n; i++){
        z[i] = x[i] + y[i];
    }
}
```

```c
void Build_mpi_type(double *a_p, double *b_p, int* n_p, MPI_Datatype *input_mpi_t_p)
{
    //1.定义MPI_Type_create_struct的三个参数
    int array_of_blocklengths[3] = {1,1,1};//每个数据的长度，因为都是一个数，所以都是1，a如果是数据，那该array_of_blocklengths[0]就是这length(a)
    MPI_Datatype array_of_types[3] = {MPI_DOUBLE,MPI_DOUBLE,MPI_INT};//数据类型
    MPI_Aint a_addr, b_addr, n_addr;//MPI_Aint是一个整型，它的长度足以保存地址
    MPI_Aint array_of_displacements[3] = {0};//地址偏移量，先全部初始化为0
    
    //2.获取地址偏移量
    MPI_Get_address(a_p, &a_addr);
    MPI_Get_address(b_p, &b_addr);
    MPI_Get_address(n_p, &n_addr);
    array_of_displacements[1] = b_addr - a_addr;
    array_of_displacements[2] = n_addr - a_addr;//这里应该是与a_addr之间的偏移量,不能是n_addr - a_addr

    //3. 使用MPI_Type_create_struct创建新的MPI数据类型
    MPI_Type_create_struct(3,array_of_blocklengths,array_of_displacements,array_of_types,input_mpi_t_p);

    //4.将该新的MPI数据类型标记为可以参与通信函数的数据类型。跟MPI_INT，MPI_DOUBLE一样
    MPI_Type_commit(input_mpi_t_p);
}
```

> 注意：
> 1. 如果由p个进程运行并行奇偶交换排序算法，则p个阶段后，输入列表排序完毕。因此phase最大为comm_sz - 1（从0开始）。
> 2. 事实上，不太可能p=n的情况下运行该算法，因为不可能有成百上千个处理器，另外即使有这样的处理器，每一次 `比较-交换` 时发送和接收消息导致的额外消息开销会严重降低程序的运行时间，记住通讯开销远比“局部”计算开销大。

```c
void Merge_smaller(double my_keys[], double recv_keys[], int local_n)
{
    double temp[local_n];    // 临时数组
    int i = 0, j = 0, k = 0; // 分别为my_keys,recv_keys,temp三个数组的指针
    while (k < local_n)      // 这个地方必须是k!!!
    {
        if (my_keys[i] < recv_keys[j])
        {
            temp[k++] = my_keys[i++];
        }
        else
        {
            temp[k++] = recv_keys[j++];
        }
    }
    // 赋值给本地my_keys
    for (i = 0; i < local_n; i++)
    {
        my_keys[i] = temp[i];
```

> 我们总结一下并行排序算法的流程：
> 1. 将各个进程中的数据用串行算法排序
> 2. 分奇偶阶段交换数据，每次交换数据的对象有所不同
> 3. 在交换数据过程中，首先需要进程通信，拿到对方进程的数据
> 4. 编号较小的进程保存两个进程中较小的一半数据，编号较大的进程保存两个进程中较大的一半数据
> 5. 重复奇偶阶段交换数据直至全局有序

### 3.8 MPI程序的安全性（通信）

```c
MPI_Send(my_keys, local_n, MPI_DOUBLE, partner, 0, MPI_COMM_WORLD);
MPI_Recv(recv_keys, local_n, MPI_DOUBLE, partner, 0, MPI_COMM_WORLD, MPI_STATUS_IGNORE);
```

> 1. MPI_Send有两种方式来实现：
>    1. 简单地将消息复制到MPI设置的缓冲区并返回
>    2. 直到对应MPI_Recv出现前都阻塞
> 2. 许多MPI函数都设置了使系统从缓冲到阻塞切换的 `阈值` ，即相对 `较小` 的消息就交由MPI_Send缓冲，但对于 `大型数据` 就选择阻塞模式。
> 3. 问题1：如果每一个进程都阻塞在MPI_Send上，则没有进程去调用MPI_Recv，此时程序就会死锁或挂起。
> 4. 问题2：依赖MPI提供的缓冲机制是 `不安全的` ，这样的程序在运行小的输入集时没有问题，在运行大的输入集时则会导致崩溃或挂起。

> 我们知道，MPI程序的MPI_Send函数可能阻塞也可能缓冲，取决于发送消息的大小（当消息大小大于系统设置的阈值，则阻塞该进程；否则MPI缓冲该消息，调用返回）；在上述程序中，由于每个进程都要收发数据，可能存在某一时刻所有进程都执行send操作，而且由于数据过大都进入了阻塞状态，那么此时的程序就会死锁，因为没有进程可以调用recv函数来接收数据，每个进程都在等待一个不会发生的事情
> 依赖于MPI提供的缓冲机制是不安全的，对于较小的数据可能程序照常运行，但是一旦数据较大就可能出现挂起甚至崩溃的情况
> 怎样修改程序才能使我们的并行排序算法的进程通信安全呢？如果某个进程既需要发送数据又需要接收数据，MPI提供的MPI_Sendrecv方法提供了自己的调度通信的方法：这里不推荐MPI_Ssend，这个函数有问题。

```c

#include <mpi.h>
#include <stdlib.h>
#include <time.h>

int Global_sum(int my_int, int my_rank, int comm_sz, MPI_Comm comm)
{
    // 1.进行树形通信计算全局总和
    int mask = 1;
    while (mask < comm_sz)
    {
        mask <<= 1;
        if (my_rank % mask == 0 && my_rank + mask / 2 < comm_sz)//my_rank + mask / 2 < comm_sz就让非2次幂的线程数可以正确求和
        {
            int partner = my_rank + mask / 2;
            int temp;
            MPI_Recv(&temp, 1, MPI_INT, partner, 0, MPI_COMM_WORLD, MPI_STATUS_IGNORE);
            my_int += temp;
            //printf("receive processing  = %d, partner = %d\n",my_rank , partner);
        }
        else if(my_rank % mask == mask / 2)
        {
            int partner = my_rank - mask / 2;
            MPI_Send(&my_int, 1, MPI_INT, partner, 0, MPI_COMM_WORLD);
            //printf("send processing  = %d, partner = %d\n",my_rank , partner);
        }
        
        MPI_Barrier(comm);
    }
```

```c
#include <stdio.h>
#include <mpi.h>
#include <stdlib.h>
#include <time.h>

int Global_sum(int my_int, int my_rank, int comm_sz, MPI_Comm comm)
{
    int remain = comm_sz, sum = my_int, half, rm, temp;
    printf("%d = %d\n", my_rank, my_int);
    while (remain != 1)
    {
        half = remain / 2;
        rm = remain % 2;
        if (my_rank < half)
        {
            MPI_Recv(&temp, 1, MPI_INT, my_rank + half + rm, 0, comm, MPI_STATUS_IGNORE);
            sum += temp;
        }
        else if (my_rank >= half + rm && my_rank < remain)
        {
            MPI_Send(&sum, 1, MPI_INT, my_rank - half - rm, 0, comm);
        }
        remain = half + rm;
    }

    if (my_rank == 0)
    {
        return sum;
    }
}

int main()
{
    int comm_sz;
```
